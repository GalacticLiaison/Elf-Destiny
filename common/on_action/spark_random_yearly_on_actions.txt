# With lower elf fertility this make sure that elves are at least having some kids
elf_prego_pulse = {
	trigger = {
		root = {
			OR = {
				AND = {
					is_female = yes
					is_pregnant = no
				}
				is_female = no
			}
			age > 16
			is_any_elf_type = yes
			any_child = {
				count = 0
			}
			OR = {
				is_married = yes
				is_concubine = yes
			}
		}
	}
	effect = {
		if = {
			limit = {
				is_female = yes
			}
			if = {
				limit = {
					is_married = yes
				}
				debug_log = "evaluating elf preg encouragment married"
	
				random_consort = {
					save_scope_as = sex_partner
				}
	
				debug_log_scopes = yes
	
				random = {
					chance = 20
					debug_log = "pregnancy given"
					make_pregnant = {
						father = scope:sex_partner
					}
				}
			}
			else_if = {
				limit = {
					is_concubine = yes
					is_concubine_of = liege
				}
				debug_log = "evaluating elf preg encouragment concubine"
				debug_log_scopes = yes
	
				random = {
					chance = 20
					debug_log = "pregnancy given"
					make_pregnant = {
						father = liege
					}
				}
			}
		}
		else = {
			debug_log = "evaluating elf preg encouragment married male perspective"
	
			random_consort = {
				limit = {
					is_pregnant = no
				}
				save_scope_as = sex_partner
			}

			debug_log_scopes = yes

			if = {
				limit = {
					exists = scope:sex_partner
				}
				random = {
					chance = 20
					debug_log = "pregnancy given"
					scope:sex_partner = {
						make_pregnant = {
							father = root
						}
					}
				}
			}

		}

	}
}

random_yearly_playable_pulse = {
	on_actions = {
		set_ai_ruler_aeluran_authority_law_level
	}
}

# This is a hold over solution to on occasion change up the level of aeluran control for AI characters
# - they wont use the newly built UI yet
set_ai_ruler_aeluran_authority_law_level = {
	trigger = {
		faith = {
			has_tenet = tenet_aeluran_sisters
		}
		is_ai = yes
		OR = {
			has_realm_law = aeluran_realm_authority_1
			has_realm_law = aeluran_realm_authority_2
			has_realm_law = aeluran_realm_authority_3
		}
	}
	effect = {
		debug_log = "set_ai_ruler_aeluran_authority_law_level"
		if = {
			limit = {
				has_realm_law = aeluran_realm_authority_1
			}

			random = {
				chance = 5
				add_realm_law = aeluran_realm_authority_2
			}
		}
		else_if = {
			limit = {
				has_realm_law = aeluran_realm_authority_2
			}

			random = {
				chance = 5
				add_realm_law = aeluran_realm_authority_1
			}
			random = {
				chance = 5
				add_realm_law = aeluran_realm_authority_3
			}
		}
		else_if = {
			limit = {
				has_realm_law = aeluran_realm_authority_3
			}

			random = {
				chance = 5
				add_realm_law = aeluran_realm_authority_2
			}
		}
	}
}


elf_wander_pop_control_pulse = {
	trigger = {
		global_var:wandering_enabled = yes
		NOT = { has_character_flag = resisted_wandering }
		NOT = { has_character_modifier = wander_immune }
		
		save_temporary_scope_as = wandering_candidate

		is_any_elf_type = yes
		# is_ai =  yes

		trigger_if = {
			limit = {
				has_global_variable = wandering_age
				global_var:wandering_age = flag:age_range_1
			}
			is_adult = yes
		}
		trigger_else_if = {
			limit = {
				has_global_variable = wandering_age
				global_var:wandering_age = flag:age_range_2
			}
			age > 50
		}
		trigger_else_if = {
			limit = {
				has_global_variable = wandering_age
				global_var:wandering_age = flag:age_range_3
			}
			age > 75
		}
		trigger_else_if = {
			limit = {
				has_global_variable = wandering_age
				global_var:wandering_age = flag:age_range_4
			}
			age > 100
		}

		## FAMILY OF PLAYER ##
		trigger_if = { 
			limit = {
				has_global_variable = wandering_family
				global_var:wandering_family = flag:no_family
			}
			# No Family Excluded
		}
		trigger_else_if = {
			limit = {
				has_global_variable = wandering_family
				global_var:wandering_family = flag:close_family
			}
			NOT = {
				any_player = {
					is_close_family_of = prev
				}
			}
		}
		trigger_else_if = {
			limit = {
				has_global_variable = wandering_family
				global_var:wandering_family = flag:extended_family
			}
			NOT = {
				any_player = {
					is_extended_family_of = prev
				}
			}
		}
		trigger_else_if = {
			limit = {
				has_global_variable = wandering_family
				global_var:wandering_family = flag:dynasty
			}
			NOT = {
				any_player = {
					dynasty = prev.dynasty
				}
			}
		}

		## RULER ##
		trigger_if = { 
			limit = {
				has_global_variable = wandering_rulers
				global_var:wandering_rulers = flag:rulers_and_heirs
			}
			NOT = { is_ruler = yes }
			NOT = {
				any_ruler = {
					scope:wandering_candidate = {
						is_primary_heir_of = prev
					}
				}
			}
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_rulers
				global_var:wandering_rulers = flag:rulers
			}
			NOT = { is_ruler = yes }
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_rulers
				global_var:wandering_rulers = flag:rulers_not_immune
			}
			# Rulers can wander
		}

		## Councillor
		trigger_if = { 
			limit = {
				has_global_variable = wandering_councilors_wont_wander
				global_var:wandering_councilors_wont_wander = yes
			}
			is_councillor = no
		}
		## Court Position
		trigger_if = { 
			limit = {
				has_global_variable = wandering_common_court_position_wont_wander
				global_var:wandering_common_court_position_wont_wander = yes
			}
			has_any_court_position = no
		}

		## Elf Type
		trigger_if = { 
			limit = {
				has_global_variable = wandering_elf_rank
				global_var:wandering_elf_rank = flag:tier_1_and_below
			}
			is_elf_blood = yes
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_elf_rank
				global_var:wandering_elf_rank = flag:tier_2_and_below
			}
			OR = {
				is_elf_blood = yes
				is_elf = yes
			}
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_elf_rank
				global_var:wandering_elf_rank = flag:tier_3_and_below
			}
			is_low_elf_tier = yes
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_elf_rank
				global_var:wandering_elf_rank = flag:fae_tier_and_below
			}
			OR = {
				is_low_elf_tier = yes
				is_fae_elf_tier = yes
			}
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_elf_rank
				global_var:wandering_elf_rank = flag:celestial_tier_and_below
			}
			OR = {
				is_low_elf_tier = yes
				is_fae_elf_tier = yes
				is_celestial_elf_tier = yes
			}
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_elf_rank
				global_var:wandering_elf_rank = flag:ainur_tier_and_below
			}
			# Any elf types will work
		}

		## Aelurans
		trigger_if = { 
			limit = {
				has_global_variable = wandering_aelurans_wont_wander
				global_var:wandering_aelurans_wont_wander = yes
			}
			is_aeluran_sister_or_higher = no
		}

		## Bloodline
		trigger_if = { 
			limit = {
				has_global_variable = wandering_bloodline_count
				global_var:wandering_bloodline_count = flag:bloodline_trait_0
			}
			# All Bloodline Trait characters will Wander
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_bloodline_count
				global_var:wandering_bloodline_count = flag:bloodline_trait_1
			}
			bloodline_trait_score >= 1
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_bloodline_count
				global_var:wandering_bloodline_count = flag:bloodline_trait_2
			}
			bloodline_trait_score >= 2
		}
		trigger_else_if = { 
			limit = {
				has_global_variable = wandering_bloodline_count
				global_var:wandering_bloodline_count = flag:bloodline_trait_3
			}
			bloodline_trait_score >= 3
		}
	}

	effect = {
		elf_wander_pop_control_effect = yes
	}
}
