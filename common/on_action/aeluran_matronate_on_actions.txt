# The flock's piety: every quarter a Matron receives piety from the Aeluran rulers whose capital lies in her Matronate,
# and the High Matriarch from those outside every Matronate (aeluran_flock_piety_value). The monthly total is kept in
# var:aeluran_flock_piety for the Order window.
quarterly_playable_pulse = {
	on_actions = {
		aeluran_flock_piety_pulse
	}
}

aeluran_flock_piety_pulse = {
	trigger = {
		OR = {
			has_title = title:d_high_matriarch
			any_held_title = { aeluran_is_matronate_title_trigger = yes }
		}
	}
	effect = {
		save_scope_as = flock_shepherd
		set_variable = { name = aeluran_flock_piety value = 0 }
		every_ruler = {
			limit = {
				aeluran_flock_member_trigger = yes
				this != scope:flock_shepherd
				OR = {
					# In her Matronate
					capital_county.clerical_region_title.holder ?= scope:flock_shepherd
					# The High Matriarch's: outside every Matronate
					AND = {
						scope:flock_shepherd = { has_title = title:d_high_matriarch }
						NOT = { capital_county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
					}
				}
			}
			save_temporary_scope_value_as = { name = flock_piety value = aeluran_flock_piety_value }
			scope:flock_shepherd = { change_variable = { name = aeluran_flock_piety add = scope:flock_piety } }
		}
		add_piety = {
			value = var:aeluran_flock_piety
			multiply = 3
		}
	}
}

# An AI High Matriarch founds Matronates and replaces Matrons who hate her
yearly_global_pulse = {
	on_actions = {
		aeluran_matronate_ai_yearly
		aeluran_see_yearly_sweep
	}
}

aeluran_matronate_ai_yearly = {
	effect = {
		title:d_high_matriarch.holder ?= {
			if = {
				limit = { is_ai = yes }
				aeluran_matronate_ai_yearly_effect = yes
			}
		}
	}
}

# root = the character about to die, who still holds her titles: a Matronate she holds passes to a raised Sister
# before her family inherits the rest.
on_death = {
	on_actions = {
		aeluran_matronate_on_death
	}
}

aeluran_matronate_on_death = {
	trigger = {
		any_held_title = { aeluran_is_matronate_title_trigger = yes }
	}
	effect = {
		aeluran_matronate_vacate_held_effect = yes

		# A player's Matron dying is not a game over. The Matronate is granted away above and her Seat dies with her,
		# so a landless Matron leaves nothing to inherit: with no heir of her dynasty to continue as, the player
		# continues as the Sister who takes up her Matronate (as aeluran_matriarch_player_continuity_effect does)
		if = {
			limit = {
				is_ai = no
				NOT = { exists = player_heir }
				exists = scope:matronate_successor
				scope:matronate_successor = {
					is_ai = yes
					this != root
				}
			}
			set_player_character = scope:matronate_successor
		}
	}
}

# Fired by code when a clerical region's holder and its capital county stop sharing a faith, or the faith stops supporting regions.
# root = region holder, scope:title = region title, scope:county = its capital county, scope:reason = flag
# Vanilla's handlers (pam_clergy_events.0051/.0052) destroy the title and strip the holder's lands; the mod's fork skips
# Matronates and sees whose seat turned Aeluran. A Matron who leaves the faith gives up her Matronate to a raised Sister.
# A foreign see whose seat turned Aeluran is claimed for the Order once an elven ruler holds the seat and half its
# counties are Aeluran (PM 2026-10-04); until then it stands and is re-checked as counties convert or change hands.
on_clerical_region_faith_mismatch = {
	on_actions = {
		aeluran_matronate_faith_mismatch
	}
}

aeluran_matronate_faith_mismatch = {
	trigger = {
		OR = {
			scope:title = { aeluran_is_matronate_title_trigger = yes }
			aeluran_see_seat_turned_aeluran_trigger = yes
		}
	}
	effect = {
		# A foreign region (an archdiocese) whose seat turned Aeluran
		if = {
			limit = { aeluran_see_seat_turned_aeluran_trigger = yes }
			if = {
				limit = { scope:title = { aeluran_see_claimable_trigger = yes } }
				debug_log = "Archdiocese seat turned Aeluran under an elven ruler, half its counties Aeluran: the Order claims the see"
				scope:title = { aeluran_see_claim_effect = yes }
			}
			else = {
				debug_log = "Archdiocese seat turned Aeluran: the see stands until an elven ruler holds the seat and half its counties are Aeluran"
				debug_log_scopes = yes
			}
		}
		# A Matronate whose capital left the faith moves its capital to another Aeluran county of it
		else_if = {
			limit = { scope:reason = flag:county_converted }
			debug_log = "MATRONATE mismatch: county_converted"
			aeluran_matronate_capital_lost_effect = yes
		}
		else_if = {
			limit = { scope:reason = flag:character_converted }
			debug_log = "MATRONATE mismatch: character_converted"
			save_scope_as = outgoing_matron
			scope:title = { aeluran_matronate_succession_effect = yes }
		}
		else_if = {
			limit = { scope:reason = flag:changed_tenets_or_doctrines }
			debug_log = "MATRONATE mismatch: changed_tenets_or_doctrines"
		}
		else = { debug_log = "MATRONATE mismatch: unknown reason" }
		debug_log_scopes = yes
	}
}

# A foreign see whose seat is Aeluran is re-checked when a county inside it turns Aeluran or changes hands, and once a
# year, and claimed once it qualifies (aeluran_see_claimable_trigger)
# Root = the county that turned Aeluran
aeluran_see_check_converted_county = {
	trigger = {
		clerical_region_title ?= { aeluran_see_claimable_trigger = yes }
	}
	effect = {
		debug_log = "County turned Aeluran inside a foreign see that now qualifies: the Order claims the see"
		clerical_region_title = { aeluran_see_claim_effect = yes }
	}
}

# Root = the new holder, scope:title = the county gained
aeluran_see_check_gained_county = {
	trigger = {
		scope:title = {
			tier = tier_county
			clerical_region_title ?= { aeluran_see_claimable_trigger = yes }
		}
	}
	effect = {
		debug_log = "County changed hands inside a foreign see that now qualifies: the Order claims the see"
		scope:title.clerical_region_title = { aeluran_see_claim_effect = yes }
	}
}

aeluran_see_yearly_sweep = {
	effect = { aeluran_see_sweep_effect = yes }
}

# Matronates grow with their flock's realms: a county that changes hands, or turns Aeluran, joins the Matronate its
# realm's capital lies in when it borders it (PM 2026-10-01)
on_title_gain = {
	on_actions = {
		aeluran_matronate_absorb_gained_county
		aeluran_see_check_gained_county
	}
}

aeluran_matronate_absorb_gained_county = {
	trigger = {
		scope:title = {
			tier = tier_county
			aeluran_matronate_uncovered_county_trigger = yes
		}
	}
	effect = {
		scope:title = { aeluran_matronate_absorb_county_effect = yes }
	}
}

on_county_faith_change = {
	on_actions = {
		aeluran_matronate_absorb_converted_county
		aeluran_see_check_converted_county
	}
}


# Root = the county
aeluran_matronate_absorb_converted_county = {

	}
	effect = {
		aeluran_matronate_absorb_county_effect = yes
	}
}
