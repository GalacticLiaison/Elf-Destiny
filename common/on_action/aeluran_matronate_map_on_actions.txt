# Before the lobby, so the new temples are on the map players pick from. The Red Sisters who arrive later take them
# through the lease.
on_game_start = {
	on_actions = {
		aeluran_matronate_seed_temples
	}
}

aeluran_matronate_seed_temples = {
	effect = {
		aeluran_matronate_seed_temples_effect = yes
	}
}

# Red Sister temple markers (common/situation/situations/elf_dest_rs_temple_marker_situation.txt). Leases fire no
# on_action, so: a full pass after the lobby and yearly, a check on each ruler's quarterly pulse, and one when A Matter
# of Land settles (aeluran_matter_of_land_settled_effect). Each marker checks itself monthly and closes when its Sister
# is gone (the marker type's on_monthly).
on_game_start_after_lobby = {
	on_actions = {
		aeluran_rs_temple_markers_sync
	}
}

yearly_global_pulse = {
	on_actions = {
		aeluran_rs_temple_markers_sync
	}
}

aeluran_rs_temple_markers_sync = {
	effect = {
		# A save from before 2026-10-04 may still carry the Order situation: end it
		aeluran_matronate_map_retire_effect = yes
		aeluran_matronate_map_sync_temple_markers_effect = yes
	}
}

# Root = a landed ruler
quarterly_playable_pulse = {
	on_actions = {
		aeluran_rs_temple_markers_quarterly
	}
}

aeluran_rs_temple_markers_quarterly = {
	trigger = {
		exists = cp:councillor_aeluran_advisor
	}
	effect = {
		aeluran_rs_temple_markers_for_ruler_effect = yes
	}
}
