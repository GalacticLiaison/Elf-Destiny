# Characters the game creates directly (pool wanderers, guests, court populators, mercenary captains, generated heirs and
# landholders, create_character in events) get their culture's ethnicity, so elf-culture ones look like elves, but
# nothing gives them the race. This makes them elves of their culture's kind, as the game-start pass does for the
# characters that exist at game start (dark_elf_convert_ethnicity_elves). Born children use the birth on_actions.
# Registered through on_actions so vanilla's own on_character_created effect block is left alone.
on_character_created = {
	on_actions = {
		elf_character_created_race
		elf_character_created_traits
	}
}

# Earned traits (heroic, passionate, their Valar forms, tier-4 genetics) on a character who hasn't earned them get
# swapped for the base trait straight away, instead of waiting for the five-year pulse (fix_potential_trait_characters).
# random_creation_weight = 0 keeps the personality ones out of random generation; this catches anything else.
elf_character_created_traits = {
	trigger = {
		# The ruler designer family keeps what the player picked
		NOT = { scope:creation_reason = flag:ruler_designer }
	}
	on_actions = {
		# Wait a day, like the race pass: templates add traits, race and Valar status in the same tick
		delay = { days = 1 }
		elf_character_created_traits_apply
	}
}

elf_character_created_traits_apply = {
	trigger = {
		is_alive = yes
	}
	on_actions = {
		fix_potential_trait_characters
	}
}

elf_character_created_race = {
	trigger = {
		# The ruler designer character and family get their race from the designer (after-lobby race logic)
		NOT = { scope:creation_reason = flag:ruler_designer }
	}
	on_actions = {
		# Wait a day: character templates (e.g. the Aeluran matchmaking ones) set their own race flag and race-init in the
		# same tick. The delayed entry's trigger is checked again after the delay, so those characters are skipped.
		delay = { days = 1 }
		elf_character_created_race_apply
	}
}

elf_character_created_race_apply = {
	trigger = {
		is_alive = yes
		elf_needs_culture_race_trigger = yes
	}
	effect = {
		elf_apply_culture_race_effect = yes
	}
}
