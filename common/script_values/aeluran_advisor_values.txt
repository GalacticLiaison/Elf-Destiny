### Leased temples (PM 2026-10-07): every temple the ruler's Red Sister holds under lease in the ruler's own counties
### yields magi too, whether or not she holds anything else. Titled Order vassals keep their temples (the lease skips
### them), so they contribute through the vassal contract instead.

# Magi one leased temple adds to a delivery. Temple buildings will add to this later.
aeluran_temple_magi_yield = {
	value = 10
}

# The Matronate's cut of the temple yield. 0 until the Matron's reserve exists; the split is decided later.
aeluran_temple_magi_matronate_share = {
	value = 0
}

# Church holdings in the ruler's own counties that the Order tends. Two forms, both seen in game 2026-10-07:
#  - leased: the ruler stays the holder and the temple is leased to her Red Sister (Virion of Mytharil: is_leased_out = yes);
#  - held outright by a Sister, Matron or Matriarch (a landed Red Sister, or an Order-government ruler's own temple;
#    the lease skips those, so is_leased_out = no there).
# Holy-order leases are not the Order's.
aeluran_leased_temple_count = {
	value = 0
	if = {
		limit = { exists = cp:councillor_aeluran_advisor }
		every_held_title = {
			limit = { tier = tier_county }
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_barony
					title_province = { has_holding_type = church_holding }
					OR = {
						AND = {
							is_leased_out = yes
							NOT = { is_under_holy_order_lease = yes }
						}
						holder ?= { is_aeluran_sister_or_higher = yes }
					}
				}
				add = 1
			}
		}
	}
}

# The Realm Authority bonus on the whole delivery: +25% / +50% / +75%

	}

	if = {
		limit = { # have a decimal remainder
			NOT = { aeluran_war_magi_reinforcements_modulo = 0 }
		}

		if = {
			limit = { # modulo numerator was smaller than the denominator
				aeluran_war_magi_reinforcements_modulo = total_aeluran_magi_reinforcements
			}

			if = {
				limit = {
					aeluran_reinforcements_war_magi_share <= 0.5
				}
				floor = yes
			}
			else = {
				ceiling = yes
			}
		}
		else = {
			if = {
				limit = { aeluran_war_magi_reinforcements_decimal_remainder <= 0.5 }
				floor = yes
			}
			else = {
				ceiling = yes
			}
		}
	}
}

### The Aeluran host (aeluran_repeating_scripted_effects.txt)

# Realm Authority level that sizes the host. The law's on_pass sets set_aeluran_army_level because has_realm_law
# still reports the old law while on_pass runs; aeluran_apply_host_size_effect clears it again.
aeluran_army_level_value = {
	if = {
		limit = { exists = var:set_aeluran_army_level }
		value = var:set_aeluran_army_level
	}
	else_if = {
		limit = { has_realm_law = aeluran_realm_authority_3 }
		value = 3
	}
	else_if = {
		limit = { has_realm_law = aeluran_realm_authority_2 }
		value = 2
	}
	else = {
		value = 1
	}
}

# Sub-regiments each regiment of the host may hold: 3 / 5 / 10 by Realm Authority, plus one per leased temple
# (PM 2026-10-07, "law plus temples"; temple buildings will add more later)
aeluran_host_regiment_size_target = {
	if = {
		limit = { aeluran_army_level_value = 3 }
		value = 10
	}
	else_if = {
		limit = { aeluran_army_level_value = 2 }
		value = 5
	}
	else = {
		value = 3
	}
	add = aeluran_leased_temple_count
}

# Men that is: sub-regiments x stack (aeluran_war_magi 50, aeluran_artillery 25; men_at_arms_types/spark_maa_types.txt)
aeluran_host_war_magi_cap = {
	value = aeluran_host_regiment_size_target
	multiply = 50
}
aeluran_host_artillery_cap = {
	value = aeluran_host_regiment_size_target
	multiply = 25
}

# Sub-regiments to add so a regiment the host holds reaches the level's size (negative = already larger)

aeluran_host_artillery_size_delta = {
	value = aeluran_host_regiment_size_target
