# 1 star (tier 0) x1, 2 stars (tier 3) x1.5, 3 stars (tier 4) x2

# A healthy piety reward: a major sum for a 1 star task, more the further the god sends you
deity_task_piety_reward = {
	value = major_piety_gain
	if = {
		limit = { task_contract_tier >= 4 }
		value = massive_piety_gain
	}
	else_if = {
		limit = { task_contract_tier >= 3 }
		multiply = 1.5
	}
}

# Blood Rite. Quality is summed over every prisoner marked for the rite
blood_rite_max_marked = 3
blood_rite_fair = 5
blood_rite_strong = 10
blood_rite_exceptional = 15

blood_rite_prisoner_value = {
	value = 0
	# Rank
	if = {
		limit = { highest_held_title_tier = tier_barony }
		add = 1
	}
	else_if = {
		limit = { highest_held_title_tier = tier_county }
		add = 2
	}
	else_if = {
		limit = { highest_held_title_tier = tier_duchy }
		add = 3
	}
	else_if = {
		limit = { highest_held_title_tier = tier_kingdom }
		add = 5
	}
	else_if = {
		limit = { highest_held_title_tier = tier_empire }
		add = 8
	}
	# Elf blood
	if = {
		limit = { is_any_elf_type = yes }
		add = 2
	}
	if = {
		limit = { has_trait = pure_blooded }
		add = 2
	}
	else_if = {
		limit = { has_trait = purer_blooded }
		add = 3
	}
	else_if = {
		limit = { has_trait = purest_blooded }
		add = 4
	}
	# The blood carries what the Serpent wants
	if = {
		limit = { has_trait = physique_good_1 }
		add = 2
	}
	if = {
		limit = { has_trait = physique_good_2 }
		add = 2
	}
	if = {
		limit = { has_trait = physique_good_3 }
		add = 2
	}
	if = {
		limit = { has_trait = intellect_good_1 }
		add = 2
	}
	if = {
		limit = { has_trait = intellect_good_2 }
		add = 2
	}
	if = {
		limit = { has_trait = intellect_good_3 }
		add = 2
	}
	if = {
		limit = { has_trait = beauty_good_1 }
		add = 2
	}
	if = {
		limit = { has_trait = beauty_good_2 }
		add = 2
	}
	if = {
		limit = { has_trait = beauty_good_3 }
		add = 2
	}
	if = {
		limit = { has_trait = physique_bad_1 }
		subtract = 1
	}
	if = {
		limit = { has_trait = physique_bad_2 }
		subtract = 1
	}
	if = {
		limit = { has_trait = physique_bad_3 }
		subtract = 1
	}
	if = {
		limit = { has_trait = intellect_bad_1 }
		subtract = 1
	}
	if = {
		limit = { has_trait = intellect_bad_2 }
		subtract = 1
	}
	if = {
		limit = { has_trait = intellect_bad_3 }
		subtract = 1
	}
	if = {
		limit = { has_trait = beauty_bad_1 }
		subtract = 1
	}
	if = {
		limit = { has_trait = beauty_bad_2 }
		subtract = 1
	}
	if = {
		limit = { has_trait = beauty_bad_3 }
		subtract = 1
	}
	# The strongest blood is your own. Root is the character performing the rite
	if = {
		limit = {
			is_close_family_of = root
		}
		add = 5
	}
}

# Root = the character performing the rite
blood_rite_quality = {
	value = 0
	every_prisoner = {
		limit = { has_character_flag = marked_for_blood_rite }
		add = blood_rite_prisoner_value
	}
}
