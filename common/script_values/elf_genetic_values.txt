    }
    else_if = {

        }
        add = aratar_genetic_threshold_value
    }
    # Dark Elf Beta - dark race flags map to the same tier thresholds; the light
    # branches above stay first (history characters may carry both flag families).
    else_if = { limit = { has_character_flag = race_dark_elf_blood } add = elf_blood_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_elf } add = elf_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_high_elf } add = high_elf_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_true_elf } add = true_elf_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_fae } add = fae_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_fae_radiant } add = fae_radiant_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_celestial } add = celestial_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_seraphic_celestial } add = seraphic_celestial_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_eldar } add = eldar_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_maiar } add = maiar_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_valar } add = valar_genetic_threshold_value }
    else_if = { limit = { has_character_flag = race_dark_aratar } add = aratar_genetic_threshold_value }
