# The Order window (gui/aeluran_order_window.gui): a Matron's flock, or the High Matriarch's.

# Is the player a shepherd of the Order (a Matron holding a Matronate, or the High Matriarch)?
aeluran_order_is_shepherd = {
	scope = character

	is_shown = {
		OR = {
			has_title = title:d_high_matriarch
			any_held_title = { aeluran_is_matronate_title_trigger = yes }
		}
	}
}

# Rebuild the flock list (var list aeluran_order_flock on the player) when the window opens
aeluran_order_refresh = {
	scope = character

	effect = {
		clear_variable_list = aeluran_order_flock
		save_scope_as = flock_shepherd
		every_ruler = {
			limit = {
				aeluran_flock_member_trigger = yes
				this != scope:flock_shepherd
				OR = {
					capital_county.clerical_region_title.holder ?= scope:flock_shepherd
					AND = {
						scope:flock_shepherd = { has_title = title:d_high_matriarch }
						NOT = { capital_county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
					}
				}
			}
			scope:flock_shepherd = { add_to_variable_list = { name = aeluran_order_flock target = prev } }
		}
	}
}

# Does this ruler (scope:target) have an Advisor post open for the player to fill?
aeluran_order_post_is_open = {
	scope = character
	saved_scopes = { target }

	is_shown = {
		scope:target = {
			var:aeluran_advisor_vacancy ?= root
			NOT = { exists = cp:councillor_aeluran_advisor }
		}
	}
}

# Open Appoint an Advisor for this ruler (scope:target)
aeluran_order_open_appoint = {
	scope = character
	saved_scopes = { target }

	effect = {
		open_interaction_window = {
			interaction = aeluran_appoint_advisor_interaction
			actor = root
			recipient = scope:target
		}
	}
}

# The player picks two unwed elves (player vars aeluran_bl_first / aeluran_bl_second) and proposes the match.

# Rebuild the board (var list aeluran_bloodlines on the player)
aeluran_bl_refresh = {
	scope = character

	effect = {
		aeluran_bloodlines_build_list_effect = yes
	}
}

# Select or deselect an elf (scope:target)
aeluran_bl_select = {
	scope = character
	saved_scopes = { target }

	effect = {
		if = {
			limit = { var:aeluran_bl_first ?= scope:target }
			remove_variable = aeluran_bl_first
		}
		else_if = {
			limit = { var:aeluran_bl_second ?= scope:target }
			remove_variable = aeluran_bl_second
		}
		else_if = {
			limit = { NOT = { has_variable = aeluran_bl_first } }
			set_variable = { name = aeluran_bl_first value = scope:target }
		}
		else = {
			set_variable = { name = aeluran_bl_second value = scope:target }
		}
	}
}

# Is this elf (scope:target) one of the chosen pair?
aeluran_bl_is_selected = {
	scope = character
	saved_scopes = { target }

	is_shown = {
		OR = {
			var:aeluran_bl_first ?= scope:target
			var:aeluran_bl_second ?= scope:target
		}
	}
}

# Can the chosen pair be proposed?
aeluran_bl_propose = {
	scope = character

	is_valid = {
		exists = var:aeluran_bl_first
		exists = var:aeluran_bl_second
		custom_tooltip = {
			text = aeluran_bl_cannot_wed_tt
			var:aeluran_bl_first = { aeluran_bloodlines_pair_trigger = { OTHER = root.var:aeluran_bl_second } }
		}
		custom_tooltip = {
			text = aeluran_bl_awaiting_answer_tt
			trigger_if = {
				limit = { has_variable = aeluran_bl_pending }
				var:aeluran_bl_pending <= 0
			}
		}
	}

	effect = {
		var:aeluran_bl_first = { save_scope_as = bl_a }
		var:aeluran_bl_second = { save_scope_as = bl_b }
		remove_variable = aeluran_bl_first
		remove_variable = aeluran_bl_second
		aeluran_bloodlines_propose_effect = yes
		aeluran_bloodlines_build_list_effect = yes
	}
}
