# The Order window (gui/aeluran_order_window.gui): a Matron's flock, or the High Matriarch's.

# Is the player a shepherd of the Order (a Matron holding a Matronate, or the High Matriarch)?
aeluran_order_is_shepherd = {
	scope = character

	is_shown = {
		OR = {
			has_title = title:d_high_matriarch
			any_held_title = { aeluran_is_matronate_title_trigger = yes }
		}
	}
}

# Rebuild the flock list (var list aeluran_order_flock on the player) when the window opens. Also works out the flock's
# monthly piety now (the quarterly pulse pays it), and opens the post of any ruler in her care still without an Advisor,
# so she can fill it at once instead of waiting for the Order's next quarterly round.
aeluran_order_refresh = {
	scope = character

	effect = {
		clear_variable_list = aeluran_order_flock
		save_scope_as = flock_shepherd
		aeluran_order_set_focus_effect = yes
		# Sisters still promised to a ruler purged from the save hold an empty appointed_to_advise: clear it so they can
		# be appointed again
		every_living_character = {
			limit = {
				has_trait = aeluran_sister
				has_variable = appointed_to_advise
				NOT = { exists = var:appointed_to_advise }
			}
			remove_variable = appointed_to_advise
		}
		set_variable = { name = aeluran_flock_piety value = 0 }
		every_ruler = {
			limit = {
				aeluran_flock_member_trigger = yes
				this != scope:flock_shepherd
				OR = {
					capital_county.clerical_region_title.holder ?= scope:flock_shepherd
					AND = {
						scope:flock_shepherd = { has_title = title:d_high_matriarch }
						NOT = { capital_county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
					}
				}
			}
			scope:flock_shepherd = { add_to_variable_list = { name = aeluran_order_flock target = prev } }
			# Their spouses, for the row's spouse portraits (the GUI has no spouse list outside the character window)
			save_temporary_scope_as = order_ruler
			clear_variable_list = aeluran_order_spouses
			every_spouse = {
				scope:order_ruler = { add_to_variable_list = { name = aeluran_order_spouses target = prev } }
			}
			# The Red Sister seated on their council, for the row's portrait (the GUI cannot read a council seat itself)
			if = {
				limit = { exists = cp:councillor_aeluran_advisor }
				set_variable = { name = aeluran_order_red_sister value = cp:councillor_aeluran_advisor }
			}
			else_if = {
				limit = { has_variable = aeluran_order_red_sister }
				remove_variable = aeluran_order_red_sister
			}
			save_temporary_scope_value_as = { name = flock_piety value = aeluran_flock_piety_value }
			scope:flock_shepherd = { change_variable = { name = aeluran_flock_piety add = scope:flock_piety } }
			if = {
				limit = {
					NOT = { exists = cp:councillor_aeluran_advisor }
					NOT = { has_character_flag = aeluran_advisor_is_on_the_way }
					NOT = { has_variable = aeluran_advisor_vacancy }
					faith = { has_tenet = tenet_aeluran_sisters }
				}
				aeluran_open_advisor_post_effect = yes
			}
		}
	}
}

# Sisters #
##########
# The Sisters under a shepherd: those whose religious superior she is (the Advisors holding her temple leases), and
# every Sister living in her Matronate (the High Matriarch's: outside every Matronate). Advisors are listed first.
# var:aeluran_order_tithe_share is the shepherd's share of a lessee's temple taxes (aeluran_lease): 15% for a Matron,
# 20% for the High Matriarch where no Matronate stands between them (lease liege and top liege both).

# Rebuild the list (var list aeluran_order_sisters on the player)
aeluran_order_sisters_refresh = {
	scope = character

	effect = {
		clear_variable_list = aeluran_order_sisters
		save_scope_as = flock_shepherd
		if = {
			limit = { any_held_title = { aeluran_is_matronate_title_trigger = yes } }
			random_held_title = {
				limit = { aeluran_is_matronate_title_trigger = yes }
				save_scope_as = matronate
			}
			set_variable = { name = aeluran_order_tithe_share value = 0.15 }
		}
		else = {
			set_variable = { name = aeluran_order_tithe_share value = 0.2 }
		}
		# Advisors first, then the rest
		every_living_character = {
			limit = {
				has_trait = aeluran_sister
				has_council_position = councillor_aeluran_advisor
				aeluran_order_sister_in_care_trigger = yes
			}
			scope:flock_shepherd = { add_to_variable_list = { name = aeluran_order_sisters target = prev } }
		}
		every_living_character = {
			limit = {
				has_trait = aeluran_sister
				NOT = { has_council_position = councillor_aeluran_advisor }
				aeluran_order_sister_in_care_trigger = yes
			}
			scope:flock_shepherd = { add_to_variable_list = { name = aeluran_order_sisters target = prev } }
		}
	}
}

# Is this Sister (scope:target) one of the Order's Advisors, a Red Sister on a ruler's council?

# Bloodlines board #
####################
# The player picks two unwed elves (player vars aeluran_bl_first / aeluran_bl_second) and proposes the match.

# Rebuild the board (var list aeluran_bloodlines on the player)
aeluran_bl_refresh = {
	scope = character

	effect = {
		aeluran_bloodlines_build_list_effect = yes
		if = {
			limit = {
				exists = var:aeluran_bl_first
				NOT = { is_target_in_variable_list = { name = aeluran_bloodlines target = var:aeluran_bl_first } }
			}
			remove_variable = aeluran_bl_first
		}
		aeluran_bloodlines_build_matches_effect = yes
	}
}

# Select or deselect an elf (scope:target)
aeluran_bl_select = {
	scope = character
	saved_scopes = { target }

	effect = {
		if = {
			limit = { var:aeluran_bl_first ?= scope:target }
			remove_variable = aeluran_bl_first
		}
		else_if = {
			limit = { var:aeluran_bl_second ?= scope:target }
			remove_variable = aeluran_bl_second
		}
		else_if = {
			limit = { NOT = { has_variable = aeluran_bl_first } }
			set_variable = { name = aeluran_bl_first value = scope:target }
		}
		else = {
			set_variable = { name = aeluran_bl_second value = scope:target }
		}
	}
}

# Is this elf (scope:target) one of the chosen pair?
aeluran_bl_is_selected = {
	scope = character
	saved_scopes = { target }

	is_shown = {
		OR = {
			var:aeluran_bl_first ?= scope:target
			var:aeluran_bl_second ?= scope:target
		}
	}
}

# Can the chosen pair be proposed?
aeluran_bl_propose = {
	scope = character

	is_valid = {
		custom_tooltip = {
			text = aeluran_bl_pick_noble_tt
			exists = var:aeluran_bl_first
		}
		custom_tooltip = {
			text = aeluran_bl_pick_match_tt
			exists = var:aeluran_bl_second
		}
		custom_tooltip = {
			text = aeluran_bl_cannot_wed_tt
			var:aeluran_bl_first = { aeluran_bloodlines_pair_trigger = { OTHER = root.var:aeluran_bl_second } }
		}
		custom_tooltip = {
			text = aeluran_bl_awaiting_answer_tt
			trigger_if = {
				limit = { has_variable = aeluran_bl_pending }
				var:aeluran_bl_pending <= 0
			}
		}
	}

	effect = {
		var:aeluran_bl_first = { save_scope_as = bl_a }
		var:aeluran_bl_second = { save_scope_as = bl_b }
		if = {
			limit = { has_variable = aeluran_bl_matrilineal }
			save_scope_value_as = { name = bl_matrilineal_choice value = flag:matrilineal }
		}
		else = {
			save_scope_value_as = { name = bl_matrilineal_choice value = flag:patrilineal }
		}
		if = { limit = { has_variable = aeluran_bl_matrilineal } remove_variable = aeluran_bl_matrilineal }
		if = { limit = { has_variable = aeluran_bl_house_parent } remove_variable = aeluran_bl_house_parent }
		if = { limit = { has_variable = aeluran_bl_match_matron } remove_variable = aeluran_bl_match_matron }
		remove_variable = aeluran_bl_first
		remove_variable = aeluran_bl_second
		aeluran_bloodlines_propose_effect = yes
		aeluran_bloodlines_build_list_effect = yes
		aeluran_bloodlines_build_matches_effect = yes
	}
}

# Matchmaking (two columns)  #
##############################
# Left: the nobles in her care with room for a spouse (aeluran_bloodlines). Choose one (var:aeluran_bl_first) and the
# right column lists their possible matches (aeluran_bl_matches), each with a Propose button.

# Choose this noble (scope:target), or choose again to clear
aeluran_bl_choose = {
	scope = character
	saved_scopes = { target }

	effect = {
		if = { limit = { has_variable = aeluran_bl_matrilineal } remove_variable = aeluran_bl_matrilineal }
		if = { limit = { has_variable = aeluran_bl_house_parent } remove_variable = aeluran_bl_house_parent }
		if = { limit = { has_variable = aeluran_bl_match_matron } remove_variable = aeluran_bl_match_matron }
		if = { limit = { has_variable = aeluran_bl_second } remove_variable = aeluran_bl_second }
		if = {
			limit = { var:aeluran_bl_first ?= scope:target }
			remove_variable = aeluran_bl_first
		}
		else = {
			set_variable = { name = aeluran_bl_first value = scope:target }
		}
		aeluran_bloodlines_build_matches_effect = yes
	}
}

# Propose the chosen noble and this match (scope:target)

	saved_scopes = { target }

			text = aeluran_bl_cannot_wed_tt

			trigger_if = {

				var:aeluran_bl_pending <= 0
			}
		}
	}

		scope:target = { save_scope_as = bl_b }
		remove_variable = aeluran_bl_first
		aeluran_bloodlines_propose_effect = yes
		aeluran_bloodlines_build_list_effect = yes
		aeluran_bloodlines_build_matches_effect = yes
	}
}

# Select or deselect this match (scope:target) for the chosen noble; Propose Match (aeluran_bl_propose) sends it
aeluran_bl_select_match = {
	scope = character
	saved_scopes = { target }

	effect = {
		if = { limit = { has_variable = aeluran_bl_match_matron } remove_variable = aeluran_bl_match_matron }
		if = {
			limit = { var:aeluran_bl_second ?= scope:target }
			remove_variable = aeluran_bl_second
		}
		else = {
			set_variable = { name = aeluran_bl_second value = scope:target }
			# Her Matron, for the Propose Match panel: her Matronate's holder, beyond the Matronates the High Matriarch
			scope:target = {
				if = {
					limit = { exists = var:aeluran_bl_home.holder }
					var:aeluran_bl_home.holder = { save_scope_as = bl_match_matron }
				}
				else = {
					title:d_high_matriarch.holder ?= { save_scope_as = bl_match_matron }
				}
			}
			scope:bl_match_matron ?= { root = { set_variable = { name = aeluran_bl_match_matron value = prev } } }
		}
		aeluran_bloodlines_default_marriage_type_effect = yes
	}
}

aeluran_bl_is_match_selected = {

# The Matchmaking panel's Matrilineal Marriage checkbox (needs both picked)
aeluran_bl_toggle_matrilineal = {
	scope = character

	is_valid = {
		exists = var:aeluran_bl_first
		exists = var:aeluran_bl_second
	}

	effect = {
		if = {
			limit = { has_variable = aeluran_bl_matrilineal }
			remove_variable = aeluran_bl_matrilineal
		}
		else = {
			set_variable = { name = aeluran_bl_matrilineal value = yes }
		}
		aeluran_bloodlines_update_house_parent_effect = yes
	}
}

aeluran_bl_is_matrilineal = {
	scope = character

	is_shown = {
		has_variable = aeluran_bl_matrilineal
	}
}
