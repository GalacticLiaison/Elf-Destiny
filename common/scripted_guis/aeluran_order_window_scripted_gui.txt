# The Order window (gui/aeluran_order_window.gui): a Matron's flock, or the High Matriarch's.

# Is the player a shepherd of the Order (a Matron holding a Matronate, or the High Matriarch)?
aeluran_order_is_shepherd = {
	scope = character

	is_shown = {
		OR = {
			has_title = title:d_high_matriarch
			any_held_title = { aeluran_is_matronate_title_trigger = yes }
		}
	}
}

# Rebuild the flock list (var list aeluran_order_flock on the player) when the window opens
aeluran_order_refresh = {
	scope = character

	effect = {
		clear_variable_list = aeluran_order_flock
		save_scope_as = flock_shepherd
		every_ruler = {
			limit = {
				aeluran_flock_member_trigger = yes
				this != scope:flock_shepherd
				OR = {
					capital_county.clerical_region_title.holder ?= scope:flock_shepherd
					AND = {
						scope:flock_shepherd = { has_title = title:d_high_matriarch }
						NOT = { capital_county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
					}
				}
			}
			scope:flock_shepherd = { add_to_variable_list = { name = aeluran_order_flock target = prev } }
		}
	}
}

# Does this ruler (scope:target) have an Advisor post open for the player to fill?
aeluran_order_post_is_open = {
	scope = character
	saved_scopes = { target }

	is_shown = {
		scope:target = {
			var:aeluran_advisor_vacancy ?= root
			NOT = { exists = cp:councillor_aeluran_advisor }
		}
	}
}

# Open Appoint an Advisor for this ruler (scope:target)
aeluran_order_open_appoint = {
	scope = character
	saved_scopes = { target }

	effect = {
		open_interaction_window = {
			interaction = aeluran_appoint_advisor_interaction
			actor = root
			recipient = scope:target
		}
	}
}

# Who sees the Red Sister temple cards on the map (gui/map_icon_layer.gui, the situation_map_pin_widget fork): members of
# the Order only. Checked against the viewing player, so the pins exist for everyone but draw for nobody else.
aeluran_order_sees_temple_cards = {
	scope = character

	is_shown = {
		OR = {
			has_trait = aeluran_sister
			has_trait = aeluran_matron
			has_trait = aeluran_high_matriarch
			has_government = aeluran_weavers_government
			has_title = title:d_high_matriarch
			any_held_title = { aeluran_is_matronate_title_trigger = yes }
		}
	}
}
