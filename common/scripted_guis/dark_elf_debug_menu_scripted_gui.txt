# Dark Elf Beta
# Toggles the four dark elf bloodline traits from the debug-menu trait-selection
# widget (elf_destiny_debug_menu.022 -> event_window_widget_trait_selection.gui).
# Mirrors Elf Destiny's elf_bloodline_trait_selection_button but for the dark
# houses; additive (a NEW scripted gui), so ED's version is left untouched.
# The widget's Dark Elf Bloodlines row passes trait = 1..4 to this.
dark_elf_bloodline_trait_selection_button = {
	scope = character
	saved_scopes = { trait }

	effect = {
		if = {
			limit = { scope:trait = 1 }
			if = {
				limit = { has_trait = royal_elf_drenmyre }
				remove_trait = royal_elf_drenmyre
			}
			else = {
				add_trait = royal_elf_drenmyre
			}
		}
		else_if = {
			limit = { scope:trait = 2 }
			if = {
				limit = { has_trait = royal_elf_rynvela }
				remove_trait = royal_elf_rynvela
			}
			else = {
				add_trait = royal_elf_rynvela
			}
		}
		else_if = {
			limit = { scope:trait = 3 }
			if = {
				limit = { has_trait = royal_elf_myrvessa }
				remove_trait = royal_elf_myrvessa
			}
			else = {
				add_trait = royal_elf_myrvessa
			}
		}
		else_if = {
			limit = { scope:trait = 4 }
			if = {
				limit = { has_trait = royal_elf_syldranis }
				remove_trait = royal_elf_syldranis
			}
			else = {
				add_trait = royal_elf_syldranis
			}
		}
	}
}
