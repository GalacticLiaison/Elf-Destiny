# Culture-Specific Innovations
innovation_elven_adaptability = {
	group = culture_group_regional
	culture_era = culture_era_tribal
	icon = "gfx/interface/icons/culture_innovations/innovation_elven_adaptability.dds"
	
	potential = {
		has_cultural_pillar = heritage_elf
	}
	
	culture_modifier = {
		culture_tradition_max_add = 100
	}
}

# Discoverable elven innovations (PM 2026-11-15). No culture knows them at game start. An expedition discovery finds
# one (events expedition_unique_encounters.0041 / .0042). After that it spreads the normal way, slowly, to elven
# cultures that accept a culture which already knows it (elf_dest_elven_innovation_spreads_trigger).
innovation_lembas_baking = {
	group = culture_group_regional
	culture_era = culture_era_tribal
	icon = "gfx/interface/icons/culture_innovations/innovation_lembas_baking.dds"

	potential = {
		has_cultural_pillar = heritage_elf
	}

	can_progress = {
		custom_description = {
			text = "innovation_elven_discovery_only"
			elf_dest_elven_innovation_spreads_trigger = { INNOVATION = innovation_lembas_baking }
		}
	}

	parameters = {
		unlocks_lembas_bakery = yes
	}
}
