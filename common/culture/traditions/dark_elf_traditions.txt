# Dark Elf Beta - dark elf culture traditions.
# tradition_the_old_ways_dark shares its display name with Elf Destiny's
# tradition_the_old_ways by design: both elf types keep their own version of
# the ancient ways. Art uses the vanilla intrigue backdrop instead of ED's
# forest layer until custom dark art exists.
# Effects per PM 2026-08-22: same as the light Old Ways but the herbalist
# bonus is swapped for knight effectiveness per intrigue, plus the tattooing
# custom (dark_elf_tattooing_common drives the tradition tattoo portrait
# modifiers in gfx/portraits/portrait_modifiers/dark_elf_tradition_tattoos.txt).
tradition_the_old_ways_dark = {
	category = elven

	layers = {
		0 = forest_bottom_layer_dark.dds
		1 = pattern_elf_1.dds
		4 = tradition_the_old_ways_dark_item.dds
	}

	is_shown = {
		any_player = {
			is_dark_elf_type = yes
		}
	}
	can_pick = {

	}

	parameters = {
		# Grants Fey Archers (same as the light Old Ways)
		unlock_maa_fey_archers = yes
		# Mystic More Common (same as the light Old Ways)
		mystic_trait_more_common = yes
		mystic_trait_gives_bonuses = yes
		# Tattooing custom: tooltip line + gates the tradition tattoo portrait modifiers
		dark_elf_tattooing_common = yes
	}

	character_modifier = {
		# Replaces the light version's herbalist bonus (PM choice)
		knight_effectiveness_per_intrigue = 0.01
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 0
	}
}
