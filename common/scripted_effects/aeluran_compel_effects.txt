# Compelling a defiant Sister: when a Sister refuses a shepherd of the Order (a Matron or the High Matriarch), the
# shepherd may let it go or force her with a battle of wills (aeluran_compel.001).
#
# Refusals that lead here (scope:compel_task):
#   flag:whispers  a Red Sister refuses Gather Whispers   (scope:actor = shepherd, scope:recipient = Sister)
#   flag:summon    a Sister refuses Summon to Court        (scope:actor = shepherd, scope:recipient = Sister)
#   flag:advisor   a Sister refuses an Advisor posting     (scope:actor = shepherd, scope:recipient = the ruler)
#
# Winning: she obeys, and resents it. Losing: backlash on the shepherd, scaled by how far the Sister was favoured
# (aeluran_compel_margin_value), and the Sister is emboldened.

# Called where the refusal happens. SISTER = the Sister, TASK = whispers / summon / advisor.
aeluran_compel_begin_effect = {
	$SISTER$ = { save_scope_as = compel_sister }
	save_scope_value_as = { name = compel_task value = flag:$TASK$ }
	scope:actor = { trigger_event = aeluran_compel.001 }
}

# Root = the shepherd. She lets the refusal stand.
aeluran_compel_let_go_effect = {
	if = {
		limit = { scope:compel_task = flag:whispers }
		add_piety = scope:compel_sister.aeluran_whispers_piety_cost
		if = {
			limit = {
				scope:compel_sister = { aeluran_whispers_betrayal_trigger = { SHEPHERD = root } }
			}
			aeluran_whispers_betray_effect = yes
		}
	}
}

# Root = the shepherd, who won. The Sister does as she was told.
aeluran_compel_win_effect = {
	scope:compel_sister = {
		add_opinion = {
			target = root
			modifier = aeluran_bent_to_her_will_opinion
		}
	}
	if = {
		limit = { scope:compel_task = flag:whispers }
		aeluran_whispers_start_effect = yes
	}
	else_if = {
		limit = { scope:compel_task = flag:summon }
		add_courtier = scope:compel_sister
	}
	else_if = {
		limit = { scope:compel_task = flag:advisor }
		save_scope_as = advisor_appointer
		scope:compel_sister = { save_scope_as = aeluran_sister_advisor }
		scope:recipient = { aeluran_appoint_chosen_advisor_effect = yes }
	}
}

# Root = the shepherd, who lost. The refusal stands, with backlash.
aeluran_compel_lose_effect = {
	aeluran_compel_backlash_effect = yes
	# A Red Sister who broke her shepherd's will tells her ruler what she was asked
	if = {
		limit = { scope:compel_task = flag:whispers }
		aeluran_whispers_betray_effect = yes
	}
}

# Root = the shepherd who lost a battle of wills to scope:compel_sister (scope:challenger / scope:defender /
# scope:battle_type still saved from battle_of_wills_effect)
aeluran_compel_backlash_effect = {
	aeluran_compel_backlash_self_effect = yes
	# The Sister is emboldened
	scope:compel_sister = {
		add_prestige = 100
		add_to_variable_list = { name = aeluran_unbroken_against target = root }
	}
	# Word spreads among the Sisters in her care
	save_scope_as = humbled_shepherd
	every_living_character = {
		limit = {
			has_trait = aeluran_sister
			NOT = { this = scope:compel_sister }
			OR = {
				superior ?= scope:humbled_shepherd
				location.county.clerical_region_title.holder ?= scope:humbled_shepherd
			}
		}
		add_opinion = {
			target = scope:humbled_shepherd
			modifier = aeluran_humbled_by_a_sister_opinion
		}
	}
}

# Root = whoever lost a battle of wills (scope:challenger / scope:defender / scope:battle_type saved). The backlash on
# the loser herself: stress and a weakened will, worse (with piety loss and a wound chance) for a crushing defeat.
# Call before marking the winner unbroken, which shifts the battle of wills value the margin is measured with.
aeluran_compel_backlash_self_effect = {
	if = {
		limit = { aeluran_compel_margin_value >= 15 }
		# Crushing: she was the stronger will by far
		add_stress = massive_stress_impact_gain
		add_character_modifier = {
			modifier = aeluran_will_shattered_modifier
			years = 5
		}
		add_piety = -100
		random = {
			chance = 50
			increase_wounds_no_death_effect = { REASON = wounds }
		}
	}
	else = {
		add_stress = major_stress_impact_gain
		add_character_modifier = {
			modifier = aeluran_will_recoiling_modifier
			years = 5
		}
	}
}

#################################################
# Pressing a ruler (aeluran_press_authority)   #
#################################################
# Root = the shepherd. scope:compel_ruler = the ruler who refused; scope:actor / scope:recipient from the interaction.
# The battle is fought by the shepherd or by the ruler's Red Sister (scope:press_champion).

# Won: the law rises and the ruler resents being forced
aeluran_press_compel_win_effect = {
	aeluran_press_authority_success_effect = yes
	scope:compel_ruler = {
		add_opinion = {
			target = root
			modifier = aeluran_coerced_by_the_order_opinion
		}
		if = {
			limit = { NOT = { scope:press_champion = root } }
			add_opinion = {
				target = scope:press_champion
				modifier = aeluran_coerced_by_the_order_opinion
			}
		}
		send_interface_toast = {
			type = event_toast_effect_bad
			title = aeluran_press_forced_ruler_toast
			left_icon = scope:press_champion
			right_icon = root
			custom_tooltip = aeluran_press_forced_ruler_toast_desc
		}
	}
}

# Lost, by the shepherd herself: her backlash, and the ruler is emboldened against her
aeluran_press_compel_lose_self_effect = {
	aeluran_compel_backlash_self_effect = yes
	scope:compel_ruler = {
		add_prestige = 100
		add_to_variable_list = { name = aeluran_unbroken_against target = root }
	}
}

# Lost, by the Red Sister: she bears the backlash and blames the shepherd; an AI ruler may dismiss her
aeluran_press_compel_lose_sister_effect = {
	scope:press_champion = {
		add_stress = major_stress_impact_gain
		add_opinion = {
			target = root
			modifier = aeluran_sent_me_to_fail_opinion
		}
	}
	scope:compel_ruler = {
		add_prestige = 50
		if = {
			limit = { is_ai = yes }
			random = {
				chance = 50
				fire_councillor = scope:press_champion
				scope:press_champion = {
					if = {
						limit = { has_variable = appointed_to_advise }
						remove_variable = appointed_to_advise
					}
				}
				aeluran_open_advisor_post_effect = yes
			}
		}
	}
}
