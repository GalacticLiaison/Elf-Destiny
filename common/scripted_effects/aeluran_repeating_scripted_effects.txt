determine_new_aeluran_advisor_effect = {
    if = { # First check if any of your vassals can do it
        limit = {
            any_vassal = {
                is_aeluran_sister_or_higher = yes
                is_capable_adult = yes
            }
        }

        save_scope_value_as = {
            name = advisor_type
            value = flag:vassal
        }

        random_vassal = {
            limit = {
                is_aeluran_sister_or_higher = yes
                is_capable_adult = yes
            }

            save_scope_as = aeluran_sister_advisor
        }
    }
    else_if = { # Next check if you have a courtier who can do it
        limit = {
            any_courtier = {
                is_aeluran_sister_or_higher = yes
                is_capable_adult = yes
            }
        }

        save_scope_value_as = {
            name = advisor_type
            value = flag:courtier
        }

        random_courtier = {
            limit = {
                is_aeluran_sister_or_higher = yes
                is_capable_adult = yes
            }

            save_scope_as = aeluran_sister_advisor
        }
    }
    else = { # Couldn't find an advisor, create and send one from Deepwood
        save_scope_value_as = {
            name = advisor_type
            value = flag:sent_from_deepwood
        }

        save_scope_value_as = {
            name = aeluran_advisor_coming_from_sanctum
            value = yes
        }

        create_aeluran_advisor_effect = yes
    }

    scope:aeluran_sister_advisor = {
        set_variable = {
            name = appointed_to_advise
            value = root
        }
        # A Sister found at court takes the court's chapter: her robes follow the ruler's tenet (PM 2026-10-05). A landed
        # Sister keeps her own rite and wears her own chapter's colour.
        if = {
            limit = {
                is_ruler = no
                exists = root.rite
            }
            set_character_rite = root.rite
        }
        aeluran_set_sister_clothing_effect = yes
    }
}


create_aeluran_advisor_effect = {
    # The body follows the ruler's race: a Dark Elf ruler is sent a dark-elf Sister of their own culture. Her robes follow
    # the ruler's tenet, set below once she has the rite (aeluran_set_sister_clothing_effect, PM 2026-10-05).
    if = {
        limit = {
            is_dark_elf = yes
        }
        create_character = {
            template = dark_aeluran_sister_advisor
            dynasty = none
            location = title:b_cykma.title_province
            faith = root.faith
            culture = root.culture
            save_scope_as = aeluran_sister_advisor

            after_creation = {
                add_character_modifier = {
                    modifier = on_official_aeluran_business
                }
            }
        }
    }
    else = {
        create_character = {
            template = aeluran_sister_advisor
            dynasty = none
            location = title:b_cykma.title_province
            # faith = global_var:high_matriarch.faith # swapped out so advisor matches players aeluran faith variant and protects against matriarch grabbing human faith
            faith = root.faith
            # culture = root.culture
            random_culture_in_group = root.culture
            save_scope_as = aeluran_sister_advisor

            after_creation = {
                add_character_modifier = {
                    modifier = on_official_aeluran_business
                }
            }
        }
    }
    # She follows the ruler's rite, not just the faith
    if = {
        limit = {
            exists = scope:aeluran_sister_advisor
            exists = rite
        }
        scope:aeluran_sister_advisor = {
            set_character_rite = root.rite
        }
    }
    # Robes by the ruler's tenet, whatever her race
    scope:aeluran_sister_advisor ?= { aeluran_set_sister_clothing_effect = yes }
}

# Root = an Aeluran Sister. Raises her to Matron: trait, clothing, Matron's Seat (which makes a courtier a landless
# ruler of the Order), and the Council's list. Matrons are raised only through Matronate succession
# (aeluran_matronate_succession_effect), which notifies a player; the old random replacement on a Matron's death is gone.
aeluran_raise_to_matron_effect = {
    save_scope_as = new_matron
    if = {
        limit = { has_trait = aeluran_sister }
        remove_trait = aeluran_sister
    }
    if = {
        limit = { NOT = { has_trait = aeluran_matron } }
        add_trait = aeluran_matron
    }
    aeluran_set_sister_clothing_effect = yes
    # A Matron who already rules land moves to the Order government (no title gain fires for her)
    if = {
        limit = {
            is_landed = yes
            NOT = { has_government = aeluran_weavers_government }
        }
        change_government = aeluran_weavers_government
    }
    aeluran_grant_matron_seat_effect = yes
    add_to_global_variable_list = {
        name = aeluran_matrons
        target = scope:new_matron
    }
}

#################################################################################################################
# Aeluran magi reinforcements
#
# Every years_between_magi_reinforcements the Red Sister's Magistration task hands her ruler the magi that the
# ruler's Aeluran Order vassals owe (aeluran_advisor_values.txt). The host is event troops: it never reinforces on
# its own, only the Order refills it. Script can only reach an event regiment through the army it stands in, and
# that army only exists while the ruler is at war, so:
#   - at peace the magi are PLEDGED (var:aeluran_pledged_war_magi / var:aeluran_pledged_artillery, capped at what
#     the Realm Authority law lets the host hold);
#   - the first time the ruler is at war (on_war_started / on_join_war_as_secondary in
#     aeluran_reinforcement_on_actions.txt) the pledge is spawned onto her at her capital and the regiment scopes
#     are kept in var:aeluran_war_magi_army_regiment / var:aeluran_artillery_army_regiment;
#   - from then on every delivery refills those regiments directly, raised or not.
# The host is inheritable; aeluran_reinforcements_on_death_effect hands the scopes and any pledge to the heir.
#################################################################################################################

# Root = the ruler. Called by aeluran_advisor_task.01011 (the first pledge, visible) and .01012 (hidden).
manage_aeluran_reinforcements_effect = {
	# the per-vassal breakdown the event widget lists
	clear_variable_list = aeluran_vassals
	every_vassal = {
		limit = { has_government = aeluran_weavers_government }
		prev = { add_to_variable_list = { name = aeluran_vassals target = prev } }
	}
	cp:councillor_aeluran_advisor ?= { save_scope_as = aeluran_advisor }
	# for the task description: when the Order comes again
	set_variable = { name = aeluran_next_magi_delivery_year value = aeluran_next_delivery_year_value }

	if = { # nothing feeds the host: no leased temple, no titled Order vassal
		limit = { total_aeluran_magi_reinforcements <= 0 }
		send_interface_message = {
			type = aeluran_army_reinforcements_arrived
			title = aeluran_reinforcements_none_title
			desc = aeluran_reinforcements_none_desc
			right_icon = scope:aeluran_advisor
		}
	}
	else = {
		# regiments the host already holds: let the law size them, then refill them
		aeluran_apply_host_size_effect = yes
		aeluran_refill_host_effect = yes
		# regiment types the host does not hold yet: pledge them, and deliver at once if already at war
		aeluran_pledge_magi_effect = yes
		if = {
			limit = {
				is_at_war = yes
				aeluran_has_pledged_magi_trigger = yes
			}
			aeluran_deliver_pledged_magi_effect = yes
		}

		if = {
			limit = { aeluran_host_exists_trigger = yes }
			if = {
				limit = { aeluran_host_is_full_trigger = yes }
				send_interface_message = {
					type = aeluran_army_reinforcements_arrived
					title = aeluran_reinforcements_arrived_title_max_capacity
					desc = aeluran_reinforcements_message_max_capacity
					right_icon = scope:aeluran_advisor
				}
			}
			else = {
				send_interface_message = {
					type = aeluran_army_reinforcements_arrived
					title = aeluran_reinforcements_arrived_title
					desc = {
						desc = aeluran_reinforcements_message_total_magi_reinforcements
						desc = aeluran_reinforcements_aeluran_war_magi_total
						desc = aeluran_reinforcements_aeluran_artillery_total
					}
					right_icon = scope:aeluran_advisor
				}
			}
		}
		else = {
			send_interface_message = {
				type = aeluran_army_reinforcements_arrived
				title = aeluran_reinforcements_pledged_title
				desc = aeluran_reinforcements_pledged_desc
				right_icon = scope:aeluran_advisor
			}
		}
	}
}

# Root = the ruler. Banks this delivery's magi for each regiment type the host does not hold yet, up to the law cap.
aeluran_pledge_magi_effect = {
	if = {
		limit = {
			NOT = { exists = var:aeluran_war_magi_army_regiment }
			aeluran_pledge_war_magi_add > 0
		}
		if = {
			limit = { has_variable = aeluran_pledged_war_magi }
			change_variable = { name = aeluran_pledged_war_magi add = aeluran_pledge_war_magi_add }
		}
		else = {
			set_variable = { name = aeluran_pledged_war_magi value = aeluran_pledge_war_magi_add }
		}
	}
	if = {
		limit = {
			NOT = { exists = var:aeluran_artillery_army_regiment }
			aeluran_pledge_artillery_add > 0
		}
		if = {
			limit = { has_variable = aeluran_pledged_artillery }
			change_variable = { name = aeluran_pledged_artillery add = aeluran_pledge_artillery_add }
		}
		else = {
			set_variable = { name = aeluran_pledged_artillery value = aeluran_pledge_artillery_add }
		}
	}
}

# Root = the ruler. Adds this delivery's magi to each regiment the host holds (the engine stops at the regiment's max).
aeluran_refill_host_effect = {
	save_scope_value_as = { name = aeluran_refill_war_magi value = aeluran_war_magi_reinforcements }
	save_scope_value_as = { name = aeluran_refill_artillery value = aeluran_artillery_reinforcements }
	var:aeluran_war_magi_army_regiment ?= { change_maa_troops_count = scope:aeluran_refill_war_magi }
	var:aeluran_artillery_army_regiment ?= { change_maa_troops_count = scope:aeluran_refill_artillery }
}

# Root = a ruler at war (on_war_started, on_join_war_as_secondary, or a delivery that falls during a war).
aeluran_try_deliver_pledged_magi_effect = {
	if = {
		limit = {
			is_at_war = yes
			aeluran_has_pledged_magi_trigger = yes
		}
		aeluran_deliver_pledged_magi_effect = yes
	}
}

# Scope = a ruler at war holding a pledge (not root: from the war on_actions root is the casus belli). Spawns the
# pledged magi at her capital as her own event troops (no war = , so the host stays after the war), keeps the
# regiment scopes, sizes them to the law and clears the pledge.
aeluran_deliver_pledged_magi_effect = {
	save_scope_as = aeluran_host_ruler
	if = {
		limit = { aeluran_army_level_value = 3 }
		aeluran_spawn_pledged_host_effect = { ARMY_NAME = stationed_aeluran_magi_3 }
	}
	else_if = {
		limit = { aeluran_army_level_value = 2 }
		aeluran_spawn_pledged_host_effect = { ARMY_NAME = stationed_aeluran_magi_2 }
	}
	else = {
		aeluran_spawn_pledged_host_effect = { ARMY_NAME = stationed_aeluran_magi_1 }
	}

	if = {
		limit = { exists = scope:aeluran_reinforcements }
		scope:aeluran_reinforcements = {
			random_army_maa_regiment = {
				limit = { is_maa_type = aeluran_war_magi }
				scope:aeluran_host_ruler = { set_variable = { name = aeluran_war_magi_army_regiment value = prev } }
			}
			random_army_maa_regiment = {
				limit = { is_maa_type = aeluran_artillery }
				scope:aeluran_host_ruler = { set_variable = { name = aeluran_artillery_army_regiment value = prev } }
			}
		}
		if = {
			limit = {
				exists = var:aeluran_war_magi_army_regiment
				has_variable = aeluran_pledged_war_magi
			}
			remove_variable = aeluran_pledged_war_magi
		}
		if = {
			limit = {
				exists = var:aeluran_artillery_army_regiment
				has_variable = aeluran_pledged_artillery
			}
			remove_variable = aeluran_pledged_artillery
		}
		aeluran_apply_host_size_effect = yes

		if = {
			limit = { is_ai = no }
			send_interface_toast = {
				type = new_aeluran_army_arrived
				title = new_aeluran_army_arrived_title
				left_icon = scope:aeluran_host_ruler
				custom_tooltip = {
					text = new_aeluran_army_arrived_desc
				}
			}
			scope:aeluran_reinforcements = { pan_camera_to_province = location }
		}
	}
	else = {
		debug_log = "aeluran_deliver_pledged_magi_effect: spawn_army spawned no army (ruler not at war?); the pledge is kept"
	}
}

# Root = the ruler. Spawns the pledged regiment types (men = the pledge) as one stack named $ARMY_NAME$ at the capital
# and saves it as scope:aeluran_reinforcements. One block per combination: spawn_army cannot skip a listed type.
aeluran_spawn_pledged_host_effect = {
	if = {
		limit = {
			aeluran_pledged_war_magi_value > 0
			aeluran_pledged_artillery_value > 0
		}
		spawn_army = {
			men_at_arms = { type = aeluran_war_magi men = aeluran_pledged_war_magi_value }
			men_at_arms = { type = aeluran_artillery men = aeluran_pledged_artillery_value }
			location = capital_province
			inheritable = yes
			uses_supply = yes
			name = $ARMY_NAME$
			save_scope_as = aeluran_reinforcements
		}
	}
	else_if = {
		limit = { aeluran_pledged_war_magi_value > 0 }
		spawn_army = {
			men_at_arms = { type = aeluran_war_magi men = aeluran_pledged_war_magi_value }
			location = capital_province
			inheritable = yes
			uses_supply = yes
			name = $ARMY_NAME$
			save_scope_as = aeluran_reinforcements
		}
	}
	else_if = {
		limit = { aeluran_pledged_artillery_value > 0 }
		spawn_army = {
			men_at_arms = { type = aeluran_artillery men = aeluran_pledged_artillery_value }
			location = capital_province
			inheritable = yes
			uses_supply = yes
			name = $ARMY_NAME$
			save_scope_as = aeluran_reinforcements
		}
	}
}

# Root = the ruler. Grows each regiment the host holds to the sub-regiment count the Realm Authority law allows
# (never shrinks: fewer sub-regiments would delete soldiers). Also called from the laws' on_pass, where the law's
# hand-over variable set_aeluran_army_level stands in for has_realm_law; the variable is cleared afterwards.
aeluran_apply_host_size_effect = {
	if = {
		limit = { exists = var:aeluran_war_magi_army_regiment }
		save_scope_value_as = { name = aeluran_host_size_delta value = aeluran_host_war_magi_size_delta }
		if = {
			limit = { scope:aeluran_host_size_delta > 0 }
			var:aeluran_war_magi_army_regiment = {
				change_maa_regiment_size = { size = scope:aeluran_host_size_delta reinforce = no }
			}
		}
	}
	if = {
		limit = { exists = var:aeluran_artillery_army_regiment }
		save_scope_value_as = { name = aeluran_host_size_delta value = aeluran_host_artillery_size_delta }
		if = {
			limit = { scope:aeluran_host_size_delta > 0 }
			var:aeluran_artillery_army_regiment = {
				change_maa_regiment_size = { size = scope:aeluran_host_size_delta reinforce = no }
			}
		}
	}
	if = {
		limit = { has_variable = set_aeluran_army_level }
		remove_variable = set_aeluran_army_level
	}
}

# Root = a ruler about to die (on_death). The host is inheritable, so the heir gets the regiments; hand her the
# scopes and any pledge too, or her next delivery would pledge and raise a second host beside the first.
aeluran_reinforcements_on_death_effect = {
	if = {
		limit = { exists = primary_heir }
		primary_heir = {
			if = {
				limit = {
					NOT = { has_variable = aeluran_war_magi_army_regiment }
					prev = { has_variable = aeluran_war_magi_army_regiment }
				}
				set_variable = { name = aeluran_war_magi_army_regiment value = prev.var:aeluran_war_magi_army_regiment }
			}
			if = {
				limit = {
					NOT = { has_variable = aeluran_artillery_army_regiment }
					prev = { has_variable = aeluran_artillery_army_regiment }
				}
				set_variable = { name = aeluran_artillery_army_regiment value = prev.var:aeluran_artillery_army_regiment }
			}
			if = {
				limit = { prev = { has_variable = aeluran_pledged_war_magi } }
				if = {
					limit = { has_variable = aeluran_pledged_war_magi }
					change_variable = { name = aeluran_pledged_war_magi add = prev.var:aeluran_pledged_war_magi }
				}
				else = {
					set_variable = { name = aeluran_pledged_war_magi value = prev.var:aeluran_pledged_war_magi }
				}
			}
			if = {
				limit = { prev = { has_variable = aeluran_pledged_artillery } }
				if = {
					limit = { has_variable = aeluran_pledged_artillery }
					change_variable = { name = aeluran_pledged_artillery add = prev.var:aeluran_pledged_artillery }
				}
				else = {
					set_variable = { name = aeluran_pledged_artillery value = prev.var:aeluran_pledged_artillery }
				}
			}
		}
	}
}

# Root = the ruler. Removes one regiment of the host and forgets it.
destroy_existing_magi_maa_regiment = {
	if = {
		limit = { exists = var:$REGIMENT_TYPE$_army_regiment }
		var:$REGIMENT_TYPE$_army_regiment = { destroy_maa_regiment = yes }
		remove_variable = $REGIMENT_TYPE$_army_regiment
	}
}

# Root = a ruler who has left the faith (aeluran_reinforcement_on_actions.txt). The Order takes its host back and
# forgets any pledge (PM 2026-10-07: only on leaving the faith).
aeluran_recall_host_effect = {
	destroy_existing_magi_maa_regiment = { REGIMENT_TYPE = aeluran_war_magi }
	destroy_existing_magi_maa_regiment = { REGIMENT_TYPE = aeluran_artillery }
	if = {
		limit = { has_variable = aeluran_pledged_war_magi }
		remove_variable = aeluran_pledged_war_magi
	}
	if = {
		limit = { has_variable = aeluran_pledged_artillery }
		remove_variable = aeluran_pledged_artillery
	}
	if = {
		limit = { is_ai = no }
		send_interface_message = {
			type = aeluran_host_recalled
			title = aeluran_host_recalled_title
			desc = aeluran_host_recalled_desc
			left_icon = root
		}
	}
}
