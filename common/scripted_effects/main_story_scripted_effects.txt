move_the_grand_portal_ruins_here_effect = {
	title_province = {
		set_variable = { # Tell the game that the Grand Portal has been transported
			name = title_grand_portal_transported
			value = yes
		}

		add_special_building_slot = holy_site_spark_grand_portal_01
		add_special_building = holy_site_spark_grand_portal_01
	}
}

remove_the_grand_portal_effect = {
	title_province = { # cant delete so replacing with a generic temple instead
		custom_tooltip = {
			text = remove_the_grand_portal_effect

			remove_building = holy_site_spark_grand_portal_01

			add_special_building_slot = holy_site_spark_grand_temple_01
			add_special_building = holy_site_spark_grand_temple_01
		}
	}
}

restore_the_grand_portal_scripted_effect = {
	# this is the County title.
	title_province = {
		set_variable = { # Tell the game that the Grand Portal has been restored
			name = title_grand_portal_restored
			value = yes
		}

		if = { # IF for some reason the special slot isnt there yet, add it and level 2 portal
			limit = {
				has_special_building_slot = no 
			}
			add_special_building_slot = holy_site_spark_grand_portal_02
			add_special_building = holy_site_spark_grand_portal_02
		}
		else_if = { # IF slot exists, its almost guaranteed to be level 1 portal, upgrade it (Expected Path)
			limit = {
				has_special_building = yes
			}
			remove_building = holy_site_spark_grand_portal_01
			add_special_building = holy_site_spark_grand_portal_02
		}
		else = {
			debug_log = "Unexpected result in restore_the_grand_portal_scripted_effect"
		}
	}
}

get_julia_melwood_effect = {
	if = {
		limit = {
			any_living_character = {
				has_character_flag = julia_melwood
			}
		}

		random_living_character = {
			limit = {
				has_character_flag = julia_melwood
			}

			save_scope_as = archeologist
		}
	}
	else = {
		# Julia keeps the creed of her homeland as it stands today (Paris): Carolingian Christianity in 867, the Roman
		# rite of the Catholic Church once the Great Schism has passed. Should Paris have left Christendom, the Roman rite.
		if = {
			limit = { title:c_ile_de_france.faith.religion = religion:christianity_religion }
			title:c_ile_de_france.rite = { save_temporary_scope_as = julia_rite }
		}
		else = {
			rite:roman_rite = { save_temporary_scope_as = julia_rite }
		}
		create_character = {
			template = julia_melwood
			faith = scope:julia_rite.faith
			rite = scope:julia_rite
			culture = culture:french
			# dynasty_house = 696976
			location = root.location
			save_scope_as = archeologist
		}
	}
}

# Julia Melwood follows her patron into a new faith, wherever she is: she joins the spouses_and_family_to_convert list
# (scope:actor's rite is the one being taken). Never creates her.
add_julia_melwood_to_converts_effect = {
	random_living_character = {
		limit = {
			has_character_flag = julia_melwood
			NOT = { rite = scope:actor.rite }
			NOT = { is_in_list = spouses_and_family_to_convert }
		}
		add_to_list = spouses_and_family_to_convert
	}
}

# The household on the spouses_and_family_to_convert list (built by grab_spouses_and_family_to_convert_effect) takes
# scope:actor's rite as welcomed converts. Since 1.20 vanilla's convert_family_to_faith_effect runs every convert
# through convert_to_rite_with_consequences_effect: whoever has not studied the rite loses their piety and devotion,
# gains massive stress and carries a five-year convert modifier, and courtiers can never study a rite.
welcome_household_to_faith_effect = {
	every_in_list = {
		list = spouses_and_family_to_convert
		clean_convert_modifier_effect = yes
		set_character_rite = scope:actor.rite
	}
	scope:recipient = { bastard_to_wild_oat_conversion_interaction_effect = yes }
}

# Scope: the barony the Grand Portal is moved to; root = the ruler moving it.
# 1.20 holy sites can be re-created elsewhere: the eminent Grand Portal holy site follows the portal, its relics
# re-enshrined there (a site without one is inactive). The religion's main holy site link stays on Spirits' Landing.
move_the_grand_portal_holy_site_effect = {
	save_scope_as = new_portal_barony
	holy_site:grand_portal ?= {
		every_holy_site_enshrined_artifact = { add_to_temporary_list = grand_portal_relics }
	}
	faith:aeluran_weavers = {
		if = {
			limit = { any_holy_site = { this = holy_site:grand_portal } }
			remove_holy_site = holy_site:grand_portal
		}
		create_holy_site = {
			type = grand_portal
			eminent = yes
			barony = scope:new_portal_barony
			county = scope:new_portal_barony.county
			actor = root
		}
	}
	# Re-enshrined as vanilla does for starting relics (finalize_starting_relic_effect): owner set and equipped,
	# skipping the enshrine rules (the mover need not own the relic or hold the county)
	every_in_list = {
		list = grand_portal_relics
		set_holy_site_owner = {
			target = holy_site:grand_portal
			generate_history = no
		}
		equip_artifact_to_owner = yes
	}
	# The expedition portal exclusion and the restore decision look for the portal here
	global_var:portal_location_barony ?= { remove_variable = portal_location }
	set_variable = portal_location
	set_global_variable = {
		name = portal_location_barony
		value = this
	}
}
