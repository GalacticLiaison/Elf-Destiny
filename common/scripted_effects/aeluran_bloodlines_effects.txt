# The Bloodlines board: a shepherd of the Order (a Matron holding a Matronate, or the High Matriarch) matches the unwed
# noble elves in her care. (Matron gameplay Phase C, reports/proposals/2026-09-30-matron-gameplay-proposal.md.)
#
# A proposal goes to each ward's matchmaker (their guardian: themselves if they rule). An AI guardian answers at once,
# by opinion of the shepherd, the partner's blood and their own devotion; a player guardian gets aeluran_bloodlines.001.
# Accepted: two adults wed, otherwise they are betrothed; the shepherd gains piety by the pair's worth, and again when a
# child of the match is born gifted. Refused: neither ward can be proposed again for five years.

# Root = the shepherd. Rebuilds her board (var list aeluran_bloodlines), the worthiest first.
aeluran_bloodlines_build_list_effect = {
	clear_variable_list = aeluran_bloodlines
	save_scope_as = bl_shepherd
	ordered_living_character = {
		limit = {
			allowed_more_spouses = yes
			is_betrothed = no
			aeluran_bloodlines_candidate_trigger = { SHEPHERD = scope:bl_shepherd }
		}
		order_by = genetic_worthiness
		min = 0
		max = 99
		check_range_bounds = no
		# Their spouses, for the row's spouse portraits
		save_temporary_scope_as = bl_noble
		clear_variable_list = aeluran_bl_spouses
		every_spouse = { scope:bl_noble = { add_to_variable_list = { name = aeluran_bl_spouses target = prev } } }
		scope:bl_shepherd = { add_to_variable_list = { name = aeluran_bloodlines target = prev } }
	}
}

# Root = the shepherd; scope:bl_a and scope:bl_b = the pair. Sends the proposal to both guardians.
aeluran_bloodlines_propose_effect = {
	save_scope_as = bl_matron
	clear_saved_scope = bl_declined
	set_variable = { name = aeluran_bl_pending value = 0 }
	scope:bl_a = { save_scope_as = bl_ward }
	scope:bl_b = { save_scope_as = bl_partner }
	aeluran_bloodlines_ask_guardian_effect = yes
	scope:bl_b = { save_scope_as = bl_ward }
	scope:bl_a = { save_scope_as = bl_partner }
	aeluran_bloodlines_ask_guardian_effect = yes

	if = {
		limit = { exists = scope:bl_declined }
		aeluran_bloodlines_refused_effect = yes
	}
	else_if = {
		limit = { var:aeluran_bl_pending <= 0 }
		aeluran_bloodlines_finalize_effect = yes
	}
	else = {
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = aeluran_bloodlines_proposed_toast
			left_icon = scope:bl_a
			right_icon = scope:bl_b
			custom_tooltip = aeluran_bloodlines_proposed_toast_desc
		}
	}
}

# Scope:bl_ward's guardian answers. Needs scope:bl_matron, scope:bl_ward, scope:bl_partner, scope:bl_a, scope:bl_b.
aeluran_bloodlines_ask_guardian_effect = {
	scope:bl_ward.matchmaker ?= { save_scope_as = bl_guardian }
	if = {
		limit = { NOT = { exists = scope:bl_guardian } }
		scope:bl_ward = { save_scope_as = bl_guardian }
	}
	# The shepherd needs no one's leave for her own wards
	if = {
		limit = {
			NOT = { scope:bl_guardian = scope:bl_matron }
			scope:bl_guardian = { is_ai = yes }
			# A player's proposal (scope:bl_gated) was only sendable with this guardian's acceptance above 0
			NOT = { exists = scope:bl_gated }
		}
		random = {
			chance = {
				value = 100
				subtract = scope:bl_guardian.aeluran_bloodlines_accept_chance
			}
			save_scope_value_as = { name = bl_declined value = yes }
			scope:bl_guardian = { save_scope_as = bl_decliner }
		}
	}
	else_if = {
		limit = { NOT = { scope:bl_guardian = scope:bl_matron } }
		scope:bl_matron = { change_variable = { name = aeluran_bl_pending add = 1 } }
		scope:bl_guardian = { trigger_event = aeluran_bloodlines.001 }
	}
	clear_saved_scope = bl_guardian
}

# A player guardian answered yes (aeluran_bloodlines.001). When no answer is still awaited, the match is made.
aeluran_bloodlines_player_accepted_effect = {
	scope:bl_matron = {
		change_variable = { name = aeluran_bl_pending add = -1 }
		if = {
			limit = {
				var:aeluran_bl_pending <= 0
				scope:bl_a = { aeluran_bloodlines_pair_trigger = { OTHER = scope:bl_b } }
				scope:bl_a = { allowed_more_spouses = yes is_betrothed = no }
				scope:bl_b = { allowed_more_spouses = yes is_betrothed = no }
			}
			aeluran_bloodlines_finalize_effect = yes
		}
	}
}

# The match is refused (an AI guardian said no, or a player guardian declined).
aeluran_bloodlines_refused_effect = {
	scope:bl_a = { set_variable = { name = aeluran_bloodlines_refused value = yes years = 5 } }
	scope:bl_b = { set_variable = { name = aeluran_bloodlines_refused value = yes years = 5 } }
	scope:bl_matron = {
		set_variable = { name = aeluran_bl_pending value = 0 }
		send_interface_toast = {
			type = event_toast_effect_bad
			title = aeluran_bloodlines_refused_toast
			left_icon = scope:bl_a
			right_icon = scope:bl_b
			custom_tooltip = aeluran_bloodlines_refused_toast_desc
		}
	}
}

# The match is made. Two adults wed, otherwise they are betrothed. Matrilineal as the proposer chose on the Matchmaking
# panel (scope:bl_matrilineal_choice); without a choice (AI), when the woman rules and outranks the man.
aeluran_bloodlines_finalize_effect = {
	if = {
		limit = { scope:bl_a = { is_female = yes } }
		scope:bl_a = { save_scope_as = bl_woman }
		scope:bl_b = { save_scope_as = bl_man }
	}
	else = {
		scope:bl_b = { save_scope_as = bl_woman }
		scope:bl_a = { save_scope_as = bl_man }
	}
	if = {
		limit = { exists = scope:bl_matrilineal_choice }
		if = {
			limit = { scope:bl_matrilineal_choice = flag:matrilineal }
			save_scope_value_as = { name = bl_matrilineal value = yes }
		}
	}
	else_if = {
		limit = {
			scope:bl_woman = { is_ruler = yes }
			OR = {
				scope:bl_man = { is_ruler = no }
				scope:bl_woman.highest_held_title_tier > scope:bl_man.highest_held_title_tier
			}
		}
		save_scope_value_as = { name = bl_matrilineal value = yes }
	}
	if = {
		limit = {
			scope:bl_woman = { is_adult = yes }
			scope:bl_man = { is_adult = yes }
		}
		if = {
			limit = { exists = scope:bl_matrilineal }
			scope:bl_woman = { marry_matrilineal = scope:bl_man }
		}
		else = {
			scope:bl_man = { marry = scope:bl_woman }
		}
	}
	else = {
		if = {
			limit = { exists = scope:bl_matrilineal }
			scope:bl_woman = { create_betrothal_matrilineal = scope:bl_man }
		}
		else = {
			scope:bl_man = { create_betrothal = scope:bl_woman }
		}
	}
	scope:bl_a = { set_variable = { name = aeluran_order_match value = scope:bl_matron years = 60 } }
	scope:bl_b = { set_variable = { name = aeluran_order_match value = scope:bl_matron years = 60 } }
	scope:bl_matron = {
		set_variable = { name = aeluran_bl_pending value = 0 }
		add_piety = aeluran_bloodlines_match_piety
		send_interface_toast = {
			type = event_toast_effect_good
			title = aeluran_bloodlines_match_made_toast
			left_icon = scope:bl_a
			right_icon = scope:bl_b
			custom_tooltip = aeluran_bloodlines_match_made_toast_desc
		}
	}
	scope:bl_a.matchmaker ?= {
		if = {
			limit = { NOT = { this = scope:bl_matron } }
			add_opinion = { target = scope:bl_matron modifier = aeluran_arranged_a_fine_match_opinion }
		}
	}
	scope:bl_b.matchmaker ?= {
		if = {
			limit = {
				NOT = { this = scope:bl_matron }
				NOT = { this = scope:bl_a.matchmaker }
			}
			add_opinion = { target = scope:bl_matron modifier = aeluran_arranged_a_fine_match_opinion }
		}
	}
}

# Root = an AI shepherd, yearly. Matches the worthiest unwed elf in her care with the worthiest partner for them.
aeluran_bloodlines_ai_yearly_effect = {
	save_scope_as = bl_shepherd
	clear_saved_scope = bl_a
	clear_saved_scope = bl_b
	ordered_living_character = {
		limit = {
			allowed_more_spouses = yes
			is_betrothed = no
			aeluran_bloodlines_candidate_trigger = { SHEPHERD = scope:bl_shepherd }
		}
		order_by = genetic_worthiness
		save_scope_as = bl_a
	}
	scope:bl_a ?= {
		ordered_living_character = {
			limit = {
				allowed_more_spouses = yes
				is_betrothed = no
				# A partner from anywhere in the Order
				aeluran_bloodlines_eligible_trigger = { SHEPHERD = scope:bl_shepherd }
				aeluran_bloodlines_pair_trigger = { OTHER = scope:bl_a }
			}
			order_by = genetic_worthiness
			save_scope_as = bl_b
		}
	}
	if = {
		limit = { exists = scope:bl_b }
		aeluran_bloodlines_propose_effect = yes
	}
}

# Root = the shepherd; var:aeluran_bl_first = the noble she chose. Rebuilds the matches for them (var list
# aeluran_bl_matches): eligible nobles from anywhere in the Order (every Matronate, and the High Matriarch's care), and
# Sisters of her own care noble or not, they
# could wed, the worthiest 40 first. Also records each match's spouses for the row's spouse portraits.
aeluran_bloodlines_build_matches_effect = {
	clear_variable_list = aeluran_bl_matches
	if = {
		limit = { exists = var:aeluran_bl_first }
		save_scope_as = bl_shepherd
		var:aeluran_bl_first = { save_scope_as = bl_chosen }
		ordered_living_character = {
			limit = {
				NOT = { this = scope:bl_chosen }
				aeluran_bloodlines_match_trigger = { SHEPHERD = scope:bl_shepherd }
				aeluran_bloodlines_pair_trigger = { OTHER = scope:bl_chosen }
			}
			order_by = genetic_worthiness
			min = 0
			max = 40
			check_range_bounds = no
			save_temporary_scope_as = bl_noble
			clear_variable_list = aeluran_bl_spouses
			every_spouse = { scope:bl_noble = { add_to_variable_list = { name = aeluran_bl_spouses target = prev } } }
			# Their Matronate (by capital if they rule land, else by where they live), for the row: unset = the High
			# Matriarch's care
			if = { limit = { has_variable = aeluran_bl_home } remove_variable = aeluran_bl_home }
			if = { limit = { has_variable = aeluran_bl_matron } remove_variable = aeluran_bl_matron }
			if = {
				limit = { is_landed = yes capital_county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
				set_variable = { name = aeluran_bl_home value = capital_county.clerical_region_title }
			}
			else_if = {
				limit = { is_landed = no location.county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
				set_variable = { name = aeluran_bl_home value = location.county.clerical_region_title }
			}
			# Her Matron, for the row's portrait
			var:aeluran_bl_home.holder ?= { save_temporary_scope_as = bl_home_matron }
			if = {
				limit = { exists = scope:bl_home_matron }
				set_variable = { name = aeluran_bl_matron value = scope:bl_home_matron }
			}
			scope:bl_shepherd = { add_to_variable_list = { name = aeluran_bl_matches target = prev } }
		}
	}
}

# Root = the shepherd, with var:aeluran_bl_first and var:aeluran_bl_second picked on the Matchmaking panel. Defaults the
# marriage type the way an AI match would be made (matrilineal when the woman rules and outranks the man).
aeluran_bloodlines_default_marriage_type_effect = {
	if = { limit = { has_variable = aeluran_bl_matrilineal } remove_variable = aeluran_bl_matrilineal }
	if = {
		limit = { exists = var:aeluran_bl_first exists = var:aeluran_bl_second }
		if = {
			limit = { var:aeluran_bl_first = { is_female = yes } }
			var:aeluran_bl_first = { save_scope_as = bl_woman }
			var:aeluran_bl_second = { save_scope_as = bl_man }
		}
		else = {
			var:aeluran_bl_second = { save_scope_as = bl_woman }
			var:aeluran_bl_first = { save_scope_as = bl_man }
		}
		if = {
			limit = {
				scope:bl_woman = { is_ruler = yes }
				OR = {
					scope:bl_man = { is_ruler = no }
					scope:bl_woman.highest_held_title_tier > scope:bl_man.highest_held_title_tier
				}
			}
			set_variable = { name = aeluran_bl_matrilineal value = yes }
		}
	}
	aeluran_bloodlines_update_house_parent_effect = yes
}

# Root = the shepherd. var:aeluran_bl_house_parent = whose house the children of the picked pair would be born into: the
# woman's when matrilineal (var:aeluran_bl_matrilineal), else the man's. For the panel's marriage-type line.
aeluran_bloodlines_update_house_parent_effect = {
	aeluran_bloodlines_score_effect = yes
	if = { limit = { has_variable = aeluran_bl_house_parent } remove_variable = aeluran_bl_house_parent }
	if = {
		limit = { exists = var:aeluran_bl_first exists = var:aeluran_bl_second }
		if = {
			limit = {
				OR = {
					AND = { has_variable = aeluran_bl_matrilineal var:aeluran_bl_first = { is_female = yes } }
					AND = { NOT = { has_variable = aeluran_bl_matrilineal } var:aeluran_bl_first = { is_female = no } }
				}
			}
			set_variable = { name = aeluran_bl_house_parent value = var:aeluran_bl_first }
		}
		else = {
			set_variable = { name = aeluran_bl_house_parent value = var:aeluran_bl_second }
		}
	}
}

# Root = the shepherd. Records the parties of the picked pair for the Matchmaking panel: each partner's guardian
# (var:aeluran_bl_guardian_a / _b: their matchmaker, or themselves), and forgets hooks chosen for an earlier pair.
aeluran_bloodlines_update_parties_effect = {
	clear_variable_list = aeluran_bl_hooks
	if = { limit = { has_variable = aeluran_bl_guardian_a } remove_variable = aeluran_bl_guardian_a }
	if = { limit = { has_variable = aeluran_bl_guardian_b } remove_variable = aeluran_bl_guardian_b }
	var:aeluran_bl_first ?= {
		if = {
			limit = { exists = matchmaker }
			matchmaker = { save_scope_as = bl_party }
		}
		else = { save_scope_as = bl_party }
	}
	scope:bl_party ?= { root = { set_variable = { name = aeluran_bl_guardian_a value = prev } } }
	clear_saved_scope = bl_party
	var:aeluran_bl_second ?= {
		if = {
			limit = { exists = matchmaker }
			matchmaker = { save_scope_as = bl_party }
		}
		else = { save_scope_as = bl_party }
	}
	scope:bl_party ?= { root = { set_variable = { name = aeluran_bl_guardian_b value = prev } } }
	clear_saved_scope = bl_party
	aeluran_bloodlines_score_effect = yes
}

# Root = the shepherd, after a player's proposal (scopes bl_a, bl_b, bl_guardian_b, bl_match_matron). The answer from the
# match's side, as a letter: from her Matron when that is not the shepherd herself, else from her guardian, else from her.
# Only for an AI guardian (a player guardian answers through aeluran_bloodlines.001).
aeluran_bloodlines_answer_letter_effect = {
	if = {
		limit = {
			scope:bl_guardian_b ?= { is_ai = yes }
			NOT = { scope:bl_guardian_b = root }
		}
		if = {
			limit = {
				exists = scope:bl_match_matron
				NOT = { scope:bl_match_matron = root }
			}
			scope:bl_match_matron = { save_scope_as = bl_letter_sender }
			save_scope_value_as = { name = bl_letter_role value = flag:matron }
		}
		else_if = {
			limit = { NOT = { scope:bl_guardian_b = scope:bl_b } }
			scope:bl_guardian_b = { save_scope_as = bl_letter_sender }
			save_scope_value_as = { name = bl_letter_role value = flag:guardian }
		}
		else = {
			scope:bl_b = { save_scope_as = bl_letter_sender }
			save_scope_value_as = { name = bl_letter_role value = flag:self }
		}
		trigger_event = aeluran_bloodlines.010
	}
}

# Root = the shepherd. Works out both guardians' acceptance (aeluran_mm_accept_value) for the picked pair and keeps it in
# var:aeluran_bl_score_a / _b for the Matchmaking panel's score lines. Run whenever the pair, the hooks or the marriage
# type change.
aeluran_bloodlines_score_effect = {
	if = { limit = { has_variable = aeluran_bl_score_a } remove_variable = aeluran_bl_score_a }
	if = { limit = { has_variable = aeluran_bl_score_b } remove_variable = aeluran_bl_score_b }
	if = {
		limit = {
			exists = var:aeluran_bl_first
			exists = var:aeluran_bl_second
			exists = var:aeluran_bl_guardian_a
			exists = var:aeluran_bl_guardian_b
		}
		save_scope_as = mm_shepherd
		# Our side: the noble's guardian answers
		var:aeluran_bl_guardian_a = { save_scope_as = recipient }
		var:aeluran_bl_first = { save_scope_as = secondary_recipient }
		var:aeluran_bl_guardian_b = { save_scope_as = actor save_scope_as = puppet_or_actor }
		var:aeluran_bl_second = { save_scope_as = secondary_actor }
		save_scope_as = mm_matron
		scope:recipient = { save_temporary_scope_value_as = { name = mm_score value = aeluran_mm_accept_value } }
		set_variable = { name = aeluran_bl_score_a value = scope:mm_score }
		# Their side: the match's guardian answers, under the eye of her Matron
		var:aeluran_bl_guardian_b = { save_scope_as = recipient }
		var:aeluran_bl_second = { save_scope_as = secondary_recipient }
		var:aeluran_bl_guardian_a = { save_scope_as = actor save_scope_as = puppet_or_actor }
		var:aeluran_bl_first = { save_scope_as = secondary_actor }
		clear_saved_scope = mm_matron
		var:aeluran_bl_match_matron ?= { save_scope_as = mm_matron }
		scope:recipient = { save_temporary_scope_value_as = { name = mm_score_b value = aeluran_mm_accept_value } }
		set_variable = { name = aeluran_bl_score_b value = scope:mm_score_b }
	}
}
