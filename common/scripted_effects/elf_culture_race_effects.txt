# Character scope, guarded by elf_needs_culture_race_trigger: make the character an elf of their culture's kind.
# Drow, Dokkalfar and the basic dark elf culture become dark elves, every other elf-heritage culture standard elves. Only the race is set: their
# DNA and look already come from the culture's ethnicity (never set var:dark_elf_skin_color here, that is for converts).
elf_apply_culture_race_effect = {
	if = {
		limit = {
			OR = {
				culture = culture:elf_culture_drow
				culture = culture:elf_culture_dokkalfar
				culture = culture:elf_culture_dark_elf
			}
		}
		add_character_flag = race_dark_elf
	}
	else = {
		add_character_flag = race_elf
	}
	urf_for_all_phenotypes = { APPLY = urf_base_race_init }
	set_appropriate_elf_immortality_age = yes
}
