create_elf_kingdom_effect = {
	save_scope_as = founder
	primary_title = {
		save_scope_as = old_title
		# Remember the de jure empire before the duchies are moved out of their old kingdom
		empire ?= { save_scope_as = old_empire }
	}

	debug_log = "create_elf_kingdom_effect"
	debug_log_scopes = yes

	create_dynamic_title = {
		tier = kingdom
		name = NEW_CREATED_TITLE_NAME
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	
	scope:new_title = {
		change_title_holder = {
			holder = scope:founder
			change = scope:title_change
		}
	}

	debug_log_scopes = yes
	
	resolve_title_and_vassal_change = scope:title_change

	every_sub_realm_duchy = {
		set_de_jure_liege_title = scope:new_title
	}

	scope:new_title = {
		set_coa = scope:old_title
		set_color_from_title = scope:old_title
		set_capital_county = scope:old_title.title_capital_county
		# Keep the new kingdom inside an empire: a kingdom with no de jure empire leaves its
		# counties with a null `empire`, which vanilla scripts (found_empire_decision,
		# tournaments) dereference unguarded
		if = {
			limit = { exists = scope:old_empire }
			set_de_jure_liege_title = scope:old_empire
		}
	}
	set_primary_title_to = scope:new_title
}

# Dark elf realms: map colour and name, set at game start (dark_elf_on_actions.txt).
# Do NOT move this back into common/landed_titles: re-declaring a vanilla title there, even
# only for its colour, cuts it out of its de jure liege and out of every geographical region
# that names it. Menzoberranzan had no kingdom, Valusia and Svartalfaheim no empire, and the
# guests the game generates for those courts at start came out with no culture and no name.
# Svartalfaheim = near-black; Menzoberranzan = dark blood red (the Myrvessa blood-red hair
# mark); Valusia = serpent emerald. The names stay with the title whoever holds it later.
# Display names: localization/english/dark_elf_titles_l_english.yml.
dark_elf_realm_titles_effect = {
	title:k_sapmi = {
		set_title_color = { 26 27 33 }
		set_title_name = cn_svartalfaheim
	}

		set_title_color = { 40 7 9 }
		set_title_name = cn_menzoberranzan
	}
	title:k_kamarupa = {
		set_title_color = { 28 66 50 }
		set_title_name = cn_valusia
	}
	title:d_kamarupanagara = {
		set_title_color = { 34 74 56 }
		set_title_name = cn_kamula
	}
}
