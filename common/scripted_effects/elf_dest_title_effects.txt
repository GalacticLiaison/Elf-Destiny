create_elf_kingdom_effect = {
	save_scope_as = founder
	primary_title = {
		save_scope_as = old_title
		# Remember the de jure empire before the duchies are moved out of their old kingdom
		empire ?= { save_scope_as = old_empire }
	}

	debug_log = "create_elf_kingdom_effect"
	debug_log_scopes = yes

	create_dynamic_title = {
		tier = kingdom
		name = NEW_CREATED_TITLE_NAME
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	
	scope:new_title = {
		change_title_holder = {
			holder = scope:founder
			change = scope:title_change
		}
	}

	debug_log_scopes = yes
	
	resolve_title_and_vassal_change = scope:title_change

	every_sub_realm_duchy = {
		set_de_jure_liege_title = scope:new_title
	}

	scope:new_title = {
		set_coa = scope:old_title
		set_color_from_title = scope:old_title
		set_capital_county = scope:old_title.title_capital_county
		# Keep the new kingdom inside an empire: a kingdom with no de jure empire leaves its
		# counties with a null `empire`, which vanilla scripts (found_empire_decision,
		# tournaments) dereference unguarded
		if = {
			limit = { exists = scope:old_empire }
			set_de_jure_liege_title = scope:old_empire
		}
	}
	set_primary_title_to = scope:new_title
}
