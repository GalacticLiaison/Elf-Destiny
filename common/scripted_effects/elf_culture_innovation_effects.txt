# Give Elven Adaptability to every culture of elven heritage that lacks it (elf_culture_innovation_on_actions.txt).
elf_culture_innovation_sweep_effect = {
	every_culture_global = {
		limit = {
			has_cultural_pillar = heritage_elf
			NOT = { has_innovation = innovation_elven_adaptability }
		}
		add_innovation = innovation_elven_adaptability
	}
}
