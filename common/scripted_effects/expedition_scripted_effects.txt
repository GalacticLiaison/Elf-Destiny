register_expedition_site = {
	title:$BARONY_NAME$ = {
		set_variable = valid_expedition_location
		set_variable = {
			name = expedition_site_loc
			value = flag:expedition_site_$SITE_NAME$_desc
		}
		set_variable = {
			name = expedition_holy_site
			value = flag:$SITE_NAME$
		}
		# The holy site this barony becomes once an expedition discovers it
		set_variable = {
			name = expedition_holy_site_scope
			value = holy_site:$HOLY_SITE$
		}
	}
}

maunally_set_first_expedition_site = {
	if = {
		limit = {
			is_target_in_global_variable_list = {
				name = first_expedition_location_candidates
				target = title:$BARONY_NAME$
			}
		}

		remove_list_global_variable = {
			name = first_expedition_location_candidates
			target = title:$BARONY_NAME$
		}

		title:$BARONY_NAME$ = {
			set_variable = valid_expedition_location
			set_variable = {
				name = expedition_site_loc
				value = flag:expedition_site_fortress_tomb_of_the_dead_emperor_desc
			}
			set_variable = {
				name = expedition_holy_site
				value = flag:fortress_tomb_of_the_dead_emperor
			}
		}
	
	}
}

# Discovered expedition sites become holy sites of the Aeluran Weavers (never the discoverer's own faith).
# Expects scope:activity; root is the expedition host.
activate_discovered_holy_site_effect = {
	scope:activity.activity_location.barony ?= {
		expedition_activate_barony_holy_site_effect = yes
	}
}

# Scope: the expedition site barony. Root is the discovering character.
expedition_activate_barony_holy_site_effect = {
	if = {
		limit = { exists = var:expedition_holy_site_scope }
		var:expedition_holy_site_scope = { save_scope_as = discovered_holy_site }
	}

	if = {
		limit = { exists = scope:discovered_holy_site }

		# add_holy_site errors if the faith already has the site (repeat expeditions, the Grand Portal)
		if = {
			limit = {
				NOT = {
					faith:aeluran_weavers = {
						any_holy_site = { this = scope:discovered_holy_site }
					}
				}
			}
			faith:aeluran_weavers = {
				add_holy_site = {
					target = scope:discovered_holy_site
					actor = root
				}
			}
		}

		if = {
			limit = {
				scope:discovered_holy_site = { holy_site_has_artifact = no }
			}
			expedition_enshrine_placeholder_relic_effect = yes
		}
	}
}

# PLACEHOLDER until site-specific relics exist: a common Aeluran relic enshrined in the discovered site.
# Uses vanilla's generic holy relic setup: branch_general_template counts an artifact as a holy relic
# for var:relic_religion, and finalize_starting_relic_effect enshrines it (skips if the site is full).
expedition_enshrine_placeholder_relic_effect = {
	scope:discovered_holy_site = { save_scope_as = relic_site }
	faith:aeluran_weavers = { save_scope_as = relic_faith }
	root = { save_scope_as = relic_owner }

	set_artifact_rarity_common = yes
	scope:relic_owner = {
		create_artifact = {
			name = artifact_expedition_placeholder_relic_name
			description = artifact_expedition_placeholder_relic_desc
			type = pedestal
			template = branch_general_template
			visuals = reliquary
			wealth = scope:wealth
			quality = scope:quality
			history = {
				type = discovered
				actor = scope:relic_owner
				recipient = scope:relic_owner
				location = scope:relic_site.holy_site_barony.title_province
			}
			modifier = artifact_placeholder_modifier
			save_scope_as = newly_created_artifact
		}
	}
	scope:newly_created_artifact = {
		set_variable = {
			name = relic_religion
			value = scope:relic_faith.religion
		}
	}
	finalize_starting_relic_effect = yes
}
