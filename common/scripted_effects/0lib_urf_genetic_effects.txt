urf_set_race_phenotype = {
    set_variable = {
        name = urf_phenotype
        value = flag:$RACE_PHENOTYPE$
    }
}

urf_set_genotype = {
    set_variable = {
        name = urf_genotype_$RACE$
        value = $VAL$
    }
}

urf_set_genotype_base = {
    urf_set_genotype = { RACE = $RACE$ VAL = 100.0 }
}

urf_genotype_phenotype_threshold = {
    if = {
        limit = {
            urf_race_initialized = no # Elf Destiny
            urf_racial_threshold_trigger = { RACE = $RACE$ }
        }
        # Elf Destiny
        if = {
            limit = {
                flag:$RACE$ = flag:elf
            }
            urf_elf_race_sub_init = yes
        }
        # End Elf Destiny

        else = {
            urf_set_race_phenotype = {
                RACE_PHENOTYPE = $RACE$
            }
            urf_base_modifier_init = { RACE = $RACE$ }
        }

    }
}

urf_inherit_genotype_assignment = {
    $CHILD$ = {
        set_variable = {
            name = urf_genotype_$RACE$
            value = {
                value = 0
                # Elf Destiny
                if = {
                    limit = {
                        $MOTHER$ ?= {
                            is_alive = yes
                            has_variable = urf_genotype_$RACE$
                        }
                    }

                    add = $MOTHER$.var:urf_genotype_$RACE$
                }
                else = {
                    add = 0
                }

                if = {
                    limit = {
                        $FATHER$ ?= {
                            is_alive = yes
                            has_variable = urf_genotype_$RACE$
                        }
                    }

                    add = $FATHER$.var:urf_genotype_$RACE$
                }
                else = {
                    add = 0
                }

                divide = 2

                # Controlled variance (PM 2026-10-01): up to 4 either way (half an elf tier is 4.17), then kept
                # between the two parents, so a child is never below its lower parent (two elves never have a
                # half-elf) and an elf x high elf child (on the tier boundary) can come out either way
                add = { fixed_range = { min = -4 max = 4 } }
                # Not below the lower parent (a missing parent counts as 0, as above)
                min = {
                    value = 0
                    if = {
                        limit = { $MOTHER$ ?= { is_alive = yes has_variable = urf_genotype_$RACE$ } }
                        add = $MOTHER$.var:urf_genotype_$RACE$
                    }
                    max = {
                        value = 0
                        if = {
                            limit = { $FATHER$ ?= { is_alive = yes has_variable = urf_genotype_$RACE$ } }
                            add = $FATHER$.var:urf_genotype_$RACE$
                        }
                    }
                }
                # Not above the higher parent
                max = {
                    value = 0
                    if = {
                        limit = { $MOTHER$ ?= { is_alive = yes has_variable = urf_genotype_$RACE$ } }
                        add = $MOTHER$.var:urf_genotype_$RACE$
                    }
                    min = {
                        value = 0
                        if = {
                            limit = { $FATHER$ ?= { is_alive = yes has_variable = urf_genotype_$RACE$ } }
                            add = $FATHER$.var:urf_genotype_$RACE$
                        }
                    }
                }
                #End Elf Destiny
            }
        }
    }
}
