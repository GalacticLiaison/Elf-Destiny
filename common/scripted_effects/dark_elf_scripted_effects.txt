set_aeluran_clothing_colors = {
	if = {
		limit = {
			NOT = {
				has_character_flag = ignore_aeluran_clothes
			}
		}

		if = {
			limit = {
				# The tenet decides, not culture or race (PM 2026-10-05): a dark elf on the Aeluran Sisters tenet wears red
				aeluran_follows_dark_sisters_trigger = yes
			}

			set_black_aeluran_clothing_colors = yes
		}
		else = {
			manual_set_clothes_colors = {
				channel = 1
				RED = 0.5
				GREEN = 0
				BLUE = 0
			}
			manual_set_clothes_colors = {
				channel = 2
				RED = 0.5
				GREEN = 0
				BLUE = 0
			}
		}

	}
}

set_black_aeluran_clothing_colors = {
	manual_set_clothes_colors = {
		channel = 1
		RED = 0
		GREEN = 0
		BLUE = 0
	}
	manual_set_clothes_colors = {
		channel = 2
		RED = 0
		GREEN = 0
		BLUE = 0
	}
}
