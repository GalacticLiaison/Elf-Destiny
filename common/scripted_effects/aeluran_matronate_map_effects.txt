# Red Sister temple markers: a marker situation (aeluran_rs_temple_marker, common/situation/situations/) per temple a
# Red Sister keeps, centred on the temple so the engine draws its pin there (the only way to pin an arbitrary barony).
# A Sister keeps a temple when her ruler's lease hands it to her (common/lease_contracts/elf_dest_aeluran_lease.txt), or,
# in saves from before Beta_v2.0.4, when she holds it herself (aeluran_temple_kept_by_sister_trigger).
# Markers live in the global list aeluran_rs_temple_markers; each temple province points at its marker in
# var:aeluran_rs_temple_marker, and each marker carries its Sister in var:aeluran_rs_temple_sister. The pin itself is in
# gui/map_icon_layer.gui and draws only for Order members (aeluran_order_sees_temple_cards).
# Leases fire no on_action, so markers are placed when A Matter of Land settles, on each ruler's quarterly pulse, after
# the lobby and yearly (common/on_action/aeluran_matronate_map_on_actions.txt); each marker closes itself monthly.

# Full pass, any scope: close stale markers, then open one for every kept temple without one
aeluran_matronate_map_sync_temple_markers_effect = {
	every_in_global_list = {
		variable = aeluran_rs_temple_markers
		aeluran_rs_temple_marker_check_effect = yes
	}
	aeluran_rs_temple_markers_close_stale_effect = yes
	every_ruler = {
		limit = { exists = cp:councillor_aeluran_advisor }
		aeluran_rs_temple_markers_for_ruler_effect = yes
	}
}

# Scope = a ruler. Opens a marker on every temple her Red Sister keeps that has none.
aeluran_rs_temple_markers_for_ruler_effect = {
	clear_saved_scope = rs_sister
	cp:councillor_aeluran_advisor ?= { save_scope_as = rs_sister }
	if = {
		limit = { exists = scope:rs_sister }
		every_sub_realm_barony = {
			limit = {
				aeluran_temple_kept_by_sister_trigger = { SISTER = scope:rs_sister }
				title_province = { NOT = { exists = var:aeluran_rs_temple_marker } }
			}
			title_province = { aeluran_rs_temple_marker_start_effect = yes }
		}
	}
}

# Scope = a marker situation. Lists it for closing when its Sister is dead, off the Red Sister seat or no longer keeps
# the temple. Closing happens in aeluran_rs_temple_markers_close_stale_effect, outside any list iteration.
aeluran_rs_temple_marker_check_effect = {
	clear_saved_scope = rs_check_sister
	var:aeluran_rs_temple_sister ?= { save_scope_as = rs_check_sister }
	if = {
		limit = {
			NOT = {
				AND = {
					exists = scope:rs_check_sister
					scope:rs_check_sister = {
						is_alive = yes
						has_council_position = councillor_aeluran_advisor
					}
					exists = situation_center_province
					situation_center_province.barony = {
						aeluran_temple_kept_by_sister_trigger = { SISTER = scope:rs_check_sister }
					}
				}
			}
		}
		add_to_temporary_list = rs_stale_markers
	}
}

aeluran_rs_temple_markers_close_stale_effect = {
	every_in_list = {
		list = rs_stale_markers
		situation_center_province ?= { remove_variable = aeluran_rs_temple_marker }
		remove_list_global_variable = { name = aeluran_rs_temple_markers target = this }
		end_situation = yes
	}
}

# Root = a marker situation (its on_monthly)
aeluran_rs_temple_marker_self_check_effect = {
	aeluran_rs_temple_marker_check_effect = yes
	aeluran_rs_temple_markers_close_stale_effect = yes
}

# Scope = a temple province; scope:rs_sister = the Red Sister who keeps it
aeluran_rs_temple_marker_start_effect = {
	save_scope_as = rs_temple
	start_situation = {
		type = aeluran_rs_temple_marker
		save_scope_as = rs_marker_new
	}
	scope:rs_marker_new ?= {
		set_situation_center_province = scope:rs_temple
		# The engine draws no pin for a situation with no land: give it the temple's county
		situation_sub_region:aeluran_rs_marker_area = { add_dejure_title_to_sub_region = scope:rs_temple.county }
		set_variable = { name = aeluran_rs_temple_sister value = scope:rs_sister }
		add_to_global_variable_list = { name = aeluran_rs_temple_markers target = this }
		scope:rs_temple = { set_variable = { name = aeluran_rs_temple_marker value = scope:rs_marker_new } }
	}
	if = {
		limit = { NOT = { exists = scope:rs_marker_new } }
		debug_log = "RS MARKER: start_situation made no situation"
		debug_log_scopes = yes
	}
}

# Game start: up to two new temples in each Matronate, on empty non-capital baronies, one per county, in counties an
# Aeluran ruler holds herself, skipping counties that already have one. Their Red Sisters take them through the lease.
# Any scope.
aeluran_matronate_seed_temples_effect = {
	faith:aeluran_weavers = {
		every_clerical_region_title = {
			limit = { aeluran_is_matronate_title_trigger = yes }
			save_temporary_scope_as = seed_matronate
			while = {
				count = 2
				random_county_in_clerical_region = {
					limit = { aeluran_matronate_seed_temple_county_trigger = yes }
					random_county_province = {
						limit = { aeluran_matronate_seed_temple_province_trigger = yes }
						set_holding_type = church_holding
					}
				}
			}
		}
	}
}
