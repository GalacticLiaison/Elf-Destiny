set_aeluran_clothing_colors = {
	if = {
		limit = {
			NOT = {
				has_character_flag = ignore_aeluran_clothes
			}
		}
		manual_set_clothes_colors = {
			channel = 1
			RED = 0.5
			GREEN = 0
			BLUE = 0
		}
		manual_set_clothes_colors = {
			channel = 2
			RED = 0.5
			GREEN = 0
			BLUE = 0
		}
	}
}

set_merchants_clothes = {
	manual_set_clothes_colors = {
		channel = 1
		RED = 0.7
		GREEN = 0.7
		BLUE = 0.7
	}
	manual_set_clothes_colors = {
		channel = 3
		RED = 0
		GREEN = 0.05
		BLUE = 0
	}
	manual_set_clothes_colors = {
		channel = 4
		RED = 0
		GREEN = 0.05
		BLUE = 0
	}
	manual_set_clothes_colors = {
		channel = 5
		RED = 1
		GREEN = 1
		BLUE = 1
	}
}

manual_set_clothes_colors = {
    set_variable = {
		name = POD_portrait_clothing_$channel$_r
		value = {
			value = $RED$
			# divide = 255
			min = 0.004 # minimum value (not zero so that the decal doesn't disappear)
			max = 1.0   # maximum value that shaders can read
		}
	}
	set_variable = {
		name = POD_portrait_clothing_$channel$_g
		value = {
			value = $GREEN$
			# divide = 255
			min = 0.004 # minimum value (not zero so that the decal doesn't disappear)
			max = 1.0   # maximum value that shaders can read
		}
	}
	set_variable = {
		name = POD_portrait_clothing_$channel$_b
		value = {
			value = $BLUE$
			# divide = 255
			min = 0.004 # minimum value (not zero so that the decal doesn't disappear)
			max = 1.0   # maximum value that shaders can read
		}
	}
}

clear_aeluran_clothing_colors = {
	remove_variable = POD_portrait_clothing_1_r
    remove_variable = POD_portrait_clothing_1_g
    remove_variable = POD_portrait_clothing_1_b

    remove_variable = POD_portrait_clothing_2_r
    remove_variable = POD_portrait_clothing_2_g
    remove_variable = POD_portrait_clothing_2_b
}
# Robes by chapter: black for a Sister who follows the Dark Sisters, red otherwise (PM 2026-10-05: the tenet decides,
# race plays no part). Use this wherever a Sister is created, raised or seated, so a promotion never repaints her.
aeluran_set_sister_clothing_effect = {
	if = {
		limit = { NOT = { has_character_flag = ignore_aeluran_clothes } }
		if = {
			limit = { aeluran_follows_dark_sisters_trigger = yes }
			debug_log = "ROBES black (Dark Sisters)"
			set_black_aeluran_clothing_colors = yes
		}
		else = {
			debug_log = "ROBES red (Aeluran Sisters)"
			manual_set_clothes_colors = { channel = 1 RED = 0.5 GREEN = 0 BLUE = 0 }
			manual_set_clothes_colors = { channel = 2 RED = 0.5 GREEN = 0 BLUE = 0 }
		}
		debug_log_scopes = yes
	}
}
