# The champion's house: light houses for light elf rulers, the four dark houses for dark elf rulers (PM 2026-10-09).
# Sets scope:bloodline_strain, scope:champion_bloodline (read by the bloodline_champion template) and scope:champion_culture.
determine_bloodline_champion_bloodline_effect = {
	if = {
		limit = { is_dark_elf_type = yes }
		save_scope_value_as = { name = bloodline_strain value = flag:dark }
	}
	else = {
		save_scope_value_as = { name = bloodline_strain value = flag:light }
	}
	random_list = {
		10 = {
			trigger = { scope:bloodline_strain = flag:light }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:daelurin
			}
		}
		10 = {
			trigger = { scope:bloodline_strain = flag:light }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:thundarael
			}
		}
		10 = {
			trigger = { scope:bloodline_strain = flag:light }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:gwynthorn
			}
		}
		10 = {
			trigger = { scope:bloodline_strain = flag:light }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:serelion
			}
		}
		10 = {
			trigger = { scope:bloodline_strain = flag:light }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:valerith
			}
		}
		10 = {
			trigger = { scope:bloodline_strain = flag:light }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:lormelis
			}
		}
		10 = {
			trigger = { scope:bloodline_strain = flag:dark }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:drenmyre
			}
		}
		10 = {
			trigger = { scope:bloodline_strain = flag:dark }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:rynvela
			}
		}
		10 = {
			trigger = { scope:bloodline_strain = flag:dark }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:syldranis
			}
		}
		10 = {
			trigger = { scope:bloodline_strain = flag:dark }
			save_scope_value_as = {
				name = champion_bloodline
				value = flag:myrvessa
			}
		}
	}
	if = {
		limit = {
			OR = {
				scope:champion_bloodline = flag:drenmyre
				scope:champion_bloodline = flag:rynvela
			}
		}
		culture:elf_culture_dokkalfar = { save_scope_as = champion_culture }
	}
	else_if = {
		limit = {
			OR = {
				scope:champion_bloodline = flag:syldranis
				scope:champion_bloodline = flag:myrvessa
			}
		}
		culture:elf_culture_drow = { save_scope_as = champion_culture }
	}
	else = {
		culture:culture_elf = { save_scope_as = champion_culture }
	}
}

		location = root.location

		culture = scope:champion_culture
		save_scope_as = champion
	}
}
