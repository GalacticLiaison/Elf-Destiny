# Why Override this?
#  Divine Marriage makes Incestuous a virtue, and a religion cannot switch off a virtue granted by one of its tenets (it can only turn it into a sin).
#  Vanilla checks this tenet by key in ~60 places (marriage acceptance, schemes, factions), so a renamed copy would lose all of that.
#  The only change from vanilla is the removed "incestuous" line in traits. Re-diff against vanilla after every game patch.
tenet_divine_marriage = {
	icon = tenet_divine_marriage

	name = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_latin_as_liturgical_language_trigger = yes
				}
				desc = tenet_divine_marriage_latin_name
			}
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_greek_as_liturgical_language_trigger = yes
				}
				desc = tenet_divine_marriage_greek_name
			}
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_greek_as_liturgical_language_trigger = no
					pam_latin_as_liturgical_language_trigger = no
				}
				desc = tenet_divine_marriage_default_name
			}
			triggered_desc = {
				trigger = {
					rite_religion_tag = zoroastrianism_religion
				}
				desc = tenet_divine_marriage_xwedodah 
			}
			desc = tenet_divine_marriage_name
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_latin_as_liturgical_language_trigger = yes
				}
				desc = tenet_divine_marriage_latin_desc
			}
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_greek_as_liturgical_language_trigger = yes
				}
				desc = tenet_divine_marriage_greek_desc
			}
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_greek_as_liturgical_language_trigger = no
					pam_latin_as_liturgical_language_trigger = no
				}
				desc = tenet_divine_marriage_default_desc
			}
			desc = tenet_divine_marriage_desc
		}
	}

	piety_cost = {
		if = {
			limit = {
				OR = {
					rite_religion_tag = islam_religion
					rite_religion_tag = dualism_religion
					rite_religion_tag = buddhism_religion
					rite_religion_tag = jainism_religion
				}
			}
			value = faith_tenet_cost_high
		}
		else = {
			value = faith_tenet_cost_mid
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_divine_marriage }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	can_pick = {
		custom_description = {
			text = doctrine_unrestricted_consanguinity_trigger
			doctrine:doctrine_consanguinity_unrestricted = { is_in_list = selected_doctrines }
		}
	}

	parameters = {
		divine_marriage_piety_gain_active
		tenet_divine_marriage_incestuous_liege_faction_reluctance
	}

	personal_tenet_parameters = {
		tenet_divine_marriage_spouse_contribution
	}

	special_parameters = {
		divine_marriage_opinion = 10
	}

	character_modifier = {
		spouse_opinion = 10
		positive_random_genetic_chance = 0.15
		negative_random_genetic_chance = 0.15
		positive_inactive_inheritance_chance = 0.15
		negative_inactive_inheritance_chance = 0.15
		genetic_trait_strengthen_chance = 0.1
	}

	traits = {
		virtues = {
			pure_blooded = { weight = 30 scale = 1 }
			loyal = { weight = 15 scale = 1 }
		}
	}
}
