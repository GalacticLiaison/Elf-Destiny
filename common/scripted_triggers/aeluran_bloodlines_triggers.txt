# The Bloodlines board (Order window): the unwed noble elves in a shepherd's care, and the matches she may propose.
# Effects in common/scripted_effects/aeluran_bloodlines_effects.txt.

# Scope: a character; SHEPHERD = the shepherd matching them. Could the Order wed this elf: Aeluran faith, room for another
# spouse (the unwed, and rulers whose faith allows more than one), not betrothed? Noble or not.
aeluran_bloodlines_marriable_trigger = {
	is_alive = yes
	is_any_elf_type = yes
	faith = faith:aeluran_weavers
	age >= 12
	# Room for another spouse: the unwed, and rulers whose faith allows more than one
	allowed_more_spouses = yes
	is_betrothed = no
	is_imprisoned = no
	can_marry_common_trigger = yes
	NOT = { this = $SHEPHERD$ }
	NOR = {
		has_trait = aeluran_matron
		has_trait = aeluran_high_matriarch
	}
	NOT = { has_variable = aeluran_bloodlines_refused }
}

# Scope: a character; SHEPHERD = the shepherd. A marriable noble: who may be matched anywhere in the Order (matches are
# made across all Matronates, PM 2026-10-02).
aeluran_bloodlines_eligible_trigger = {
	aeluran_bloodlines_marriable_trigger = { SHEPHERD = $SHEPHERD$ }
	exists = dynasty
}

# Scope: a character; SHEPHERD = a Matron holding a Matronate, or the High Matriarch. An eligible noble (above) within
# her care: rulers count by their capital, everyone else by where they live. The High Matriarch's care is everywhere
# outside the Matronates. These are the nobles she seeks spouses for.
aeluran_bloodlines_candidate_trigger = {
	aeluran_bloodlines_eligible_trigger = { SHEPHERD = $SHEPHERD$ }
	aeluran_bloodlines_in_care_trigger = { SHEPHERD = $SHEPHERD$ }
}

# Scope: a character; SHEPHERD = a Matron holding a Matronate, or the High Matriarch. Within her care: rulers by their
# capital, everyone else by where they live; the High Matriarch's care is everywhere outside the Matronates.
aeluran_bloodlines_in_care_trigger = {
	trigger_if = {
		limit = { is_landed = yes }
		trigger_if = {
			limit = { $SHEPHERD$ = { has_title = title:d_high_matriarch } }
			NOT = { capital_county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
		}
		trigger_else = {
			capital_county.clerical_region_title.holder ?= $SHEPHERD$
		}
	}
	trigger_else = {
		trigger_if = {
			limit = { $SHEPHERD$ = { has_title = title:d_high_matriarch } }
			NOT = { location.county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
		}
		trigger_else = {
			location.county.clerical_region_title.holder ?= $SHEPHERD$
		}
	}
}

# Scope: a character; SHEPHERD = the shepherd. A possible match: an eligible noble from anywhere in the Order, or a Sister
# of the shepherd's own care, noble or not (PM 2026-10-02).
aeluran_bloodlines_match_trigger = {
	OR = {
		aeluran_bloodlines_eligible_trigger = { SHEPHERD = $SHEPHERD$ }
		AND = {
			has_trait = aeluran_sister
			aeluran_bloodlines_marriable_trigger = { SHEPHERD = $SHEPHERD$ }
			aeluran_bloodlines_in_care_trigger = { SHEPHERD = $SHEPHERD$ }
		}
	}
}

# Scope: a character; OTHER = the proposed partner. Could the two be wed (sex, kinship law, the Order's reach)?
aeluran_bloodlines_pair_trigger = {
	NOT = { this = $OTHER$ }
	could_marry_character_trigger = { CHARACTER = $OTHER$ }
}
