# Dark Elf Beta OVERRIDE of Elf Destiny's 0lib_urf_race_triggers.txt (full copy).
# Change: has_any_elf_flag_trigger also recognizes the race_dark_* input flags.

###########################
### URF Racial Triggers ###
###########################
urf_race_initialized = {
    has_variable = urf_phenotype
}

urf_for_all_phenotypes_trigger = {
    OR = {
        #Elf Destiny
        $APPLY$ = { RACE = elf }
        #End Elf Destiny

        $APPLY$ = { RACE = human }
    }
}
urf_for_all_phenotypes_comparator_trigger = {
    OR = {
        #Elf Destiny
        urf_elf_threshold = { RACE = elf }
        #End Elf Destiny

        $APPLY$ = { RACE = human COMPARATOR = $COMPARATOR$ }
    }
}

#Elf Destiny

urf_race_elf_trigger = {
    has_any_elf_flag_trigger = yes
}

urf_elf_phenotype_trigger = {
    var:urf_phenotype = flag:elf
}

has_any_elf_flag_trigger = {
    OR = {
        has_character_flag = race_elf_blood
        has_character_flag = race_elf
        has_character_flag = race_high_elf
        has_character_flag = race_true_elf
        has_character_flag = race_fae
        has_character_flag = race_fae_radiant
        has_character_flag = race_celestial
        has_character_flag = race_seraphic_celestial
        has_character_flag = race_eldar
        has_character_flag = race_maiar
        has_character_flag = race_valar
        has_character_flag = race_aratar
        # Dark Elf Beta
        has_character_flag = race_dark_elf_blood
        has_character_flag = race_dark_elf
        has_character_flag = race_dark_high_elf
        has_character_flag = race_dark_true_elf
        has_character_flag = race_dark_fae
        has_character_flag = race_dark_fae_radiant
        has_character_flag = race_dark_celestial
        has_character_flag = race_dark_seraphic_celestial
        has_character_flag = race_dark_eldar
        has_character_flag = race_dark_maiar
        has_character_flag = race_dark_valar
        has_character_flag = race_dark_aratar
        # End Dark Elf Beta
    }
}

urf_race_unknown_trigger = {
    has_character_flag = race_unknown
}

urf_unknown_phenotype_trigger = {
    var:urf_phenotype = flag:unknown
}

#End Elf Destiny

urf_race_human_trigger = {
    is_human_culture = yes
    has_any_elf_flag_trigger = no
}

urf_human_phenotype_trigger = {
    var:urf_phenotype = flag:human
}

urf_has_a_race_modifier = {
    OR = {
        has_character_modifier = urf_phenotype_$RACE$_modifier
        # Elf Destiny
        is_any_elf_type = yes
        # End Elf Destiny
    }
}

is_human_culture = {
    always = yes
}
