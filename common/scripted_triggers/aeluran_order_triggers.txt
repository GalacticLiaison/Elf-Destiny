is_aeluran_sister_or_higher = {
	OR = {
		has_trait = aeluran_sister
		has_trait = aeluran_matron
		has_trait = aeluran_high_matriarch
	}
}
# Follows the Dark Sisters: the rite, or failing that the faith, carries the tenet. Works on characters and on county
# titles (both have faith and rite). This, not race, decides the colour of a Sister's robes (PM 2026-10-05).
aeluran_follows_dark_sisters_trigger = {
	OR = {
		rite ?= { rite_has_tenet = tenet_dark_sisters }
		faith ?= { has_tenet = tenet_dark_sisters }
	}
}

# Follows the Order, of either Sisterhood: either tenet, on the rite or on the faith (PM 2026-10-08: every "has the
# Aeluran Sisters tenet" gate in the mod uses this). has_tenet reads the faith's mainline core tenets, so a Dark Sisters
# branch rite of the Aeluran faith passes through the faith; a standalone faith built on the Dark Sisters tenet passes
# through the tenet itself. Works on characters and on county titles.
aeluran_order_faith_trigger = {
	OR = {
		faith ?= { has_tenet = tenet_aeluran_sisters }
		faith ?= { has_tenet = tenet_dark_sisters }
		rite ?= { rite_has_tenet = tenet_aeluran_sisters }
		rite ?= { rite_has_tenet = tenet_dark_sisters }
	}
}
