is_aeluran_sister_or_higher = {
	OR = {
		has_trait = aeluran_sister
		has_trait = aeluran_matron
		has_trait = aeluran_high_matriarch
	}
}
# Follows the Dark Sisters: the rite, or failing that the faith, carries the tenet. Works on characters and on county
# titles (both have faith and rite). This, not race, decides the colour of a Sister's robes (PM 2026-10-05).
aeluran_follows_dark_sisters_trigger = {
	OR = {
		rite ?= { rite_has_tenet = tenet_dark_sisters }
		faith ?= { has_tenet = tenet_dark_sisters }
	}
}
