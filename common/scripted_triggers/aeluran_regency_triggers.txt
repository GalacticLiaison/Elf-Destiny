aeluran_domination_level_full_deference = {
	diarchy_swing <= 10
}

aeluran_domination_level_high_deference = {
	diarchy_swing > 10
	diarchy_swing <= 25
}

aeluran_domination_level_low_deference = {
	diarchy_swing > 25
	diarchy_swing <= 40
}

aeluran_domination_level_neutral = {
	diarchy_swing > 40
	diarchy_swing < 60
}

aeluran_domination_level_low_dominated = {
	diarchy_swing >= 60
	diarchy_swing < 75
}

aeluran_domination_level_highly_dominated= {
	diarchy_swing >= 75
	diarchy_swing < 90
}

aeluran_domination_level_fully_dominated = {
	diarchy_swing >= 90
}

# The ruler still holds the county and the holding the Order paid for has been built (PROVINCE = its province).
# Without the second check an unbuilt holding's barony was granted, leaving the Red Sister holding no land.
new_temple_available_for_diarch = {
	$COUNTY$.holder = root
	$PROVINCE$ = { has_holding = yes }
}
# Scope = a province, root = the ruler. A Matter of Land may found a temple here: the ruler holds the county, the barony
# is empty and nothing is being built on it. has_holding stays "no" while a temple holding is still being founded, so
# the construction flag is checked as well. Used by the survey AND by the event options: an undeveloped_holding target
# saved years earlier survives clear_saved_scope when it was inherited from the triggering event, and the Order then
# tried to found a temple on a barony the ruler had lost that already had one (2026-10-11).
aeluran_undeveloped_holding_trigger = {
	has_holding = no
	has_ongoing_construction = no
	NOT = { has_construction_with_flag = temple }
	barony_cannot_construct_holding = no
	county.holder = root
}
# Scope = a barony title. A temple that $SISTER$ (a Red Sister) keeps: leased to her by her ruler's lease, or, in saves
# from before Beta_v2.0.4, held by her outright. Used by A Matter of Land and the temple map markers.
aeluran_temple_kept_by_sister_trigger = {
	title_province = { has_holding_type = church_holding }
	OR = {
		lessee ?= $SISTER$
		holder = $SISTER$
	}
}
