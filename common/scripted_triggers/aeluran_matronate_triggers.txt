# Scope: a character, scope:matronate = a Matronate. Does she live in its region? A ruler of land by her capital,
# anyone else by where she is (her court).
aeluran_in_matronate_region_trigger = {
	trigger_if = {
		limit = { any_held_title = { is_landless_type_title = no } }
		capital_county.clerical_region_title ?= scope:matronate
	}
	trigger_else = {
		location.county.clerical_region_title ?= scope:matronate
	}
}

# Scope: a character, scope:matronate = the Matronate to fill, scope:outgoing_matron (optional) = the Matron leaving it.
# Can she be raised to Matron to take the Matronate?
aeluran_matronate_successor_candidate_trigger = {
	has_trait = aeluran_sister
	is_female = yes
	is_adult = yes
	faith = faith:aeluran_weavers
	is_imprisoned = no
	trigger_if = {
		limit = { exists = scope:outgoing_matron }
		NOT = { this = scope:outgoing_matron }
	}
	# One clerical region per ruler
	NOT = { any_held_title = { has_clerical_region = yes } }
	# A Sister who rules land must rule it inside the Matronate: a Matronate far from her own realm is not hers to keep
	trigger_if = {
		limit = { any_held_title = { is_landless_type_title = no } }
		aeluran_in_matronate_region_trigger = yes
	}
}

# Scope: a character. Is she of the kind the High Matriarch may give a Matronate to (a Sister, or a Matron without one)?
aeluran_may_receive_matronate_trigger = {
	OR = {
		has_trait = aeluran_sister
		has_trait = aeluran_matron
	}
	faith = faith:aeluran_weavers
	NOT = { has_title = title:d_high_matriarch }
	# One clerical region per ruler
	NOT = { any_held_title = { has_clerical_region = yes } }
}

# Scope: a character. She rules land under a duke or a count: a Matronate (duchy rank, like her Matron's Seat) would
# raise her above her liege and pull her land out of the realm.
aeluran_matronate_would_break_realm_trigger = {
	any_held_title = { is_landless_type_title = no }
	is_independent_ruler = no
	liege = { highest_held_title_tier <= tier_duchy }
}

# Scope: a county. Can a new Matronate be founded with this county as its capital?
aeluran_matronate_foundable_county_trigger = {
	# A region takes its faith from its capital
	faith = faith:aeluran_weavers
	NOT = { exists = clerical_region_title }
	# Spirits' Landing is the High Matriarch's own see
	NOT = { this = title:d_high_matriarch.title_capital_county }
}

# Scope: a character. Is the county she calls home (her capital, or her court's county if she rules no land) one a
# Matronate can be founded around?
aeluran_matronate_home_county_foundable_trigger = {
	trigger_if = {
		limit = { any_held_title = { is_landless_type_title = no } }
		capital_county ?= { aeluran_matronate_foundable_county_trigger = yes }
	}
	trigger_else = {
		location.county ?= { aeluran_matronate_foundable_county_trigger = yes }
	}
}

# Scope: a character, scope:target = a Matronate. Does she live in it (by capital if she rules land, else by court)?
aeluran_lives_in_target_matronate_trigger = {
	trigger_if = {
		limit = { any_held_title = { is_landless_type_title = no } }
		capital_county.clerical_region_title ?= scope:target
	}
	trigger_else = {
		location.county.clerical_region_title ?= scope:target
	}
}

# Scope: a character. Is she or he one of the Order's flock: an Aeluran ruler of count rank or above?
# Human (non-Aeluran) rulers are not; by default the Order wants them gone (PM 2026-09-30).
aeluran_flock_member_trigger = {
	is_ruler = yes
	# Landed rulers only: adventurer bands and mercenary companies have no realm for the Order to tend
	is_landed = yes
	NOT = { primary_title ?= { is_mercenary_company = yes } }
	faith = faith:aeluran_weavers
	highest_held_title_tier >= tier_county
}

# Scope: a landed title. Is it an Aeluran Matronate?
# The bookmark Matronates are named explicitly so this holds before game start tags them with aeluran_matronate.
aeluran_is_matronate_title_trigger = {
	OR = {
		has_variable = aeluran_matronate
		this = title:d_matronate_feyswood
		this = title:d_matronate_crystal_glades
		this = title:d_matronate_high_hills
		this = title:d_matronate_valusia
		this = title:d_matronate_menzoberranzan
		this = title:d_matronate_middle_sea
		this = title:d_matronate_western_shores
		this = title:d_matronate_svartalfaheim
	}
}

# Scope: a county. An Aeluran county no Matronate (or other clerical region) covers yet.
aeluran_matronate_uncovered_county_trigger = {
	faith = faith:aeluran_weavers
	NOT = { exists = clerical_region_title }
	# Spirits' Landing is the High Matriarch's own see
	NOT = { this = title:d_high_matriarch.title_capital_county }
}

# Scope: a ruler. Does her or his realm hold enough uncovered Aeluran counties to warrant a Matronate of its own?
aeluran_realm_wants_matronate_trigger = {
	any_sub_realm_county = {
		count >= 3 # aeluran_matronate_petition_min_counties (count takes a number)
		aeluran_matronate_uncovered_county_trigger = yes
	}
}

# Scope: a ruler. May she or he petition the High Matriarch for a Matronate around the realm capital?
aeluran_can_petition_matronate_trigger = {
	is_ruler = yes
	faith = faith:aeluran_weavers
	highest_held_title_tier >= tier_county
	NOT = { has_title = title:d_high_matriarch }
	exists = title:d_high_matriarch.holder
}

# Scope: a county, scope:matronate = a Matronate. Should the Matronate take this county in?
# Uncovered, Aeluran, bordering the Matronate, and ruled from a realm whose capital lies in it.
aeluran_matronate_can_absorb_county_trigger = {
	aeluran_matronate_uncovered_county_trigger = yes
	holder.top_liege.capital_county.clerical_region_title ?= scope:matronate
	scope:matronate = { is_adjacent_to_clerical_region = prev }
}

# On a clerical region faith mismatch (root = region holder, scope:title, scope:county = its capital): a foreign
# region (an archdiocese) whose seat just turned Aeluran. Vanilla's handlers leave it alone (events/dlc/pam fork) and
# the Order claims it once aeluran_see_claimable_trigger holds: seat held by an elf and half the counties Aeluran
# (PM 2026-10-04). A see whose seat turns early stands with an Aeluran seat until then.
aeluran_see_seat_turned_aeluran_trigger = {
	scope:reason = flag:county_converted
	NOT = { scope:title = { aeluran_is_matronate_title_trigger = yes } }
	scope:county = { faith = faith:aeluran_weavers }
}

# Scope: a clerical region title. A foreign see the Order may claim as a Matronate: its seat is Aeluran under an
# elven ruler, and at least half of its counties are Aeluran (PM 2026-10-04).
aeluran_see_claimable_trigger = {
	has_clerical_region = yes
	NOT = { aeluran_is_matronate_title_trigger = yes }
	title_capital_county ?= {
		faith = faith:aeluran_weavers
		holder ?= { is_any_elf_type = yes }
	}
	any_county_in_clerical_region = {
		percent >= 0.5
		faith = faith:aeluran_weavers
	}
}

# Scope: a character. Does the realm she calls home (by her capital, or her court's county) want a Matronate?
aeluran_matronate_home_realm_wants_trigger = {
	trigger_if = {
		limit = { any_held_title = { is_landless_type_title = no } }
		capital_county.holder.top_liege ?= { aeluran_realm_wants_matronate_trigger = yes }
	}
	trigger_else = {
		location.county.holder.top_liege ?= { aeluran_realm_wants_matronate_trigger = yes }
	}
}

# Scope: a character, scope:flock_shepherd = a Matron or the High Matriarch, scope:matronate (optional) = her Matronate.
# Is she in the shepherd's care: her religious superior is the shepherd (an Advisor holding the shepherd's leases), or
# she lives in the shepherd's Matronate (for a High Matriarch without one: outside every Matronate)?
aeluran_order_sister_in_care_trigger = {
	faith = faith:aeluran_weavers
	NOT = { this = scope:flock_shepherd }
	OR = {
		superior ?= scope:flock_shepherd
		AND = {
			exists = scope:matronate
			aeluran_in_matronate_region_trigger = yes
		}
		AND = {
			NOT = { exists = scope:matronate }
			scope:flock_shepherd = { has_title = title:d_high_matriarch }
			trigger_if = {
				limit = { any_held_title = { is_landless_type_title = no } }
				NOT = { capital_county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
			}
			trigger_else = {
				NOT = { location.county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
			}
		}
	}
}
