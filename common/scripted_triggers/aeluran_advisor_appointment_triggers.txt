# Placing Sisters (see common/scripted_effects/aeluran_advisor_appointment_effects.txt)

# Scope: a character. POST = the ruler whose Advisor is being chosen, APPOINTER = who chooses.
# Can she be appointed? A free Sister within the appointer's reach: her own Matronate for a Matron; for the High
# Matriarch, the ruler's own realm or anywhere outside the Matronates.
aeluran_advisor_candidate_trigger = {
	has_trait = aeluran_sister
	faith = faith:aeluran_weavers
	is_capable_adult = yes
	is_imprisoned = no
	NOT = { this = $POST$ }
	NOT = { has_council_position = councillor_aeluran_advisor }
	NOT = { var:appointed_to_advise ?= { is_alive = yes } }
	# A council seat needs a courtier or a vassal of the ruler
	trigger_if = {
		limit = { is_ruler = yes }
		is_vassal_of = $POST$
	}
	# Her reach
	trigger_if = {
		limit = {
			$POST$.capital_county.clerical_region_title ?= {
				aeluran_is_matronate_title_trigger = yes
				holder ?= $APPOINTER$
			}
		}
		$POST$.capital_county.clerical_region_title = { save_temporary_scope_as = matronate }
		aeluran_in_matronate_region_trigger = yes
	}
	trigger_else = {
		OR = {
			is_courtier_of = $POST$
			is_vassal_of = $POST$
			location.county.holder.top_liege ?= $POST$.top_liege
			NOT = { location.county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
		}
	}
}

# Scope: a character (a Matron or the High Matriarch); scope:recipient = an Aeluran ruler.
# May she recall the ruler's Advisor? The High Matriarch may recall anyone's; a Matron those of the counts and dukes of
# her Matronate.
aeluran_has_advisor_authority_over_trigger = {
	OR = {
		has_title = title:d_high_matriarch
		AND = {
			has_trait = aeluran_matron
			scope:recipient.highest_held_title_tier <= tier_duchy
			scope:recipient.capital_county.clerical_region_title ?= {
				aeluran_is_matronate_title_trigger = yes
				holder ?= root
			}
		}
	}
}

# Scope: a character, SHEPHERD = a Matron or the High Matriarch. Is she a Sister of the shepherd's care who holds no
# land and serves nowhere (no council seat, no Advisor post given or on the way), and not already at the shepherd's court?
aeluran_summonable_sister_trigger = {
	has_trait = aeluran_sister
	faith = faith:aeluran_weavers
	is_ruler = no
	is_councillor = no
	NOT = { has_variable = appointed_to_advise }
	NOT = { has_character_flag = aeluran_advisor_is_on_the_way }
	NOT = { employer ?= $SHEPHERD$ }
	OR = {
		superior ?= $SHEPHERD$
		# Lives in the shepherd's Matronate
		location.county.clerical_region_title.holder ?= $SHEPHERD$
		# The High Matriarch's: outside every Matronate
		AND = {
			$SHEPHERD$ = { has_title = title:d_high_matriarch }
			NOT = { location.county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
		}
	}
}

# Scope: a character, SHEPHERD = a Matron or the High Matriarch. Is she a Red Sister (Aeluran Advisor) on the
# council of a ruler in the shepherd's care, someone other than the shepherd herself?
aeluran_red_sister_in_care_trigger = {
	has_council_position = councillor_aeluran_advisor
	exists = liege
	NOT = { liege = $SHEPHERD$ }
	OR = {
		superior ?= $SHEPHERD$
		liege.capital_county.clerical_region_title.holder ?= $SHEPHERD$
		AND = {
			$SHEPHERD$ = { has_title = title:d_high_matriarch }
			NOT = { liege.capital_county.clerical_region_title ?= { aeluran_is_matronate_title_trigger = yes } }
		}
	}
}

# Scope: a Sister, SHEPHERD = the shepherd asking. A Sister bound to her (friend, lover, or under her hook) always
# agrees, and searches harder.
aeluran_whispers_loyal_sister_trigger = {
	OR = {
		has_relation_friend = $SHEPHERD$
		has_relation_best_friend = $SHEPHERD$
		has_relation_lover = $SHEPHERD$
		has_relation_soulmate = $SHEPHERD$
		$SHEPHERD$ = { has_hook = prev }
	}
}

# Scope: a Sister who refused, SHEPHERD = the shepherd who asked. Does she go to her ruler with it?
# Her rival, a Sister who thinks badly of her, or one who has come to love her ruler more than the Order.
aeluran_whispers_betrayal_trigger = {
	OR = {
		has_relation_rival = $SHEPHERD$
		has_relation_nemesis = $SHEPHERD$
		opinion = {
			target = $SHEPHERD$
			value <= -25
		}
		AND = {
			opinion = {
				target = liege
				value >= 40
			}
			opinion = {
				target = $SHEPHERD$
				value <= 0
			}
		}
	}
}
