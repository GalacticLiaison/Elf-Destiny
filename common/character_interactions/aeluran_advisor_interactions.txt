# Placing Sisters: a Matron (or the High Matriarch) appoints and recalls the Advisors of her flock.
# Effects in common/scripted_effects/aeluran_advisor_appointment_effects.txt. Used on the ruler who needs the Advisor.
# The AI uses the effects directly: an AI appointer fills a post at once.

# Appoint a Sister of the Matronate to an open Advisor post
aeluran_appoint_advisor_interaction = {
	category = interaction_category_religion
	icon = icon_aeluran_sister
	interface_priority = 70
	common_interaction = yes
	use_diplomatic_range = no
	desc = aeluran_appoint_advisor_interaction_desc

	# The best candidate is offered first
	redirect = {
		scope:recipient = { save_scope_as = advisor_post }
		scope:actor = {
			ordered_living_character = {
				limit = {
					has_trait = aeluran_sister
					aeluran_advisor_candidate_trigger = { POST = scope:recipient APPOINTER = scope:actor }
				}
				order_by = aeluran_advisor_candidate_score
				save_scope_as = secondary_recipient
			}
		}
	}

	populate_recipient_list = {
		scope:recipient = { save_scope_as = advisor_post }
		scope:actor = {
			every_living_character = {
				limit = {
					has_trait = aeluran_sister
					aeluran_advisor_candidate_trigger = { POST = scope:recipient APPOINTER = scope:actor }
				}
				add_to_list = characters
			}
		}
	}

	is_shown = {
		scope:recipient = {
			NOT = { this = scope:actor }
			var:aeluran_advisor_vacancy ?= scope:actor
			NOT = { exists = cp:councillor_aeluran_advisor }
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = aeluran_appoint_advisor_no_candidate_tt
			exists = scope:secondary_recipient
		}
		scope:secondary_recipient ?= {
			aeluran_advisor_candidate_trigger = { POST = scope:recipient APPOINTER = scope:actor }
		}
	}

	on_accept = {
		# A Sister who dislikes the posting refuses it: let it stand (the post stays open) or compel her
		if = {
			limit = {
				scope:secondary_recipient = { aeluran_refuses_posting_trigger = yes }
			}
			scope:secondary_recipient = {
				custom_tooltip = aeluran_appoint_advisor_refuses_tt
			}
			hidden_effect = {
				aeluran_compel_begin_effect = { SISTER = scope:secondary_recipient TASK = advisor }
			}
		}
		else = {
			scope:secondary_recipient = {
				custom_tooltip = aeluran_appoint_advisor_tt
			}
			hidden_effect = {
				scope:actor = { save_scope_as = advisor_appointer }
				scope:secondary_recipient = { save_scope_as = aeluran_sister_advisor }
				scope:recipient = { aeluran_appoint_chosen_advisor_effect = yes }
			}
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

# Recall a ruler's Advisor; the post opens again
aeluran_recall_advisor_interaction = {
	category = interaction_category_religion
	icon = icon_aeluran_sister
	interface_priority = 40
	common_interaction = no
	use_diplomatic_range = no
	desc = aeluran_recall_advisor_interaction_desc

	is_shown = {
		scope:recipient = {
			NOT = { this = scope:actor }
			exists = cp:councillor_aeluran_advisor
		}
		scope:actor = { aeluran_has_advisor_authority_over_trigger = yes }
	}

	cooldown_against_recipient = { years = 5 }

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_incapable = no
		}
	}

	cost = {
		piety = 100
	}

	on_accept = {
		scope:recipient.cp:councillor_aeluran_advisor = { save_scope_as = recalled_advisor }
		scope:recalled_advisor = {
			custom_tooltip = aeluran_recall_advisor_tt
			add_opinion = {
				target = scope:actor
				modifier = aeluran_recalled_advisor_opinion
			}
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = aeluran_recalled_advisor_opinion
			}
			custom_tooltip = aeluran_recall_advisor_post_opens_tt
		}
		hidden_effect = {
			scope:recipient = {
				fire_councillor = scope:recalled_advisor
			}
			scope:recalled_advisor = {
				if = {
					limit = { has_variable = appointed_to_advise }
					remove_variable = appointed_to_advise
				}
			}
			scope:recipient = { aeluran_open_advisor_post_effect = yes }
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

# Summon a landless Sister of the shepherd's Matronate who serves nowhere to the shepherd's own court
aeluran_summon_sister_interaction = {
	category = interaction_category_religion
	icon = icon_aeluran_sister
	interface_priority = 60
	common_interaction = yes
	use_diplomatic_range = no
	desc = aeluran_summon_sister_interaction_desc

	is_shown = {
		scope:actor = {
			OR = {
				has_title = title:d_high_matriarch
				any_held_title = { aeluran_is_matronate_title_trigger = yes }
			}
		}
		scope:recipient = {
			NOT = { this = scope:actor }
			has_trait = aeluran_sister
			aeluran_summonable_sister_trigger = { SHEPHERD = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_incapable = no
		}
		scope:recipient = {
			is_imprisoned = no
			is_hostage = no
		}
	}

	ai_instant_response = yes

	on_accept = {
		scope:recipient = {
			custom_tooltip = aeluran_summon_sister_tt
		}
		hidden_effect = {
			scope:actor = { add_courtier = scope:recipient }
		}
	}

	# She defies you: let it stand or compel her (aeluran_compel.001)
	on_decline = {
		custom_tooltip = aeluran_compel_on_decline_tt
		hidden_effect = {
			aeluran_compel_begin_effect = { SISTER = scope:recipient TASK = summon }
		}
	}

	# A Sister bound to the shepherd always comes
	auto_accept = {
		custom_description = {
			text = aeluran_whispers_loyal_sister_auto_accept
			subject = scope:recipient
			object = scope:actor
			scope:recipient = { aeluran_whispers_loyal_sister_trigger = { SHEPHERD = scope:actor } }
		}
	}

	ai_accept = {
		base = 25
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}
		# Attachment to the ruler whose court she would leave
		opinion_modifier = {
			trigger = { exists = scope:recipient.employer }
			who = scope:recipient
			opinion_target = scope:recipient.employer
			multiplier = -0.25
			desc = aeluran_summon_attached_reason
		}
		modifier = {
			add = -25
			scope:recipient = {
				is_married = yes
				any_spouse = { NOT = { is_courtier_of = scope:actor } }
			}
			desc = aeluran_summon_spouse_reason
		}
		modifier = {
			add = -100
			scope:recipient = {
				OR = {
					has_relation_rival = scope:actor
					has_relation_nemesis = scope:actor
				}
			}
			desc = aeluran_whispers_rival_reason
		}
	}

	ai_will_do = {
		base = 0
	}
}

# Gather Whispers: set a Red Sister in the shepherd's care to spy on the court of the ruler she advises for a year
# (common/scripted_effects/aeluran_whispers_effects.txt). She may refuse a shepherd she dislikes, and one who resents
# her shepherd, or loves her ruler more, tells her ruler. A Sister bound to the shepherd always agrees.
aeluran_gather_whispers_interaction = {
	category = interaction_category_religion
	icon = icon_aeluran_sister
	interface_priority = 50
	common_interaction = no
	use_diplomatic_range = no
	desc = aeluran_gather_whispers_interaction_desc
	ai_instant_response = yes

	is_shown = {
		scope:actor = {
			OR = {
				has_title = title:d_high_matriarch
				any_held_title = { aeluran_is_matronate_title_trigger = yes }
			}
		}
		scope:recipient = {
			NOT = { this = scope:actor }
			aeluran_red_sister_in_care_trigger = { SHEPHERD = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_incapable = no
		}
		scope:recipient = {
			custom_tooltip = {
				text = aeluran_whispers_already_tt
				NOT = { has_variable = aeluran_whispers_for }
			}
			is_imprisoned = no
		}
	}

	cooldown_against_recipient = { years = 1 }

	cost = {
		piety = {
			value = scope:recipient.aeluran_whispers_piety_cost
		}
	}

	auto_accept = {
		custom_description = {
			text = aeluran_whispers_loyal_sister_auto_accept
			subject = scope:recipient
			object = scope:actor
			scope:recipient = { aeluran_whispers_loyal_sister_trigger = { SHEPHERD = scope:actor } }
		}
	}

	on_accept = {
		scope:recipient = { custom_tooltip = aeluran_whispers_accept_tt }
		hidden_effect = { aeluran_whispers_start_effect = yes }
	}

	# She defies you: let it stand (piety back; she may tell her ruler) or compel her (aeluran_compel.001)
	on_decline = {
		custom_tooltip = aeluran_compel_on_decline_tt
		if = {
			limit = {
				scope:recipient = { aeluran_whispers_betrayal_trigger = { SHEPHERD = scope:actor } }
			}
			scope:recipient = { custom_tooltip = aeluran_whispers_betrayal_tt }
		}
		hidden_effect = {
			aeluran_compel_begin_effect = { SISTER = scope:recipient TASK = whispers }
		}
	}

	ai_accept = {
		base = 20
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:recipient.liege
			multiplier = -0.25
			desc = aeluran_whispers_loyal_to_ruler_reason
		}
		modifier = {
			add = -100
			scope:recipient = {
				OR = {
					has_relation_rival = scope:actor
					has_relation_nemesis = scope:actor
				}
			}
			desc = aeluran_whispers_rival_reason
		}
		modifier = {
			add = 15
			scope:recipient = { has_trait = deceitful }
			desc = aeluran_whispers_deceitful_reason
		}
		modifier = {
			add = -20
			scope:recipient = { has_trait = honest }
			desc = aeluran_whispers_honest_reason
		}
		modifier = {
			add = 10
			scope:recipient = { has_trait = zealous }
			desc = aeluran_whispers_zealous_reason
		}
	}

	ai_will_do = {
		base = 0
	}
}

# Press the Order's Claim: a shepherd, backed by the ruler's Red Sister, urges a ruler of her flock to raise Aeluran
# Realm Authority one step. A refusal can be overridden in a battle of wills (aeluran_compel.010), fought by the
# shepherd or by the Red Sister. AI shepherds send it from aeluran_matronate_ai_yearly.
aeluran_press_authority_interaction = {
	category = interaction_category_religion
	icon = icon_aeluran_matron
	interface_priority = 55
	common_interaction = no
	use_diplomatic_range = no
	desc = aeluran_press_authority_interaction_desc

	is_shown = {
		scope:actor = {
			OR = {
				has_title = title:d_high_matriarch
				any_held_title = { aeluran_is_matronate_title_trigger = yes }
			}
		}
		scope:recipient = {
			aeluran_press_authority_target_trigger = { SHEPHERD = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_incapable = no
		}
	}

	cooldown_against_recipient = { years = 5 }

	cost = {
		piety = {
			value = scope:recipient.aeluran_press_authority_piety_cost
		}
	}

	send_option = {
		is_valid = {
			scope:actor = { has_usable_hook = scope:recipient }
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	on_accept = {
		scope:recipient = { custom_tooltip = aeluran_press_authority_accept_tt }
		if = {
			limit = {
				scope:hook = yes
				scope:actor = { has_usable_hook = scope:recipient }
			}
			scope:actor = { use_hook = scope:recipient }
		}
		hidden_effect = { aeluran_press_authority_success_effect = yes }
	}

	# Refused: let it stand, or force it in a battle of wills (aeluran_compel.010)
	on_decline = {
		custom_tooltip = aeluran_press_authority_on_decline_tt
		hidden_effect = {
			scope:recipient = { save_scope_as = compel_ruler }
			scope:actor = { trigger_event = aeluran_compel.010 }
		}
	}

	# A strong hook cannot be refused
	auto_accept = {
		custom_description = {
			text = aeluran_press_strong_hook_auto_accept
			subject = scope:actor
			object = scope:recipient
			scope:hook = yes
			scope:actor = { has_strong_hook = scope:recipient }
		}
	}

	ai_accept = {
		base = -25
		modifier = {
			add = -25
			scope:recipient = { has_realm_law = aeluran_realm_authority_2 }
			desc = aeluran_press_judiciary_reason
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}
		# Devotion to the Order (spiritual fulfillment -100..100)
		modifier = {
			add = {
				value = scope:recipient.spiritual_fulfillment
				multiply = 0.3
			}
			desc = aeluran_press_devotion_reason
		}
		# The Red Sister at court: argues the shepherd's case, or (gone native) the ruler's
		modifier = {
			add = scope:recipient.cp:councillor_aeluran_advisor.aeluran_advisor_advocacy_value
			scope:recipient.cp:councillor_aeluran_advisor ?= {
				aeluran_advisor_native_gap <= 20
			}
			desc = aeluran_press_red_sister_for_reason
		}
		opinion_modifier = {
			trigger = {
				scope:recipient.cp:councillor_aeluran_advisor ?= {
					aeluran_advisor_native_gap <= 20
				}
			}
			who = scope:recipient
			opinion_target = scope:recipient.cp:councillor_aeluran_advisor
			multiplier = 0.25
			desc = aeluran_press_red_sister_trust_reason
		}
		modifier = {
			add = {
				value = scope:recipient.cp:councillor_aeluran_advisor.aeluran_advisor_advocacy_value
				multiply = -1
			}
			scope:recipient.cp:councillor_aeluran_advisor ?= {
				aeluran_advisor_native_gap > 20
			}
			desc = aeluran_press_red_sister_against_reason
		}
		modifier = {
			add = -20
			NOT = { exists = scope:recipient.cp:councillor_aeluran_advisor }
			desc = aeluran_press_no_red_sister_reason
		}
		modifier = {
			add = 50
			scope:hook = yes
			desc = SCHEME_WEAK_HOOK_USED
		}
		modifier = {
			add = 20
			scope:recipient = { is_at_war = yes }
			desc = aeluran_press_at_war_reason
		}
		modifier = {
			add = 15
			scope:recipient = { has_trait = zealous }
			desc = aeluran_press_zealous_reason
		}
		modifier = {
			add = -15
			scope:recipient = { has_trait = cynical }
			desc = aeluran_press_cynical_reason
		}
		modifier = {
			add = -10
			scope:recipient = { has_trait = ambitious }
			desc = aeluran_press_ambitious_reason
		}
		modifier = {
			add = -10
			scope:recipient = { has_trait = arrogant }
			desc = aeluran_press_arrogant_reason
		}
	}

	ai_will_do = {
		base = 100
	}
}
