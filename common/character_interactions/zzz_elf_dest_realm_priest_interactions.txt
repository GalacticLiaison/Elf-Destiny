# Override of vanilla request_realm_priest_replacement_interaction (game/common/character_interactions/pam_interactions.txt, CK3 1.20.0).
# Only change: is_available also requires an actual court chaplain to replace, and excludes Aeluran faiths
# (their Red Sister, councillor_aeluran_advisor, replaces the chaplain). Without it the AI evaluates
# is_valid_showing_failures_only with cp:councillor_court_chaplain unset -> "Event target link 'cp' returned an unset scope".
# zzz_ prefix: must load after the vanilla pam_interactions.txt to win. Re-diff against vanilla after every CK3 patch.

request_realm_priest_replacement_interaction = {
	category = interaction_category_religion
	icon = icon_request_priest
	interface_priority = 100
	common_interaction = yes
	notification_text = REQUEST_REALM_PRIEST_REPLACEMENT_NOTIFICATION

	desc = request_realm_priest_replacement_interaction_desc

	redirect = {
		# Prioritize direct superior of chaplain, over e.g. head of faith
		# But still allow right clicking on anyone in religious hierachy
		scope:actor = {
			if = {
				limit = {
					# Standard recipient triggers are true...
					can_request_priest_from_recipient_trigger = { RECIPIENT = scope:recipient }
				}
				# A realm under an antipope has only them to ask, so anyone else clicked is left to fail validity
				if = {
					limit = { is_following_religious_head_challenger = yes }
					if = {
						limit = { cp:councillor_court_chaplain ?= scope:recipient }
						religious_head_or_challenger ?= { save_scope_as = recipient }
					}
				}
				# Ensure direct superior of chaplain is always prioritized...
				else_if = {
					limit = {
						# Selected recipient is not direct superior of chaplain
						exists = cp:councillor_court_chaplain.superior
						cp:councillor_court_chaplain.superior != scope:recipient
					}
					cp:councillor_court_chaplain.superior = { save_scope_as = recipient }
				}
				# ...otherwise ensure the recipient is not the current chaplain
				else_if = {
					limit = { cp:councillor_court_chaplain ?= scope:recipient }
					if = {
						limit = { exists = faith.religious_head }
						faith.religious_head = { save_scope_as = recipient }
					}
					else_if = {
						limit = { exists = faith.main_rite.head_of_rite }
						faith.main_rite.head_of_rite = { save_scope_as = recipient }
					}
				}
			}
		}
	}

	populate_recipient_list = {
		scope:actor = {
			every_vassal_or_below = {
				every_courtier = { # All non-councillor realm courtiers of the actor's vassals
					limit = {
						can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
						is_councillor = no
					}
					add_to_list = characters
				}
			}
			every_courtier = { # All non-councillor realm courtiers of the actor
				limit = {
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
					is_councillor = no
				}
				add_to_list = characters
			}
		}
		scope:recipient = {
			every_vassal_or_below = {
				every_courtier = { # All non-councillor realm courtiers of the recipient's vassals
					limit = {
						can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
						is_councillor = no
					}
					add_to_list = characters
				}
			}
			every_courtier = { # All non-councillor realm courtiers of the recipient
				limit = {
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
					is_councillor = no
				}
				add_to_list = characters
			}
		}
		scope:actor = {
			every_vassal_or_below = { # All non-councillor theocratic vassals of the actor
				limit = {
					government_has_flag = government_is_theocracy
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
					is_councillor = no
				}
				add_to_list = characters
			}
			#Monk house members too!
			house ?= {
				every_house_member = {
					limit = {
						has_trait = devoted
						in_diplomatic_range = scope:actor
						is_adult = yes
						is_ruler = no
						can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
						is_councillor = no
						trigger_if = {
							limit = {
								exists = liege
								liege = {
									NOT = {
										this = scope:actor
									}
								}
							}
							liege = {
								is_ai = yes
							}
						}
						NOT = {
							has_relation_rival = scope:actor
						}
					}
					add_to_list = characters
				}
			}
			every_child = {
				limit = {
					has_trait = devoted	
				}
			}
		}
	}

	is_available = {
		# Temporal/Conditional use council Appoint, which fires on_fired_from_position instead
		NOT = { faith = { has_doctrine_parameter = clerical_appointment_ruler_can_replace_ecc } }
		government_allows = council
		# Nomads have no Court Chaplain position to replace, so they cannot request a new Realm Priest
		NOT = { government_has_flag = government_is_nomadic }
		has_pam_dlc_trigger = yes
		# ELF DESTINY: nothing to replace without a chaplain; Aeluran rulers have a Red Sister instead
		exists = cp:councillor_court_chaplain
		NOT = { faith = { has_tenet = tenet_aeluran_sisters } }
	}

	is_shown = {
		# The Head of Faith appoints their own chaplain freely via the council instead of petitioning themselves
		scope:recipient != scope:actor
		# Allow right clicking on anyone in religious hierachy
		scope:actor = {
			can_request_priest_from_recipient_trigger = { RECIPIENT = scope:recipient }
		}
	}

	cooldown_against_recipient = { years = 5 }

	is_valid_showing_failures_only = {
		scope:actor = {
			trigger_if = {
				limit = {
					NOT = {
						government_has_flag = government_is_theocracy
					}
				}
				custom_tooltip = {
					text = piety_level_or_external_puppet_req_tt
					OR = {
						exists = puppet:external_theological_agent_puppet
						piety_level >= 3
					}
				}
			}
			trigger_else = {
				piety_level >= 1
			}
			pam_antipope_clerical_request_trigger = { RECIPIENT = scope:recipient }
			custom_tooltip = {
				text = pam_blocked_by_promised_pilgrimage_tt
				NOT = { has_character_modifier = promised_pilgrimage_to_clergy_modifier }
			}
		}
		custom_tooltip = {
			text = recipient_will_not_replace_self_tt
			scope:recipient != scope:actor.cp:councillor_court_chaplain
		}
	}

	can_send = {
		custom_tooltip = {
			text = recipient_will_not_replace_self_tt
			scope:recipient != scope:actor.cp:councillor_court_chaplain
		}
	}

	cost = {
		piety = {
			value = 0
			# Change the piety loss amount based on the candidate's education level
			if = {
				limit = {
					scope:secondary_recipient ?= { has_education_rank_trigger = { RANK = 1 OPERATOR = equals } }
				}
				add = minor_piety_value
			}
			if = {
				limit = {
					scope:secondary_recipient ?= { has_education_rank_trigger = { RANK = 2 OPERATOR = equals } }
				}
				add = medium_piety_value
			}
			if = {
				limit = {
					scope:secondary_recipient ?= { has_education_rank_trigger = { RANK = 3 OPERATOR = equals } }
				}
				add = major_piety_value
			}
			if = {
				limit = {
					scope:secondary_recipient ?= { has_education_rank_trigger = { RANK = 4 OPERATOR = equals } }
				}
				add = massive_piety_value
			}
			if = {
				limit = {
					scope:secondary_recipient ?= { has_education_rank_trigger = { RANK = 5 OPERATOR = equals } }
				}
				add = monumental_piety_value
			}
			#Costs more to send for your monk relatives
			if = {
				limit = {
					scope:secondary_recipient = {
						NOR = {
							is_vassal_or_below_of = scope:actor
							is_courtier_of = scope:actor
						}
					}
				}
				multiply = 1.5
			}
			if = {
				limit = {
					scope:actor = {
						domicile ?= { has_domicile_parameter = estate_cheaper_realm_priest }
					}
				}
				multiply = ecclesiastical_domicile_cheaper_realm_priest_mult
			}
		}
	}

	on_accept = {
		scope:actor = {
			# Remove the current realm priest and make the candidate the realm priest
			if = {
				limit = {
					scope:secondary_recipient = {
						NOT = {
							is_courtier_of = scope:actor
						}
						is_ruler = no
					}
				}
				add_courtier = scope:secondary_recipient # Move the character to the court first if he's from another court and not a ruler.
			}
			cp:councillor_court_chaplain = {
				save_scope_as = old_chaplain
			}
			# Flag so on_fired_from_position does not treat this as the crown acting
			add_character_flag = { flag = pam_church_replacing_realm_priest days = 1 }
			assign_councillor_type = {
				type = councillor_court_chaplain
				remove_existing_councillor = yes
				target = scope:secondary_recipient
			}
			remove_character_flag = pam_church_replacing_realm_priest
			send_interface_toast = {
				type = event_toast_effect_good
				title = new_realm_priest_toast_title
				left_icon = scope:actor
				right_icon = scope:secondary_recipient
				# Make sure the candidate is grateful
				scope:actor = {
					reverse_add_opinion = {
						target = scope:secondary_recipient
						modifier = grateful_opinion
						opinion = 40
					}
				}
				if = {
					limit = {
						scope:use_hook = yes
						has_usable_hook = scope:recipient
					}
					use_hook = scope:recipient
				}
			}
			scope:old_chaplain = {
				add_opinion = { # Make sure your kicked realm priest is pissed
					target = scope:actor
					modifier = fp3_resentful_opinion
					opinion = -40
				}
			}
		}
		pam_offer_pilgrimage_effect = {
			ACTOR = scope:actor
			CLERIC = scope:recipient
		}
		# Catalysts
		if = {
			limit = {
				scope:actor.religion = religion:christianity_religion
				# Only rulers actually taking part in the situation can move it
				scope:actor = { exists = top_participant_group:the_christian_church }
				situation:the_christian_church ?= { situation_top_has_catalyst = catalyst_the_christian_church_requested_realm_priest_replacement }
			}
			situation:the_christian_church = { trigger_situation_catalyst = catalyst_the_christian_church_requested_realm_priest_replacement }
		}
	}

	on_decline = {

	}

	auto_accept = {
		# Using a hook should make him auto accept
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:use_hook = yes
			scope:actor = { has_strong_hook = scope:recipient }
		}
	}

	ai_accept = {
		base = 0

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:puppet_or_actor
			RECIPIENT = scope:recipient
		}

		# Weak hook: extra points
		modifier = {
			scope:use_hook = yes
			add = 200
			scope:actor = { NOT = { has_strong_hook = scope:recipient } }
			desc = SCHEME_WEAK_HOOK_USED
		}

		modifier = {
			add = pam_offer_pilgrimage_acceptance_value
			scope:offer_pilgrimage = yes
			desc = OFFER_PILGRIMAGE_INTERACTION_OPTION_REASON
		}

		# Respect the liege
		modifier = {
			add = 100
			scope:recipient = { target_is_liege_or_above = scope:actor }
			desc = LIEGE_REASON
		}

		modifier = {
			add = ecclesiastical_domicile_acceptance_minor_value_1
			desc = ESTATE_REALM_PRIEST_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_easier_realm_priest_replacement_1
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_minor_value_2
			desc = ESTATE_REALM_PRIEST_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_easier_realm_priest_replacement_2
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_minor_value_3
			desc = ESTATE_REALM_PRIEST_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_easier_realm_priest_replacement_3
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_1
			desc = ESTATE_SEMINARY_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_confessor_seminary_1
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_2
			desc = ESTATE_SEMINARY_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_confessor_seminary_2
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_3
			desc = ESTATE_SEMINARY_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_confessor_seminary_3
				}
			}
		}
		ecclesiastical_domicile_cathedral_awe_modifier = yes

		# Actor is from the recipient's house
		modifier = {
			add = 10
			exists = scope:actor.house
			scope:recipient = {
				OR = {
					is_close_family_of = scope:actor
					is_extended_family_of = scope:actor
				}
			}
			desc = YOU_ARE_MY_HOUSE_MEMBER
		}

		# Candidate is from the recipient's house
		modifier = {
			add = 10
			exists = scope:secondary_recipient
			exists = scope:secondary_recipient.house
			scope:recipient = {
				OR = {
					is_close_family_of = scope:secondary_recipient
					is_extended_family_of = scope:secondary_recipient
				}
			}
			desc = THEY_ARE_MY_HOUSE_MEMBER
		}

		# I have sinful traits
		modifier = {
			add = {
				add = -10
				multiply = scope:actor.num_sinful_traits
			}
			scope:actor.num_sinful_traits > 0
			desc = I_AM_SINFUL
		}

		# Candidate has sinful traits
		modifier = {
			exists = scope:secondary_recipient
			add = {
				add = -10
				multiply = scope:secondary_recipient.num_sinful_traits
			}
			scope:secondary_recipient.num_sinful_traits > 0
			desc = THEY_ARE_SINFUL
		}

		# I have virtuous traits
		modifier = {
			add = {
				add = 10
				multiply = scope:actor.num_virtuous_traits
			}
			scope:actor.num_virtuous_traits > 0
			desc = I_AM_VIRTUOUS
		}

		# Candidate has virtuous traits
		modifier = {
			exists = scope:secondary_recipient
			add = {
				add = 10
				multiply = scope:secondary_recipient.num_virtuous_traits
			}
			scope:secondary_recipient.num_virtuous_traits > 0
			desc = THEY_ARE_VIRTUOUS
		}

		# Religious superior's opinion of the actor matters!
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 1
			desc = AI_OPINION_REASON
		}

		# Religious Superior likes candidate more than the realm priest
		modifier = {
			add = 20
			exists = scope:secondary_recipient
			scope:recipient = {
				save_temporary_opinion_value_as = {
					name = secondary_recipient_opinion
					target = scope:secondary_recipient
				}
				save_temporary_opinion_value_as = {
					name = realm_priest_opinion
					target = scope:actor.cp:councillor_court_chaplain
				}
			}
			scope:secondary_recipient_opinion > {
			   add = scope:realm_priest_opinion
			   add = 20
			}
			desc = AI_OPINION_REASON_CANDIDATE_OVER_REALM_PRIEST
		}

		# Religious Superior likes realm priest more than the candidate
		modifier = {
			add = -20
			exists = scope:secondary_recipient
			scope:recipient = {
				save_temporary_opinion_value_as = {
					name = secondary_recipient_opinion
					target = scope:secondary_recipient
				}
				save_temporary_opinion_value_as = {
					name = realm_priest_opinion
					target = scope:actor.cp:councillor_court_chaplain
				}
			}
			scope:realm_priest_opinion > {
			   add = scope:secondary_recipient_opinion
			   add = 20
			}
			desc = AI_OPINION_REASON_REALM_PRIEST_OVER_CANDIDATE
		}

		modifier = { # A trait in common with the candidate
			add = 5
			exists = scope:secondary_recipient
			scope:recipient = {
				number_of_personality_traits_in_common = {
					target = scope:secondary_recipient
					value = 1
				}
			}
			desc = AI_COMMON_TRAITS
		}

		modifier = { # Two trait in common
			add = 10
			exists = scope:secondary_recipient
			scope:recipient = {
				number_of_personality_traits_in_common = {
					target = scope:secondary_recipient
					value = 2
				}
			}
			desc = AI_COMMON_TRAITS
		}

		modifier = { # Three traits in common
			add = 15
			exists = scope:secondary_recipient
			scope:recipient = {
				number_of_personality_traits_in_common = {
					target = scope:secondary_recipient
					value = 3
				}
			}
			desc = AI_COMMON_TRAITS
		}

		# Modifier based on Piety Level of the actor
		modifier = {
			add = {
				add = 5
				multiply = scope:actor.piety_level
			}
			scope:actor.piety_level > 1
			desc = I_AM_PIOUS
		}

		# Modifier based on Piety Level of the candidate
		modifier = {
			exists = scope:secondary_recipient
			add = {
				add = 5
				multiply = scope:secondary_recipient.piety_level
			}
			scope:secondary_recipient.piety_level > 1
			desc = THEY_ARE_PIOUS
		}

		# More likely to accept if you have an alliance
		modifier = {
			add = 40
			scope:actor = { is_allied_to = scope:recipient }
			desc = AI_HAS_ALLIANCE
		}

		# Fs
		modifier = {
			add = 25
			scope:recipient = {
				has_relation_friend = scope:actor
			}
			desc = WE_ARE_FRIENDS
		}

		# BFFs
		modifier = {
			add = 50
			scope:recipient = {
				has_relation_best_friend = scope:actor
			}
			desc = WE_ARE_BEST_FRIENDS
		}

		# Take dread into account
		modifier = {
			add = 50
			scope:recipient = {
				target_is_liege_or_above = scope:actor
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}

		modifier = {
			add = 100
			scope:recipient = {
				target_is_liege_or_above = scope:actor
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}

		# Below 10 Learning for negative, Above 10 Learning for positive
		modifier = {
			add = {
				value = scope:secondary_recipient.learning
				subtract = 10
				multiply = 2
			}
			trigger = {
				scope:secondary_recipient ?= {
					learning != 10
				}
			}
			desc = THEY_HAVE_LEARNING_SKILL_REASON
		}

		# Is a monk
		modifier = {
			add = 25
			scope:secondary_recipient ?= {
				has_trait = erudite
			}
			desc = THEY_HAVE_ERUDITE_TRAIT
		}
	}

	send_option = {
		is_shown = {
			scope:recipient != scope:actor
		}
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
			trigger_if = {
				limit = {
					scope:recipient = {
						is_ai = no
					}
				}
				scope:actor = { is_ai = no }
			}
		}
		flag = use_hook
		localization = EDUCATION_HOOK
	}

	send_option = {
		is_shown = {
			pam_offer_pilgrimage_option_shown_trigger = { ACTOR = scope:actor }
		}
		is_valid = {
			pam_offer_pilgrimage_option_valid_trigger = { ACTOR = scope:actor }
		}
		flag = offer_pilgrimage
		localization = OFFER_PILGRIMAGE_INTERACTION_OPTION
	}

	should_use_extra_icon = {

	ai_frequency_by_tier = {

		kingdom = 60
		empire = 60
		hegemony = 60
	}

	ai_will_do = {
		base = 0
		opinion_modifier = { # I dislike my realm priest
			opinion_target = scope:actor.cp:councillor_court_chaplain
			who = scope:actor
			multiplier = -1
		}
		opinion_modifier = { # My realm priest dislikes me
			opinion_target = scope:actor
			who = scope:actor.cp:councillor_court_chaplain
			multiplier = -1
		}
		modifier = {
			add = 50
			desc = request_realm_priest_replacement_ai_different_rite
			exists = scope:actor.cp:councillor_court_chaplain
			scope:actor.cp:councillor_court_chaplain.rite != scope:actor.rite
		}
	}
