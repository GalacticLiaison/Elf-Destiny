# The High Matriarch's Matronate powers (effects in common/scripted_effects/aeluran_matronate_scripted_effects.txt).
# Both are used on the Sister or Matron who is to receive the Matronate; a Sister is raised to Matron.
# The AI High Matriarch uses the same effects from a yearly pulse instead (aeluran_matronate_ai_yearly_effect).

# Found a new Matronate around the recipient's home county
aeluran_found_matronate_interaction = {
	category = interaction_category_religion
	icon = icon_clergy_claim_unformed_kingdom
	interface_priority = 45
	common_interaction = no
	use_diplomatic_range = no
	popup_on_receive = yes
	desc = aeluran_found_matronate_interaction_desc

	is_shown = {
		scope:actor = { has_title = title:d_high_matriarch }
		scope:recipient = {
			this != scope:actor
			aeluran_may_receive_matronate_trigger = yes
		}
	}

	cooldown = { years = 5 }

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_incapable = no
		}
		scope:recipient = {
			is_adult = yes
			is_imprisoned = no
			custom_tooltip = {
				text = aeluran_matronate_home_county_tt
				aeluran_matronate_home_county_foundable_trigger = yes
			}
			custom_tooltip = {
				text = aeluran_matronate_break_realm_tt
				aeluran_matronate_would_break_realm_trigger = no
			}
		}
	}

	cost = {
		piety = aeluran_matronate_found_piety_cost
	}

	on_accept = {
		scope:recipient = {
			if = {
				limit = { NOT = { has_trait = aeluran_matron } }
				custom_tooltip = aeluran_matronate_raised_tt
			}
			custom_tooltip = aeluran_matronate_found_tt
			hidden_effect = { aeluran_matronate_found_effect = yes }
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

# Entrust an existing Matronate to the recipient, deposing its Matron
aeluran_entrust_matronate_interaction = {
	category = interaction_category_religion
	icon = icon_clergy_move_lands
	interface_priority = 44
	common_interaction = no
	use_diplomatic_range = no
	popup_on_receive = yes
	desc = aeluran_entrust_matronate_interaction_desc
	target_type = title
	target_filter = clerical_region
	show_all_targets = no

	is_shown = {
		scope:actor = { has_title = title:d_high_matriarch }
		scope:recipient = {
			this != scope:actor
			aeluran_may_receive_matronate_trigger = yes
		}
	}

	cooldown = { years = 10 }

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_incapable = no
		}
		scope:recipient = {
			is_adult = yes
			is_imprisoned = no
			custom_tooltip = {
				text = aeluran_matronate_break_realm_tt
				aeluran_matronate_would_break_realm_trigger = no
			}
		}
		trigger_if = {
			limit = {
				exists = scope:target
				scope:recipient = { any_held_title = { is_landless_type_title = no } }
			}
			custom_tooltip = {
				text = aeluran_matronate_entrust_realm_outside_tt
				scope:recipient = { aeluran_lives_in_target_matronate_trigger = yes }
			}
		}
	}

	# Every Matronate but one the High Matriarch holds; a Sister who rules land may only take the one her realm lies in
	can_be_picked_title = {
		scope:target = {
			aeluran_is_matronate_title_trigger = yes
			NOT = { holder ?= scope:actor }
		}
		trigger_if = {
			limit = { scope:recipient = { any_held_title = { is_landless_type_title = no } } }
			scope:recipient = { aeluran_lives_in_target_matronate_trigger = yes }
		}
	}

	cost = {
		piety = aeluran_matronate_entrust_piety_cost
		prestige = aeluran_matronate_entrust_prestige_cost
	}

	on_accept = {
		scope:recipient = {
			if = {
				limit = { NOT = { has_trait = aeluran_matron } }
				custom_tooltip = aeluran_matronate_raised_tt
			}
			custom_tooltip = aeluran_matronate_entrust_tt
		}
		scope:target.holder ?= {
			custom_tooltip = aeluran_matronate_deposed_tt
		}
		hidden_effect = {
			scope:recipient = { aeluran_matronate_entrust_effect = yes }
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

# An Aeluran ruler asks the High Matriarch for a Matronate around the realm's capital (PM 2026-10-01). The realm must
# hold three or more Aeluran counties no Matronate covers. Effects in aeluran_matronate_petition_effect.
aeluran_petition_matronate_interaction = {
	category = interaction_category_religion
	icon = icon_clergy_claim_unformed_kingdom
	interface_priority = 40
	common_interaction = no
	use_diplomatic_range = no
	desc = aeluran_petition_matronate_interaction_desc

	is_shown = {
		scope:recipient = { has_title = title:d_high_matriarch }
		scope:actor = {
			this != scope:recipient
			aeluran_can_petition_matronate_trigger = yes
		}
	}

	cooldown = { years = 5 }

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_incapable = no
			custom_tooltip = {
				text = aeluran_matronate_petition_counties_tt
				aeluran_realm_wants_matronate_trigger = yes
			}
			custom_tooltip = {
				text = aeluran_matronate_petition_capital_tt
				capital_county ?= { aeluran_matronate_foundable_county_trigger = yes }
			}
		}
		scope:recipient = {
			is_imprisoned = no
		}
	}

	cost = {
		piety = aeluran_matronate_petition_piety_cost
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = aeluran_matronate_petition_tt
			hidden_effect = {
				scope:recipient = { save_scope_as = matriarch }
				aeluran_matronate_petition_effect = yes
			}
		}
		scope:recipient = {
			add_piety = { value = aeluran_matronate_petition_piety_cost multiply = 0.5 }
		}
	}

	ai_accept = {
		base = 10
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 25
			desc = aeluran_matronate_petition_ai_elves_tt
			scope:actor = { is_any_elf_type = yes }
		}
		modifier = {
			add = 15
			desc = aeluran_matronate_petition_ai_pious_tt
			scope:actor = { piety_level >= 2 }
		}
	}

	ai_potential = {
		is_independent_ruler = yes
		aeluran_can_petition_matronate_trigger = yes
		aeluran_realm_wants_matronate_trigger = yes
		capital_county ?= { aeluran_matronate_foundable_county_trigger = yes }
	}
	ai_targets = {
		ai_recipients = head_of_faith
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
	}
}
