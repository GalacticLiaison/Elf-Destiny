#########################################################################
# Dark Elf Beta - rotated full-body cameras for the ruler designer.
# Vanilla camera_ruler_designer is cylindrical{ 390 -22 0 } -- angle 0, dead
# front -- and the designer's body view is hardcoded to it, so back and arm
# tattoos can never be seen while choosing them. These are the same camera at
# quarter turns; the GUI appends the suffix to 'camera_ruler_designer'.
#########################################################################

camera_ruler_designer_r90 = {
	camera = {
		position = cylindrical{ 390 -22 90 } # Radius, height, angle
		position_node = {
			default = camera_torso_look_at
		}

		look_at = {	0 -32 0 } # L/R, U/D, F/B
		look_at_node = {
			default = camera_torso_look_at
		}
		fov = 20
		camera_near_far = { 10 800 }
	}

	unknown = camera_body
}
