# Dark Elf Beta
# DNA-first dark elf skin rendering. A character's authored/inherited
# gene_special_skin_color DNA is the default source of truth for their look.
# The forced entries below apply ONLY to characters carrying
# var:dark_elf_skin_color (flag:purple / flag:grey / flag:blue) - i.e. CONVERTS
# whose DNA has no drow genes (story conversions, debug toggles, ruler-designer
# picks made without choosing a skin variant). The variable is set at conversion
# by dark_elf_init_skin_color_effect and inherited 50/50 at birth, so convert
# bloodlines breed true down the line. Forced entries render across the full
# curve band (range 0.0-1.0, seeded per character) and pin base skin/hair/eye
# coordinates to 0,0 so the drow shift lands in its intended palette.
# Everyone without the variable renders their DNA untouched.
#
# Bloodline sovereignty: this pack only sets gene_special_skin_color. The
# base-color pins (skin_color, hair_color, eye_color) that used to live inside
# each drow entry were extracted into their own packs - dark_elf_skin_color_pin,
# dark_elf_hair_color_pin, dark_elf_eye_color_pin - so they can independently
# opt out when a bloodline marks the same gene (Thundarael for skin_color,
# Valerith for hair_color, Serelion/Lormelis for eye_color). Splitting
# the pins into separate packs is the only way to conditionally skip them:
# within a single pack entry the dna_modifiers apply all-or-nothing. Ported
# from Elf_Destiny_Lite_V2 (see UPSTREAM_DIVERGENCES.md item #6).

dark_elf_race_skin = {
    usage = game

    dark_elf_skin_purple = {
        dna_modifiers = {
            morph = {
                mode = replace
                gene = gene_special_skin_color
                template = drow_skin_purple
                range = { 0.0 1.0 }
            }
        }
        weight = {
            base = 0
            modifier = {
                add = 999
                is_dark_elf_type = yes
                exists = var:dark_elf_skin_color
                var:dark_elf_skin_color = flag:purple
            }
        }
    }

    dark_elf_skin_grey = {
        dna_modifiers = {
            morph = {
                mode = replace
                gene = gene_special_skin_color
                template = drow_skin_grey
                range = { 0.0 1.0 }
            }
        }
        weight = {
            base = 0
            modifier = {
                add = 999
                is_dark_elf_type = yes
                exists = var:dark_elf_skin_color
                var:dark_elf_skin_color = flag:grey
            }
        }
    }

    dark_elf_skin_blue = {
        dna_modifiers = {
            morph = {
                mode = replace
                gene = gene_special_skin_color
                template = drow_skin_blue
                range = { 0.0 1.0 }
            }
        }
        weight = {
            base = 0
            modifier = {
                add = 999
                is_dark_elf_type = yes
                exists = var:dark_elf_skin_color
                var:dark_elf_skin_color = flag:blue
            }
        }
    }

    # No-op winner for everyone else (mode = add, value = 0 changes nothing).
    dark_elf_skin_unchanged = {
        dna_modifiers = {
            morph = {
                mode = add
                gene = gene_special_skin_color
                value = 0
            }
        }
        weight = {
            base = 1
        }
    }
}

# ============================================================================
# Base skin_color pin (extracted from the drow entries above).
# Pins skin_color to (0,0) so the drow special-skin template renders on the
# intended base. Skips when Thundarael is present - her bloodline marks
# skin_color at (0.001, 0.001) for pale-skin appearance and must win.
#
# No-op fallback uses morph on gene_special_skin_color (mode=add value=0)
# instead of a color-gene no-op. `color = { mode = add }` is not a supported
# no-op form - it silently corrupts rendering when a color gene sits at base
# weight with mode=add. Morph-based no-op mirrors the pattern in the main pack.
# ============================================================================
dark_elf_skin_color_pin = {
    usage = game

    pin_active = {
        dna_modifiers = {
            color = {
                gene = skin_color
                mode = replace
                x = 0.0
                y = 0.0
            }
        }
        weight = {
            base = 0
            modifier = {
                add = 999
                is_dark_elf_type = yes
                exists = var:dark_elf_skin_color
                NOT = { has_trait = royal_elf_thundarael }
            }
        }
    }

    pin_inactive = {
        dna_modifiers = {
            morph = {
                mode = add
                gene = gene_special_skin_color
                value = 0
            }
        }
        weight = {
            base = 1
        }
    }
}

# ============================================================================
# Base-tone pin for ruler-designer dark elves and their descendants.
# var:dark_elf_base_tone marks characters whose base skin/hair/eye colors
# should be pinned to 0,0 so their DNA base colors don't skew the drow HSV
# shift too dark. Previously this was a single all-or-nothing pack entry on
# all three color genes; split here into one pack per axis so bloodline guards
# can skip the right axis independently. Loses to the forced color entries
# (999) when a convert carries both variables - those already include the
# same pins. Converts don't need this pin: their forced color entries carry
# the same pins.
# ============================================================================
dark_elf_base_tone_skin_pin = {
    usage = game

    pin_active = {
        dna_modifiers = {
            color = {
                gene = skin_color
                mode = replace
                x = 0.0
                y = 0.0
            }
        }
        weight = {
            base = 0
            modifier = {
                add = 500
                exists = this
                is_alive = yes # dead characters can't hold variables
                has_variable = dark_elf_base_tone
                NOT = { has_trait = royal_elf_thundarael }
            }
        }
    }

    pin_inactive = {
        dna_modifiers = {
            morph = {
                mode = add
                gene = gene_special_skin_color
                value = 0
            }
        }
        weight = {
            base = 1
        }
    }
}
