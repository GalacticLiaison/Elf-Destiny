# Dark Elf Beta
# The Spark judges the sigil-finder's soul and names them dawn- or dusk-touched;
# the expedition party transforms to match (or the ruler defies the verdict at a
# stress cost). Chain: expedition.00691 -> dark_elf_story.0001 -> expedition.00692.
# expedition.00691 itself is changed in the DEB same-path override copy of
# events/activities/expedition_activity/expedition_events.txt (single-event
# redefinition across files loses to the upstream definition, per error.log).

namespace = dark_elf_story

# The Judgment: the Spark weighs the ruler and delivers its verdict.
dark_elf_story.0001 = {
	type = activity_event

	title = dark_elf_story.0001.t
	desc = {
		desc = dark_elf_story.0001.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = { var:spark_judgment = flag:dark }
				desc = dark_elf_story.0001.desc.dark
			}
			desc = dark_elf_story.0001.desc.light
		}
	}

	theme = dead_elf_king_with_light

	left_portrait = {
		character = scope:host
		animation = shock
	}

	artifact = {
		target = scope:newly_created_artifact
		position = lower_center_portrait
	}

	immediate = {
		dark_elf_spark_judgment_effect = yes
	}

	# Judged dusk-touched: accept. The transformation happens in dark_elf_story.0002
	# once the ruler picks the party's hue.
	option = {
		trigger = { var:spark_judgment = flag:dark }
		name = dark_elf_story.0001.accept_dark

		custom_tooltip = dark_elf_choose_hue_tt
		remove_variable = spark_judgment

		ai_chance = { base = 100 }

		trigger_event = dark_elf_story.0002
	}

	# Judged dusk-touched: defy the verdict
	option = {
		trigger = { var:spark_judgment = flag:dark }
		name = dark_elf_story.0001.defy_dark

		stress_impact = {
			base = medium_stress_impact_gain
		}
		custom_tooltip = {
			text = all_party_members_transformed_into_elves
			dark_elf_sigil_transform_party_effect = { DARK = no }
		}
		remove_variable = spark_judgment

		ai_chance = { base = 0 }

		trigger_event = expedition.00692
	}

	# Judged dawn-touched: accept
	option = {
		trigger = { var:spark_judgment = flag:light }
		name = dark_elf_story.0001.accept_light

		custom_tooltip = {
			text = all_party_members_transformed_into_elves
			dark_elf_sigil_transform_party_effect = { DARK = no }
		}
		remove_variable = spark_judgment

		ai_chance = { base = 100 }

		trigger_event = expedition.00692
	}

	# Judged dawn-touched: defy the verdict. Transformation happens in dark_elf_story.0002.
	option = {
		trigger = { var:spark_judgment = flag:light }
		name = dark_elf_story.0001.defy_light

		stress_impact = {
			base = medium_stress_impact_gain
		}
		custom_tooltip = dark_elf_choose_hue_tt
		remove_variable = spark_judgment

		ai_chance = { base = 0 }

		trigger_event = dark_elf_story.0002
	}
}
