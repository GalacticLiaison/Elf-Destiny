# Task completed / failed. scope:deity_contract = the contract; the reward or penalty is already applied
###################

dark_elf_deity.3001 = {
	type = character_event
	title = dark_elf_deity.3001.t
	desc = {
		triggered_desc = {
			trigger = {
				scope:deity_contract = {
					OR = {
						has_task_contract_type = deity_task_melkor_break_them
						has_task_contract_type = deity_task_melkor_be_feared
					}
				}
			}
			desc = dark_elf_deity.3001.melkor
		}
		triggered_desc = {
			trigger = {
				scope:deity_contract = {
					OR = {
						has_task_contract_type = deity_task_lloth_thread_cut
						has_task_contract_type = deity_task_lloth_thread_tied
					}
				}
			}
			desc = dark_elf_deity.3001.lloth
		}
		triggered_desc = {
			trigger = {
				scope:deity_contract = {
					OR = {
						has_task_contract_type = deity_task_typhos_strong_blood
						has_task_contract_type = deity_task_typhos_shed_skin
					}
				}
			}
			desc = dark_elf_deity.3001.typhos
		}
	}
	theme = faith
	content_source = elf_destiny

	left_portrait = {
		character = root
		animation = personality_zealous
	}

	option = {
		name = dark_elf_deity.3001.a
	}
}

# A divine task pin was clicked. Root = the player, scope:pin_province = the province under the pin
###################

dark_elf_deity.5001 = {
	type = character_event
	title = dark_elf_deity.5001.t
	desc = {
		desc = dark_elf_deity.5001.desc
		triggered_desc = {
			trigger = { scope:pin_province = { has_variable = deity_offer_melkor_break_them } }
			desc = dark_elf_deity.5001.offer_melkor_break_them
		}
		triggered_desc = {
			trigger = { scope:pin_province = { has_variable = deity_offer_melkor_be_feared } }
			desc = dark_elf_deity.5001.offer_melkor_be_feared
		}
		triggered_desc = {
			trigger = { scope:pin_province = { has_variable = deity_offer_lloth_thread_cut } }
			desc = dark_elf_deity.5001.offer_lloth_thread_cut
		}
		triggered_desc = {
			trigger = { scope:pin_province = { has_variable = deity_offer_lloth_thread_tied } }
			desc = dark_elf_deity.5001.offer_lloth_thread_tied
		}
		triggered_desc = {
			trigger = { scope:pin_province = { has_variable = deity_offer_typhos_strong_blood } }
			desc = dark_elf_deity.5001.offer_typhos_strong_blood
		}
		triggered_desc = {
			trigger = { scope:pin_province = { has_variable = deity_offer_typhos_shed_skin } }
			desc = dark_elf_deity.5001.offer_typhos_shed_skin
		}
		triggered_desc = {
			trigger = { scope:pin_province = { has_variable = deity_pin_active } }
			desc = dark_elf_deity.5001.active
		}
		triggered_desc = {
			trigger = { scope:pin_province = { NOT = { has_variable = deity_pin_offers } } }
			desc = dark_elf_deity.5001.no_offers
		}
	}
	theme = faith
	content_source = elf_destiny

	left_portrait = {
		character = root
		animation = thinking
	}

	option = {
		name = dark_elf_deity.5001.melkor_break_them
		trigger = { scope:pin_province = { has_variable = deity_offer_melkor_break_them } }
		custom_tooltip = dark_elf_deity.task.melkor_break_them_tt
		accept_deity_offer_effect = { TASK = melkor_break_them YEARS = deity_task_years_break_them }
	}

	option = {
		name = dark_elf_deity.5001.melkor_be_feared
		trigger = { scope:pin_province = { has_variable = deity_offer_melkor_be_feared } }
		custom_tooltip = dark_elf_deity.task.melkor_be_feared_tt
		accept_deity_offer_effect = { TASK = melkor_be_feared YEARS = deity_task_years_be_feared }
	}

	option = {
		name = dark_elf_deity.5001.lloth_thread_cut
		trigger = { scope:pin_province = { has_variable = deity_offer_lloth_thread_cut } }
		custom_tooltip = dark_elf_deity.task.lloth_thread_cut_tt
		accept_deity_offer_effect = { TASK = lloth_thread_cut YEARS = deity_task_years_thread_cut }
	}

	option = {
		name = dark_elf_deity.5001.lloth_thread_tied
		trigger = { scope:pin_province = { has_variable = deity_offer_lloth_thread_tied } }
		custom_tooltip = dark_elf_deity.task.lloth_thread_tied_tt
		accept_deity_offer_effect = { TASK = lloth_thread_tied YEARS = deity_task_years_thread_tied }
	}

	option = {
		name = dark_elf_deity.5001.typhos_strong_blood
		trigger = { scope:pin_province = { has_variable = deity_offer_typhos_strong_blood } }
		custom_tooltip = dark_elf_deity.task.typhos_strong_blood_tt
		accept_deity_offer_effect = { TASK = typhos_strong_blood YEARS = deity_task_years_strong_blood }
	}

	option = {
		name = dark_elf_deity.5001.typhos_shed_skin
		trigger = { scope:pin_province = { has_variable = deity_offer_typhos_shed_skin } }
		custom_tooltip = dark_elf_deity.task.typhos_shed_skin_tt
		accept_deity_offer_effect = { TASK = typhos_shed_skin YEARS = deity_task_years_shed_skin }
	}

	option = {
		name = dark_elf_deity.5001.leave
	}
}
