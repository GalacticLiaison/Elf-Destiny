# The Order assigns a new Advisor; the letter comes from the Matron or High Matriarch who appointed her
# (scope:advisor_appointer, aeluran_fill_advisor_post_effect). A chosen Sister arrives as scope:aeluran_sister_advisor.
aeluran_diarchy.100 = {
	type = letter_event

	trigger = {
		NOT = {
			has_character_flag = aeluran_advisor_is_on_the_way
		}
		exists = scope:advisor_appointer
	}

	opening = aeluran_diarchy.100.title
	
	desc = {
		desc = aeluran_diarchy.100.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:advisor_type = flag:chosen }
				desc = aeluran_diarchy.100.desc.chosen
			}
			triggered_desc = {
				trigger = { 
					scope:advisor_type = flag:vassal
				}
				desc = aeluran_diarchy.100.desc.vassal
			}
			triggered_desc = {
				trigger = { 
					scope:advisor_type = flag:courtier
				}
				desc = aeluran_diarchy.100.desc.courtier
			}
			triggered_desc = {
				trigger = { 
					scope:advisor_type = flag:sent_from_deepwood
				}
				desc = aeluran_diarchy.100.desc.sent_from_deepwood
			}
		}
	}

	sender = scope:advisor_appointer

	# cooldown = { years = 1 } # is this global?

	immediate = {
		if = {
			limit = { NOT = { exists = scope:aeluran_sister_advisor } }
			determine_new_aeluran_advisor_effect = yes
		}
	}

	option = {
		name = aeluran_diarchy.100.ok

		if = {
			limit = {
				exists = scope:aeluran_sister_advisor
			}

			# DEBUG: will make advisor instant travel to your realm
			# if = {
			# 	limit = {
			# 		scope:aeluran_sister_advisor = {
			# 			NOT = {
			# 				is_courtier_of = root
			# 			}
			# 		}
			# 	}
			# 	add_courtier = scope:aeluran_sister_advisor
			# }

			if = {
				limit = {
					exists = scope:aeluran_advisor_coming_from_sanctum
					scope:aeluran_advisor_coming_from_sanctum = yes
				}
				add_character_flag = {
					flag = aeluran_advisor_is_on_the_way
					years = 4
				}

				scope:aeluran_sister_advisor = {
					set_variable = {
						name = new_liege_to_advise
						value = root
						# years = 5
					}
				}

				save_scope_as = lord_receiving_advisor

				aeluran_advisor_sent_from_deep_wood_effect = yes
			}
			else = {
				trigger_event = aeluran_diarchy.1011
			}

		}
	}
}

# Aeluran demands land
# A Matter of Land: the Red Sister and the realm's temples. Fires a month after she arrives (aeluran_diarchy.101).
# Her ruler's temples are leased to her by the engine (common/lease_contracts/elf_dest_aeluran_lease.txt), so when the
# realm holds a temple the event only explains the arrangement. When it holds none, the Order offers to raise one at
# its own expense (1031 when it stands), or the ruler postpones and the Order thinks less of them (1032, 1033).
# Until Beta_v2.0.4 the event also offered to give the Sister a temple outright. That made her a landed vassal of the
# ruler whose temple left the lease and paid the Order nothing, so it was dropped (PM 2026-10-05). Sisters who already
# hold such a temple keep it; the lease leaves it alone.
# Survey and settlement: common/scripted_effects/aeluran_matter_of_land_effects.txt.
aeluran_diarchy.103 = {
	type = character_event
	title = aeluran_diarchy.103.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:temple_in_care }
				desc = aeluran_diarchy.103.desc.in_care
			}
			triggered_desc = {
				trigger = { exists = scope:undeveloped_holding }
				desc = aeluran_diarchy.103.desc.build
			}
			desc = aeluran_diarchy.103.desc.none
		}
	}

	trigger = {
		scope:aeluran_sister_advisor = {
			is_alive = yes
		}
	}

	theme = aeluran_matriarch

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:aeluran_sister_advisor
		animation = personality_zealous
	}

	immediate = {
		aeluran_matter_of_land_survey_effect = yes
	}

	# The realm's temples are in her care through the lease: the matter is settled
	option = {
		name = aeluran_diarchy.103.in_care
		trigger = { exists = scope:temple_in_care }
		aeluran_matter_of_land_settled_effect = yes
	}

	# No temple in the realm: the Order raises one
	option = {
		name = aeluran_diarchy.103.undeveloped_holding
		trigger = {
			NOT = { exists = scope:temple_in_care }
			exists = scope:undeveloped_holding
		}
		scope:undeveloped_holding = {
			begin_create_holding = {
				type = church_holding
			}
		}
		trigger_event = {
			id = aeluran_diarchy.1031
			days = 1826
		}
		ai_chance = { base = 100 }
	}

	# Postponed: the Order thinks less of the ruler until a temple stands (faction_respect_values.txt)
	option = {
		name = aeluran_diarchy.103.will_assign
		trigger = {
			NOT = { exists = scope:temple_in_care }
		}
		add_character_flag = aeluran_penalty_has_unlanded_aeluran_advisor
		trigger_event = aeluran_diarchy.1032
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				NOT = { exists = scope:undeveloped_holding }
			}
		}
	}
}

aeluran_diarchy.1031 = {
	type = character_event
	title = aeluran_diarchy.1031.title
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					new_temple_available_for_diarch = { COUNTY = scope:undeveloped_holding.county PROVINCE = scope:undeveloped_holding }
				}
				desc = aeluran_diarchy.1031.desc
			}
			triggered_desc = {
				trigger = {
					NOT = {
						new_temple_available_for_diarch = { COUNTY = scope:undeveloped_holding.county PROVINCE = scope:undeveloped_holding }
					}
				}
				desc = aeluran_diarchy.1031.desc.no_longer_owned
			}
		}
	}

	trigger = {
		scope:aeluran_sister_advisor = {
			is_alive = yes
		}
		# The temple is still being built: wait for it (on_trigger_fail)
		NOT = { scope:undeveloped_holding ?= { has_ongoing_construction = yes } }
	}
	on_trigger_fail = {
		if = {
			limit = {
				scope:aeluran_sister_advisor ?= { is_alive = yes }
				scope:undeveloped_holding ?= { has_ongoing_construction = yes }
			}
			trigger_event = {
				id = aeluran_diarchy.1031
				days = 365
			}
		}
	}

	theme = aeluran_matriarch

	right_portrait = {
		character = scope:aeluran_sister_advisor
		animation = personality_bold
		triggered_animation = {
			trigger = {
				NOT = {
					new_temple_available_for_diarch = { COUNTY = scope:undeveloped_holding.county PROVINCE = scope:undeveloped_holding }
				}
			}
			animation = disapproval
		}
	}

	immediate = {
		
	}

	option = {
		name = aeluran_diarchy.1031.ok

		trigger = {
			new_temple_available_for_diarch = { COUNTY = scope:undeveloped_holding.county PROVINCE = scope:undeveloped_holding }
		}

		# The temple is the ruler's and the lease hands it to the Sister: no title changes hands (Beta_v2.0.4)
		aeluran_matter_of_land_settled_effect = yes
	}

	option = {
		name = aeluran_diarchy.1031.no_longer_owned
		trigger = {
			NOT = {
				new_temple_available_for_diarch = { COUNTY = scope:undeveloped_holding.county PROVINCE = scope:undeveloped_holding }
			}
		}
		trigger_event = aeluran_diarchy.1033
	}
}
